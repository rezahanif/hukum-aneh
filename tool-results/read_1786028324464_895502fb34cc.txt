   114→   114→# ==========================================================================
   115→   115→# FAMILY A: HIERARCHICAL STATUTE PARSER
   116→   116→# ==========================================================================
   117→   117→
   118→   118→class FamilyAParser:
   119→   119→    """
   120→   120→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
   121→   121→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
   122→   122→    """
   123→   123→
   124→   124→    # Heading patterns (order matters: higher levels first)
   125→   125→    RE_BAB = re.compile(r'^\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
   126→   126→    RE_BAGIAN = re.compile(r'^\s*BAGIAN\s+(\w+)', re.IGNORECASE)
   127→   127→    RE_PARAGRAF = re.compile(r'^\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
   128→   128→    RE_PASAL = re.compile(r'^\s*pasal\s+([IVXLCDM]+|\d+(?:\s+\d+)*)', re.IGNORECASE)
   129→   129→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
   130→   130→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
   131→   131→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
   132→   132→
   133→   133→    # Quoted amendment detection
   134→   134→    RE_AMENDMENT_TRIGGER = re.compile(
   135→   135→        r'(?:diubah|diganti|dihapus)\s+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
   136→   136→        re.IGNORECASE
   137→   137→    )
   138→   138→    RE_QUOTE_START = re.compile(r'^\s*["“]')
   139→   139→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
   140→   140→
   141→   141→    # Penutup (closing section) marker
   142→   142→    RE_PENUTUP = re.compile(r'^(?:KETENTUAN\s+PENUTUP|PENUTUP|penutup)\b')
   143→   143→    # v3: Penjelasan (elucidation) marker — can appear after penutup or directly after last pasal
   144→   144→    RE_PENJELASAN = re.compile(r'^(?:PENJELASAN|penjelasan)\b')
   145→   145→
   146→   146→    def __init__(self, lines, doc_type, nomor, year, issuer):
   147→   147→        self.lines = lines
   148→   148→        self.doc_type = doc_type
   149→   149→        self.nomor = nomor
   150→   150→        self.year = year
   151→   151→        self.issuer = issuer
   152→   152→        self.chunks = []
   153→   153→
   154→   154→        # Heading stack
   155→   155→        self.current_bab = None
   156→   156→        self.current_bab_label = None
   157→   157→        self.current_bagian = None
   158→   158→        self.current_bagian_label = None
   159→   159→        self.current_paragraf = None
   160→   160→        self.current_paragraf_label = None
   161→   161→        self.current_pasal = None
   162→   162→        self.current_ayat = None
   163→   163→
   164→   164→        # State
   165→   165→        self.in_preamble = True
   166→   166→        self.in_penutup = False
   167→   167→        self.in_penjelasan = False  # v3: track elucidation section
   168→   168→        self.in_quoted_amendment = False
   169→   169→        self.quoted_amendment_source = None  # tracks which pasal is being amended
   170→   170→        self.current_ayat_lines = []
   171→   171→        self.current_pasal_lines = []
   172→   172→        self.quoted_amendment_lines = []  # accumulate quoted text for chunking
   173→   173→        self._penjelasan_lines = []  # v3: accumulate Penjelasan text
   174→   174→        self._penjelasan_pasal = None
   175→   175→
   176→   176→    def parse(self) -> List[Dict]:
   177→   177→        """Main parse loop. Returns list of chunks."""
   178→   178→        for i, line in enumerate(self.lines):
   179→   179→            text = line["text"].strip()
   180→   180→            if not text:
   181→   181→                continue
   182→   182→
   183→   183→            text_upper = text.upper()
   184→   184→
   185→   185→            # Detect preamble end / body start
   186→   186→            if self.in_preamble:
   187→   187→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   188→   188→                    self.in_preamble = False
   189→   189→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
   190→   190→                    self.in_preamble = False
   191→   191→                    continue
   192→   192→                else:
   193→   193→                    continue
   194→   194→
   195→   195→            # Detect penutup
   196→   196→            if self.RE_PENUTUP.match(text):
   197→   197→                self._flush_ayat()
   198→   198→                self._flush_pasal()
   199→   199→                self._flush_quoted_amendment()
   200→   200→                self.in_penutup = True
   201→   201→                continue
   202→   202→
   203→   203→            # v3: Detect PENJELASAN even without penutup (many docs go straight to penjelasan)
   204→   204→            if self.RE_PENJELASAN.match(text) and not self.in_penjelasan:
   205→   205→                self._flush_ayat()
   206→   206→                self._flush_pasal()
   207→   207→                self._flush_quoted_amendment()
   208→   208→                self.in_penutup = True  # treat as post-operative
   209→   209→                self.in_penjelasan = True
   210→   210→                self._penjelasan_pasal = None
   211→   211→                continue
   212→   212→
   213→   213→            if self.in_penutup:
   214→   214→                # v3: Detect signature block and Penjelasan as hard boundaries
   215→   215→                text_lower = text.lower()
   216→   216→                # Signature block: lines like "Ditetapkan di Jakarta", "pada tanggal", or ttd/
   217→   217→                # Also detect by very short lines after penutup start
   218→   218→                if (re.search(r'ttd|disahkan|ditetapkan', text_lower) or
   219→   219→                    (len(text) < 15 and text_upper == text and len(text) > 2)):
   220→   220→                    self._flush_penjelasan()  # flush any accumulated penjelasan
   221→   221→                    continue
   222→   222→                # PENJELASAN section header
   223→   223→                if text_upper == "PENJELASAN" or text_upper.startswith("PENJELASAN "):
   224→   224→                    self._flush_penjelasan()
   225→   225→                    self.in_penjelasan = True
   226→   226→                    self._penjelasan_pasal = None
   227→   227→                    continue
   228→   228→                # Inside Penjelasan: track pasal references for IDs
   229→   229→                if self.in_penjelasan:
   230→   230→                    m_pj = re.match(r'^\s*Pasal\s+(\d+[a-zA-Z]?)', text, re.IGNORECASE)
   231→   231→                    if m_pj:
   232→   232→                        self._flush_penjelasan()
   233→   233→                        self._penjelasan_pasal = m_pj.group(1)
   234→   234→                    else:
   235→   235→                        self._penjelasan_lines.append(text)
   236→   236→                continue
   237→   237→
   238→   238→            # Check for quoted amendment TRIGGER ("Pasal X diubah sehingga berbunyi:")
   239→   239→            if not self.in_quoted_amendment:
   240→   240→                if self.RE_AMENDMENT_TRIGGER.search(text):
   241→   241→                    # Extract which pasal is being amended
   242→   242→                    m_pasal = re.search(r'Pasal\s+(\d+|\w+)\s+(?:ayat\s*\(\d+\)\s+)?(?:diubah|diganti|dihapus)',
   243→   243→                                         text, re.IGNORECASE)
   244→   244→                    source_pasal = m_pasal.group(1) if m_pasal else "?"
   245→   245→                    self.in_quoted_amendment = True
   246→   246→                    self.quoted_amendment_source = source_pasal
   247→   247→                    self._flush_ayat()
   248→   248→                    # Store the trigger sentence as context for the quoted chunk
   249→   249→                    self.quoted_amendment_lines = [text]
   250→   250→                    continue
   251→   251→
   252→   252→            # Inside quoted amendment: accumulate text
   253→   253→            if self.in_quoted_amendment:
   254→   254→                self.quoted_amendment_lines.append(text)
   255→   255→                if self.RE_QUOTE_END.search(text):
   256→   256→                    self._flush_quoted_amendment()
   257→   257→                # Also check for end of quoted section without closing quote:
   258→   258→                # if we hit a new structural marker, the quote ended implicitly
   259→   259→                if (self.RE_BAB.match(text) or self.RE_PASAL.match(text) or
   260→   260→                    self.RE_PENUTUP.match(text)):
   261→   261→                    self._flush_quoted_amendment()
   262→   262→                    # Don't skip the structural marker - let it be processed below
   263→   263→                    self.in_quoted_amendment = False
   264→   264→                elif self.RE_QUOTE_END.search(text):
   265→   265→                    pass  # already flushed above
   266→   266→                else:
   267→   267→                    continue
   268→   268→
   269→   269→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   270→   270→            m = self.RE_BAB.match(text)
   271→   271→            if m:
   272→   272→                self._flush_ayat()
   273→   273→                self._flush_pasal()
   274→   274→                self.current_bab = m.group(1)
   275→   275→                label_match = re.match(r'^\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   276→   276→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   277→   277→                # Fix #2: if label is empty or just repeats "BAB X", set to None
   278→   278→                if raw_label and raw_label.upper() != f"BAB {m.group(1).upper()}":
   279→   279→                    self.current_bab_label = raw_label
   280→   280→                else:
   281→   281→                    self.current_bab_label = None
   282→   282→                self.current_bagian = None
   283→   283→                self.current_bagian_label = None
   284→   284→                self.current_paragraf = None
   285→   285→                self.current_paragraf_label = None
   286→   286→                # v4: reset pasal when BAB changes to prevent heading-as-chunk
   287→   287→                self.current_pasal = None
   288→   288→                continue
   289→   289→
   290→   290→            m = self.RE_BAGIAN.match(text)
   291→   291→            if m:
   292→   292→                self._flush_ayat()
   293→   293→                self._flush_pasal()
   294→   294→                self.current_bagian = m.group(1)
   295→   295→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   296→   296→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   297→   297→                if raw_label:
   298→   298→                    self.current_bagian_label = raw_label
   299→   299→                else:
   300→   300→                    self.current_bagian_label = None
   301→   301→                self.current_paragraf = None
   302→   302→                self.current_paragraf_label = None
   303→   303→                continue
   304→   304→
   305→   305→            m = self.RE_PARAGRAF.match(text)
   306→   306→            if m:
   307→   307→                self._flush_ayat()
   308→   308→                self._flush_pasal()
   309→   309→                self.current_paragraf = m.group(1)
   310→   310→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   311→   311→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   312→   312→                if raw_label:
   313→   313→                    self.current_paragraf_label = raw_label
   314→   314→                else:
   315→   315→                    self.current_paragraf_label = None
   316→   316→                continue
   317→   317→
   318→   318→            m = self.RE_PASAL.match(text)
   319→   319→            if m:
   320→   320→                self._flush_ayat()
   321→   321→                self._flush_pasal()
   322→   322→                raw_num = m.group(1)
   323→   323→                # v4: normalize space-separated digits ("1 1" -> "11", "1 10" -> "110")
   324→   324→                self.current_pasal = re.sub(r'\s+', '', raw_num) if re.match(r'^\d', raw_num) else raw_num
   325→   325→                self.current_ayat = None
   326→   326→                # v4: capture any remaining text on the same line as pasal heading
   327→   327→                remaining = text[m.end():].strip().lstrip('.').strip()
   328→   328→                if remaining and len(remaining) > 5:
   329→   329→                    self.current_pasal_lines.append(remaining)
   330→   330→                continue
   331→   331→
   332→   332→            m = self.RE_AYAT.match(text)
   333→   333→            if m and self.current_pasal:
   334→   334→                self._flush_ayat()
   335→   335→                self.current_ayat = m.group(1)
   336→   336→                ayat_text = self.RE_AYAT.sub('', text).strip()
   337→   337→                if ayat_text:
   338→   338→                    self.current_ayat_lines.append(ayat_text)
   339→   339→                continue
   340→   340→
   341→   341→            # If we're inside a pasal but no ayat detected, accumulate lines
   342→   342→            if self.current_pasal:
   343→   343→                if self.current_ayat:
   344→   344→                    self.current_ayat_lines.append(text)
   345→   345→                else:
   346→   346→                    self.current_pasal_lines.append(text)
   347→   347→
   348→   348→        # Flush remaining
   349→   349→        self._flush_ayat()
   350→   350→        self._flush_pasal()
   351→   351→        self._flush_quoted_amendment()
   352→   352→
   353→   353→        return self.chunks
   354→   354→
   355→   355→    def _flush_ayat(self):
   356→   356→        """Flush accumulated ayat lines as a chunk."""
   357→   357→        if not self.current_ayat or not self.current_ayat_lines:
   358→   358→            self.current_ayat_lines = []
   359→   359→            return
   360→   360→
   361→   361→        text = ' '.join(self.current_ayat_lines).strip()
   362→   362→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   363→   363→            self.current_ayat_lines = []
   364→   364→            return
   365→   365→
   366→   366→        headings = _make_bab_headings(
   367→   367→            self.current_bab, self.current_bab_label,
   368→   368→            self.current_bagian, self.current_bagian_label,
   369→   369→            self.current_paragraf, self.current_paragraf_label,
   370→   370→            pasal_str=f"Pasal {self.current_pasal}",
   371→   371→            ayat_str=f"Ayat ({self.current_ayat})"
   372→   372→        )
   373→   373→
   374→   374→        path = build_path(headings)
   375→   375→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   376→   376→                            pasal=self.current_pasal, ayat=self.current_ayat)
   377→   377→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   378→   378→
   379→   379→        self.chunks.append({
   380→   380→            "id": chunk_id,
   381→   381→            "text": text,
   382→   382→            "metadata": {
   383→   383→                "doc_type": self.doc_type,
   384→   384→                "issuer": self.issuer,
   385→   385→                "year": self.year,
   386→   386→                "bab": self.current_bab or None,
   387→   387→                "pasal": self.current_pasal,
   388→   388→                "ayat": self.current_ayat,
   389→   389→                "parent": parent_id,
   390→   390→                "path": path,
   391→   391→                "status": "active",
   392→   392→            }
   393→   393→        })
   394→   394→        self.current_ayat_lines = []
   395→   395→
   396→   396→    def _flush_pasal(self):
   397→   397→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   398→   398→        if not self.current_pasal or not self.current_pasal_lines:
   399→   399→            self.current_pasal_lines = []
   400→   400→            return
   401→   401→
   402→   402→        text = ' '.join(self.current_pasal_lines).strip()
   403→   403→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   404→   404→            self.current_pasal_lines = []
   405→   405→            return
   406→   406→
   407→   407→        headings = _make_bab_headings(
   408→   408→            self.current_bab, self.current_bab_label,
   409→   409→            self.current_bagian, self.current_bagian_label,
   410→   410→            self.current_paragraf, self.current_paragraf_label,

   822→   822→def chunk_pdf(pdf_path: str, directory: str, family_map: Dict) -> List[Dict]:
   823→   823→    """
   824→   824→    Main entry point. Extract + parse a PDF into chunks.
   825→   825→    Returns list of chunk dicts with id, text, metadata.
   826→   826→    """
   827→   827→    dir_config = family_map.get(directory, {})
   828→   828→    family = dir_config.get("family", "A")
   829→   829→    default_doc_type = dir_config.get("doc_type", "Unknown")
   830→   830→    default_issuer = dir_config.get("issuer", "Unknown")

   832→   832→    try:
   833→   833→        lines, meta = extract_clean(pdf_path)
   834→   834→    except Exception as e:
   835→   835→        return [{"id": f"ERROR:{directory}", "text": str(e), "metadata": {"error": True}}]
   836→   836→
   837→   837→    if not lines:
   838→   838→        return [{"id": f"EMPTY:{directory}", "text": "No extractable text (possibly scanned PDF)", "metadata": {"error": True, "needs_ocr": True}}]
   839→   839→
   840→   840→    # v4 fix: Detect doc_type from title block FIRST, directory as fallback.
   841→   841→    # Use page-1-only RAW lines (pre-boilerplate) since boilerplate stripping
   842→   842→    # removes the actual document title (e.g. "PERATURAN PRESIDEN").
   843→   843→    # Only page 1 to avoid body citations polluting the detection.
   844→   844→    page1_raw = [l["text"].strip() for l in lines if l.get("page") == 1]
   845→   845→    raw_title_block = " ".join(page1_raw[:15]) if page1_raw else ""
   846→   846→    detected_type, detected_issuer = detect_doc_type_from_title(raw_title_block)
   847→   847→    
   848→   848→    # If page 1 was empty (image) or detection failed, try cleaned lines
   849→   849→    if detected_type == "Unknown":
   850→   850→        clean_title = extract_title_block(lines, max_first_n=5)
