    36→    36→        # Court rulings use nomor_perkara

    76→    76→    # Heading patterns (order matters: higher levels first)

    99→    99→        self.nomor = nomor

   102→   102→        self.chunks = []
   103→   103→

   116→   116→        self.in_penutup = False
   117→   117→        self.in_quoted_amendment = False

   122→   122→    def parse(self) -> List[Dict]:
   123→   123→        """Main parse loop. Returns list of chunks."""
   124→   124→        for i, line in enumerate(self.lines):
   125→   125→            text = line["text"].strip()
   126→   126→            if not text:
   127→   127→                continue
   128→   128→
   129→   129→            text_upper = text.upper()
   130→   130→
   131→   131→            # Detect preamble end / body start
   132→   132→            if self.in_preamble:
   133→   133→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   134→   134→                    self.in_preamble = False
   135→   135→                elif text_upper.startswith("MENETAPKAN"):
   136→   136→                    self.in_preamble = False
   137→   137→                    continue  # skip "Menetapkan" line itself
   138→   138→                else:
   139→   139→                    continue  # still in preamble
   140→   140→
   141→   141→            # Detect penutup
   142→   142→            if self.RE_PENUTUP.match(text):
   143→   143→                self.in_penutup = True
   144→   144→                # Flush any pending ayat
   145→   145→                self._flush_ayat()
   146→   146→                self._flush_pasal()
   147→   147→                continue
   148→   148→
   149→   149→            if self.in_penutup:
   150→   150→                continue  # skip penutup
   151→   151→
   152→   152→            # Check for quoted amendment
   153→   153→            if not self.in_quoted_amendment:
   154→   154→                if self.RE_AMENDMENT_TRIGGER.search(text):
   155→   155→                    self.in_quoted_amendment = True
   156→   156→                    self.quoted_amendment_note = text[:100]
   157→   157→                    self._flush_ayat()
   158→   158→                    continue
   159→   159→                if self.RE_QUOTE_START.match(text):
   160→   160→                    # Could be inline quote
   161→   161→                    self.in_quoted_amendment = True
   162→   162→                    self.quoted_amendment_note = "Quoted passage (inline)"
   163→   163→
   164→   164→            if self.in_quoted_amendment:
   165→   165→                if self.RE_QUOTE_END.search(text):
   166→   166→                    self.in_quoted_amendment = False
   167→   167→                    self.quoted_amendment_note = None
   168→   168→                continue  # skip all quoted text
   169→   169→
   170→   170→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   171→   171→            m = self.RE_BAB.match(text)
   172→   172→            if m:
   173→   173→                self._flush_ayat()
   174→   174→                self._flush_pasal()
   175→   175→                self.current_bab = m.group(1)
   176→   176→                # Get label text after BAB X
   177→   177→                label_match = re.match(r'^\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   178→   178→                self.current_bab_label = label_match.group(1).strip().rstrip('.') if label_match else f"BAB {m.group(1)}"
   179→   179→                self.current_bagian = None
   180→   180→                self.current_bagian_label = None
   181→   181→                self.current_paragraf = None
   182→   182→                self.current_paragraf_label = None
   183→   183→                continue
   184→   184→
   185→   185→            m = self.RE_BAGIAN.match(text)
   186→   186→            if m:
   187→   187→                self._flush_ayat()
   188→   188→                self._flush_pasal()
   189→   189→                self.current_bagian = m.group(1)
   190→   190→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   191→   191→                self.current_bagian_label = label_match.group(1).strip().rstrip('.') if label_match else f"Bagian {m.group(1)}"
   192→   192→                self.current_paragraf = None
   193→   193→                self.current_paragraf_label = None
   194→   194→                continue
   195→   195→
   196→   196→            m = self.RE_PARAGRAF.match(text)
   197→   197→            if m:
   198→   198→                self._flush_ayat()
   199→   199→                self._flush_pasal()
   200→   200→                self.current_paragraf = m.group(1)
   201→   201→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   202→   202→                self.current_paragraf_label = label_match.group(1).strip().rstrip('.') if label_match else f"Paragraf {m.group(1)}"
   203→   203→                continue
   204→   204→
   205→   205→            m = self.RE_PASAL.match(text)
   206→   206→            if m:
   207→   207→                self._flush_ayat()
   208→   208→                self._flush_pasal()
   209→   209→                self.current_pasal = m.group(1)
   210→   210→                self.current_ayat = None
   211→   211→                continue
   212→   212→
   213→   213→            m = self.RE_AYAT.match(text)
   214→   214→            if m and self.current_pasal:
   215→   215→                self._flush_ayat()
   216→   216→                self.current_ayat = m.group(1)
   217→   217→                # The rest of the line after (N) is the start of ayat text
   218→   218→                ayat_text = self.RE_AYAT.sub('', text).strip()
   219→   219→                if ayat_text:
   220→   220→                    self.current_ayat_lines.append(ayat_text)
   221→   221→                continue
   222→   222→
   223→   223→            # If we're inside a pasal but no ayat detected, accumulate lines
   224→   224→            if self.current_pasal:
   225→   225→                if self.current_ayat:
   226→   226→                    self.current_ayat_lines.append(text)
   227→   227→                else:
   228→   228→                    self.current_pasal_lines.append(text)
   229→   229→
   230→   230→        # Flush remaining
   231→   231→        self._flush_ayat()
   232→   232→        self._flush_pasal()
   233→   233→
   234→   234→        return self.chunks

   242→   242→        text = ' '.join(self.current_ayat_lines).strip()

   244→   244→            self.current_ayat_lines = []

   246→   246→
   247→   247→        headings = [

   274→   274→                "parent": parent_id,
   275→   275→                "path": path,
   276→   276→                "status": "active",
   277→   277→            }
   278→   278→        })
   279→   279→        self.current_ayat_lines = []
   280→   280→

   283→   283→        if not self.current_pasal or not self.current_pasal_lines:

   287→   287→        text = ' '.join(self.current_pasal_lines).strip()

   289→   289→            self.current_pasal_lines = []

   291→   291→
   292→   292→        headings = [
   293→   293→            f"BAB {self.current_bab}" if self.current_bab else None,

   312→   312→                "bab": self.current_bab or None,

   327→   327→class FamilyBParser:
   328→   328→    """
   329→   329→    Parses decree/decision documents: Menimbang > Mengingat > MEMUTUSKAN/MENETAPKAN > Diktum.
   330→   330→    Handles: Keppres, Inpres, Tap_MPR, Kepmen.
   331→   331→    """
   332→   332→
   333→   333→    RE_MENIMBANG = re.compile(r'^\s*menimbang\s*:?\s*$', re.IGNORECASE)
   334→   334→    RE_MENGINGAT = re.compile(r'^\s*mengingat\s*:?\s*$', re.IGNORECASE)
   335→   335→    RE_MEMUTUSKAN = re.compile(r'^\s*memutuskan\s*:?\s*$', re.IGNORECASE)
   336→   336→    RE_MENETAPKAN = re.compile(r'^\s*menetapkan\s*:?\s*$', re.IGNORECASE)
   337→   337→    RE_MENINSTRUKSIKAN = re.compile(r'^\s*menginstruksikan\s*:?\s*$', re.IGNORECASE)
   338→   338→    RE_ITEM_LETTER = re.compile(r'^\s*([a-z])\s*\.\s+')
   339→   339→    RE_ITEM_NUMBER = re.compile(r'^\s*(\d+)\s*\.\s+')
   340→   340→
   341→   341→    def __init__(self, lines, doc_type, nomor, year, issuer):
   342→   342→        self.lines = lines
   343→   343→        self.doc_type = doc_type
   344→   344→        self.nomor = nomor
   345→   345→        self.year = year
   346→   346→        self.issuer = issuer
   347→   347→        self.chunks = []
   348→   348→
   349→   349→        self.section = "pre"  # pre, menimbang, mengingat, keputusan, done
   350→   350→        self.current_item = None
   351→   351→        self.current_item_lines = []
   352→   352→        self.preamble_lines = []
   353→   353→        self.decision_keyword = None  # MEMUTUSKAN, MENETAPKAN, etc.
   354→   354→
   355→   355→    def parse(self) -> List[Dict]:
   356→   356→        for line in self.lines:
   357→   357→            text = line["text"].strip()
   358→   358→            if not text:
   359→   359→                continue
   360→   360→
   361→   361→            text_upper = text.upper()
   362→   362→
   363→   363→            # Section transitions
   364→   364→            if self.RE_MENIMBANG.match(text):
   365→   365→                self._flush_preamble()
   366→   366→                self.section = "menimbang"
   367→   367→                self.current_item = None
   368→   368→                continue
   369→   369→
   370→   370→            if self.RE_MENGINGAT.match(text):
   371→   371→                self._flush_preamble()
   372→   372→                self.section = "mengingat"
   373→   373→                self.current_item = None
   374→   374→                continue
   375→   375→
   376→   376→            if self.RE_MEMUTUSKAN.match(text):
   377→   377→                self._flush_item()
   378→   378→                self.section = "keputusan"
   379→   379→                self.decision_keyword = "MEMUTUSKAN"
   380→   380→                self.current_item = None
   381→   381→                continue
   382→   382→
   383→   383→            if self.RE_MENETAPKAN.match(text):
   384→   384→                self._flush_item()
   385→   385→                self.section = "keputusan"
   386→   386→                self.decision_keyword = "MENETAPKAN"
   387→   387→                self.current_item = None
   388→   388→                continue
   389→   389→
   390→   390→            if self.RE_MENINSTRUKSIKAN.match(text):
   391→   391→                self._flush_item()
   392→   392→                self.section = "keputusan"
   393→   393→                self.decision_keyword = "MENGINSTRUKSIKAN"
   394→   394→                self.current_item = None
   395→   395→                continue
   396→   396→
   397→   397→            # Check for judge signature block (end of substantive content)
   398→   398→            text_lower = text.lower()
   399→   399→            if "ttd" in text_lower and len(text) < 20:
   400→   400→                self._flush_item()
   401→   401→                self.section = "done"
   402→   402→                continue
   403→   403→
   404→   404→            if self.section == "pre":
   405→   405→                continue  # skip title block
   406→   406→
   407→   407→            if self.section == "done":
   408→   408→                continue
   409→   409→
   410→   410→            # Check for new item in keputusan section
   411→   411→            if self.section == "keputusan":
   412→   412→                m_letter = self.RE_ITEM_LETTER.match(text)
   413→   413→                m_number = self.RE_ITEM_NUMBER.match(text)
   414→   414→
   415→   415→                if m_letter or m_number:
   416→   416→                    self._flush_item()
   417→   417→                    self.current_item = m_letter.group(1) if m_letter else m_number.group(1)
   418→   418→                    # Get text after the item marker
   419→   419→                    m = m_letter or m_number
   420→   420→                    remaining = text[m.end():].strip()
   421→   421→                    if remaining:
   422→   422→                        self.current_item_lines.append(remaining)
   423→   423→                    continue
   424→   424→
   425→   425→                # Sub-items: continue accumulating under current item
   426→   426→                if self.current_item:
   427→   427→                    self.current_item_lines.append(text)
   428→   428→                continue
   429→   429→
   430→   430→            # menimbang/mengingat: accumulate as preamble (skip for chunking or tag separately)
   431→   431→            # Per the plan, optionally chunk these with section tag
   432→   432→
   433→   433→        self._flush_item()
   434→   434→        return self.chunks

   436→   436→    def _flush_item(self):

   476→   476→class FamilyCParser:
   477→   477→    """
   478→   478→    Parses court rulings: Duduk Perkara > Menimbang > MENGADILI/MENETAPKAN > Amar.
   479→   479→    Handles: Putusan MK, Putusan MA.
   480→   480→    """
   481→   481→
   482→   482→    RE_MENGADILI = re.compile(r'^\s*mengadili\s*:?\s*$', re.IGNORECASE)
   483→   483→    RE_MENETAPKAN = re.compile(r'^\s*menetapkan\s*:?\s*$', re.IGNORECASE)
   484→   484→    RE_AMAR = re.compile(r'^\s*(\d+)\s*\.\s+')
   485→   485→    RE_PUTUSAN = re.compile(r'^\s*putusan\s*:?\s*$', re.IGNORECASE)
   486→   486→
   487→   487→    def __init__(self, lines, doc_type, nomor, year, issuer, nomor_perkara=None):
   488→   488→        self.lines = lines
   489→   489→        self.doc_type = doc_type
   490→   490→        self.nomor = nomor
   491→   491→        self.year = year
   492→   492→        self.issuer = issuer
   493→   493→        self.nomor_perkara = nomor_perkara
   494→   494→        self.chunks = []
   495→   495→
   496→   496→        self.section = "pre"  # pre, menimbang, mengadili, amar, done
   497→   497→        self.current_amar = None
   498→   498→        self.current_amar_lines = []
   499→   499→        self.pertimbangan_lines = []
   500→   500→
   501→   501→    def parse(self) -> List[Dict]:
   502→   502→        for line in self.lines:
   503→   503→            text = line["text"].strip()
   504→   504→            if not text:
   505→   505→                continue
   506→   506→
   507→   507→            text_upper = text.upper()
   508→   508→
   509→   509→            if self.RE_MENGADILI.match(text):
   510→   510→                self._flush_amar()
   511→   511→                self.section = "amar"
   512→   512→                self.decision_keyword = "MENGADILI"
   513→   513→                continue
   514→   514→
   515→   515→            if self.RE_MENETAPKAN.match(text):
   516→   516→                self._flush_amar()
   517→   517→                self.section = "amar"
   518→   518→                self.decision_keyword = "MENETAPKAN"
   519→   519→                continue
   520→   520→
   521→   521→            if self.section == "pre":
   522→   522→                continue
   523→   523→
   524→   524→            if self.section == "done":
   525→   525→                continue
   526→   526→
   527→   527→            if self.section == "amar":
   528→   528→                m = self.RE_AMAR.match(text)
   529→   529→                if m:
   530→   530→                    self._flush_amar()
   531→   531→                    self.current_amar = m.group(1)
   532→   532→                    remaining = text[m.end():].strip()
   533→   533→                    if remaining:
   534→   534→                        self.current_amar_lines.append(remaining)
   535→   535→                    continue
   536→   536→
   537→   537→                if self.current_amar:
   538→   538→                    self.current_amar_lines.append(text)
   539→   539→
   540→   540→                # Detect end: signature block
   541→   541→                if re.search(r'\bttd\b|\bHAKIM\b|\bKETUA\b', text_upper) and len(text) < 30:
   542→   542→                    self._flush_amar()
   543→   543→                    self.section = "done"
   544→   544→
   545→   545→        self._flush_amar()
   546→   546→        return self.chunks
   547→   547→
   548→   548→    def _flush_amar(self):
   549→   549→        if not self.current_amar or not self.current_amar_lines:
   550→   550→            self.current_amar_lines = []

   552→   552→
   553→   553→        text = ' '.join(self.current_amar_lines).strip()
   554→   554→        if not text:
   555→   555→            self.current_amar_lines = []
   556→   556→            return
   557→   557→
   558→   558→        kw = getattr(self, 'decision_keyword', 'MENGADILI')
   559→   559→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   560→   560→                            nomor_perkara=self.nomor_perkara,
   561→   561→                            chunk_type="amar", chunk_num=self.current_amar)
   562→   562→
   563→   563→        self.chunks.append({
   564→   564→            "id": chunk_id,
   565→   565→            "text": text,
   566→   566→            "metadata": {
   567→   567→                "doc_type": self.doc_type,
   568→   568→                "issuer": self.issuer,
   569→   569→                "year": self.year,
   570→   570→                "nomor_perkara": self.nomor_perkara,
   571→   571→                "section": "amar",
   572→   572→                "decision_keyword": kw,
   573→   573→                "amar": self.current_amar,
   574→   574→                "parent": None,
   575→   575→                "path": f"{kw} > Amar {self.current_amar}",
   576→   576→                "status": "active",
   577→   577→            }
   578→   578→        })
   579→   579→        self.current_amar_lines = []

   586→   586→def extract_nomor_perkara(lines):
   587→   587→    """Extract case number (e.g. 96/PUU-XVI/2018) from court ruling lines."""
   588→   588→    for line in lines[:30]:
   589→   589→        m = re.search(r'(\d+/[A-Z]+-[A-Z]+/\d{4})', line["text"])
   590→   590→        if m:
   591→   591→            return m.group(1)
   592→   592→    return None

   595→   595→def chunk_pdf(pdf_path: str, directory: str, family_map: Dict) -> List[Dict]:
   596→   596→    """
   597→   597→    Main entry point. Extract + parse a PDF into chunks.
   598→   598→    Returns list of chunk dicts with id, text, metadata.
   599→   599→    """
   600→   600→    # Get family config
   601→   601→    dir_config = family_map.get(directory, {})
   602→   602→    family = dir_config.get("family", "A")
   603→   603→    default_doc_type = dir_config.get("doc_type", "Unknown")
   604→   604→    default_issuer = dir_config.get("issuer", "Unknown")
   605→   605→
   606→   606→    # Extract clean text
   607→   607→    try:
   608→   608→        lines, meta = extract_clean(pdf_path)
   609→   609→    except Exception as e:
   610→   610→        return [{"id": f"ERROR:{directory}", "text": str(e), "metadata": {"error": True}}]
   611→   611→
   612→   612→    if not lines:
   613→   613→        return [{"id": f"EMPTY:{directory}", "text": "No extractable text (possibly scanned PDF)", "metadata": {"error": True, "needs_ocr": True}}]
   614→   614→
   615→   615→    # Detect doc type from title block (override directory default)
   616→   616→    title_block = extract_title_block(lines)
   617→   617→    detected_type, detected_issuer = detect_doc_type_from_title(title_block)
   618→   618→    doc_type = detected_type if detected_type != "Unknown" else default_doc_type
   619→   619→    issuer = detected_issuer if detected_issuer != "Unknown" else default_issuer
   620→   620→    nomor = meta.get("nomor")
   621→   621→    year = meta.get("year")
   622→   622→
   623→   623→    # Parse based on family
   624→   624→    if family == "A":
   625→   625→        parser = FamilyAParser(lines, doc_type, nomor, year, issuer)
   626→   626→        chunks = parser.parse()
   627→   627→    elif family == "B":
   628→   628→        parser = FamilyBParser(lines, doc_type, nomor, year, issuer)
   629→   629→        chunks = parser.parse()
   630→   630→    elif family == "C":
   631→   631→        nomor_perkara = extract_nomor_perkara(lines)
   632→   632→        parser = FamilyCParser(lines, doc_type, nomor, year, issuer, nomor_perkara)
   633→   633→        chunks = parser.parse()
   634→   634→    else:
   635→   635→        chunks = [{"id": f"UNKNOWN_FAMILY:{directory}", "text": "", "metadata": {"error": True}}]
   636→   636→
   637→   637→    # Add extraction metadata to all chunks
   638→   638→    for chunk in chunks:
   639→   639→        chunk["metadata"]["source_file"] = meta["file"]
   640→   640→        chunk["metadata"]["directory"] = directory
   641→   641→        chunk["metadata"]["family"] = family
   642→   642→        if nomor:
   643→   643→            chunk["metadata"]["nomor"] = nomor
   644→   644→        if year:
   645→   645→            chunk["metadata"]["year"] = year
   646→   646→
   647→   647→    return chunks
