    42→    42→def build_id(doc_type, nomor, year, **parts):
    43→    43→    """Build chunk ID. Handles both hierarchical and flat structures."""
    44→    44→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    45→    45→        nomor_perkara = parts.get("nomor_perkara", nomor or "?")
    46→    46→        chunk_type = parts.get("chunk_type", "amar")

    49→    49→    elif parts.get("chunk_type") in ("diktum", "pertimbangan"):

    52→    52→        pasal = parts.get("pasal", "?")
    53→    53→        ayat = parts.get("ayat", "?")

    89→    89→        headings.append(bab_num_str)
    90→    90→    # Only add label if it's different from the number element and non-empty
    91→    91→    if bab_label and bab_label != bab_num_str:
    92→    92→        headings.append(bab_label)
    93→    93→    
    94→    94→    bagian_num_str = f"Bagian {bagian_num}" if bagian_num else None
    95→    95→    if bagian_num_str:
    96→    96→        headings.append(bagian_num_str)
    97→    97→    if bagian_label and bagian_label != bagian_num_str:
    98→    98→        headings.append(bagian_label)

   100→   100→    paragraf_num_str = f"Paragraf {paragraf_num}" if paragraf_num else None
   101→   101→    if paragraf_num_str:
   102→   102→        headings.append(paragraf_num_str)
   103→   103→    if paragraf_label and paragraf_label != paragraf_num_str:
   104→   104→        headings.append(paragraf_label)
   105→   105→    
   106→   106→    if pasal_str:
   107→   107→        headings.append(pasal_str)
   108→   108→    if ayat_str:

   114→   114→# ==========================================================================
   115→   115→# FAMILY A: HIERARCHICAL STATUTE PARSER
   116→   116→# ==========================================================================
   117→   117→
   118→   118→class FamilyAParser:
   119→   119→    """
   120→   120→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
   121→   121→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
   122→   122→    """
   123→   123→
   124→   124→    # Heading patterns (order matters: higher levels first)
   125→   125→    RE_BAB = re.compile(r'^\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
   126→   126→    RE_BAGIAN = re.compile(r'^\s*BAGIAN\s+(\w+)', re.IGNORECASE)
   127→   127→    RE_PARAGRAF = re.compile(r'^\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
   128→   128→    # v5: capture alpha suffixes (68A, 68B, ...) AND require heading-like ending
   129→   129→    # to prevent body-text references like "Pasal 2 diangkat" from matching.
   130→   130→    RE_PASAL = re.compile(r'^\s*pasal\s+([IVXLCDM]+|\d+(?:\s+\d+)*[a-zA-Z]?)\s*[.}\s]*$', re.IGNORECASE)
   131→   131→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
   132→   132→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
   133→   133→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
   134→   134→
   135→   135→    # Quoted amendment detection
   136→   136→    RE_AMENDMENT_TRIGGER = re.compile(
   137→   137→        r'(?:diubah|diganti|dihapus)\s+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
   138→   138→        re.IGNORECASE
   139→   139→    )
   140→   140→    RE_QUOTE_START = re.compile(r'^\s*["“]')
   141→   141→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
   142→   142→
   143→   143→    # Penutup (closing section) marker
   144→   144→    RE_PENUTUP = re.compile(r'^(?:KETENTUAN\s+PENUTUP|PENUTUP|penutup)\b')
   145→   145→    # v3: Penjelasan (elucidation) marker — can appear after penutup or directly after last pasal
   146→   146→    RE_PENJELASAN = re.compile(r'^(?:PENJELASAN|penjelasan)\b')
   147→   147→
   148→   148→    def __init__(self, lines, doc_type, nomor, year, issuer):
   149→   149→        self.lines = lines
   150→   150→        self.doc_type = doc_type
   151→   151→        self.nomor = nomor
   152→   152→        self.year = year
   153→   153→        self.issuer = issuer
   154→   154→        self.chunks = []
   155→   155→
   156→   156→        # Heading stack
   157→   157→        self.current_bab = None
   158→   158→        self.current_bab_label = None
   159→   159→        self.current_bagian = None
   160→   160→        self.current_bagian_label = None
   161→   161→        self.current_paragraf = None
   162→   162→        self.current_paragraf_label = None
   163→   163→        self.current_pasal = None
   164→   164→        self.current_ayat = None
   165→   165→
   166→   166→        # State
   167→   167→        self.in_preamble = True
   168→   168→        self.in_penutup = False
   169→   169→        self.in_penjelasan = False  # v3: track elucidation section
   170→   170→        self.in_quoted_amendment = False
   171→   171→        self.quoted_amendment_source = None  # tracks which pasal is being amended
   172→   172→        self.current_ayat_lines = []
   173→   173→        self.current_pasal_lines = []
   174→   174→        self.quoted_amendment_lines = []  # accumulate quoted text for chunking
   175→   175→        self._penjelasan_lines = []  # v3: accumulate Penjelasan text
   176→   176→        self._penjelasan_pasal = None
   177→   177→
   178→   178→    def parse(self) -> List[Dict]:
   179→   179→        """Main parse loop. Returns list of chunks."""
   180→   180→        for i, line in enumerate(self.lines):
   181→   181→            text = line["text"].strip()
   182→   182→            if not text:
   183→   183→                continue
   184→   184→
   185→   185→            text_upper = text.upper()
   186→   186→
   187→   187→            # Detect preamble end / body start
   188→   188→            if self.in_preamble:
   189→   189→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   190→   190→                    self.in_preamble = False
   191→   191→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
   192→   192→                    self.in_preamble = False
   193→   193→                    continue
   194→   194→                else:
   195→   195→                    continue
   196→   196→
   197→   197→            # Detect penutup
   198→   198→            if self.RE_PENUTUP.match(text):
   199→   199→                self._flush_ayat()
   200→   200→                self._flush_pasal()
   201→   201→                self._flush_quoted_amendment()
   202→   202→                self.in_penutup = True
   203→   203→                continue
   204→   204→
   205→   205→            # v3: Detect PENJELASAN even without penutup (many docs go straight to penjelasan)
   206→   206→            if self.RE_PENJELASAN.match(text) and not self.in_penjelasan:
   207→   207→                self._flush_ayat()
   208→   208→                self._flush_pasal()
   209→   209→                self._flush_quoted_amendment()
   210→   210→                self.in_penutup = True  # treat as post-operative
   211→   211→                self.in_penjelasan = True
   212→   212→                self._penjelasan_pasal = None
   213→   213→                continue
   214→   214→
   215→   215→            if self.in_penutup:
   216→   216→                # v3: Detect signature block and Penjelasan as hard boundaries
   217→   217→                text_lower = text.lower()
   218→   218→                # Signature block: lines like "Ditetapkan di Jakarta", "pada tanggal", or ttd/
   219→   219→                # Also detect by very short lines after penutup start
   220→   220→                if (re.search(r'ttd|disahkan|ditetapkan', text_lower) or
   221→   221→                    (len(text) < 15 and text_upper == text and len(text) > 2)):
   222→   222→                    self._flush_penjelasan()  # flush any accumulated penjelasan
   223→   223→                    continue
   224→   224→                # PENJELASAN section header
   225→   225→                if text_upper == "PENJELASAN" or text_upper.startswith("PENJELASAN "):
   226→   226→                    self._flush_penjelasan()
   227→   227→                    self.in_penjelasan = True
   228→   228→                    self._penjelasan_pasal = None
   229→   229→                    continue
   230→   230→                # Inside Penjelasan: track pasal references for IDs
   231→   231→                if self.in_penjelasan:
   232→   232→                    # v5: also capture alpha suffixes in Penjelasan pasal references
   233→   233→                    m_pj = re.match(r'^\s*Pasal\s+(\d+[a-zA-Z]*)', text, re.IGNORECASE)
   234→   234→                    if m_pj:
   235→   235→                        self._flush_penjelasan()
   236→   236→                        self._penjelasan_pasal = m_pj.group(1)
   237→   237→                    else:
   238→   238→                        self._penjelasan_lines.append(text)
   239→   239→                continue
   240→   240→
   241→   241→            # Check for quoted amendment TRIGGER ("Pasal X diubah sehingga berbunyi:")
   242→   242→            if not self.in_quoted_amendment:
   243→   243→                if self.RE_AMENDMENT_TRIGGER.search(text):
   244→   244→                    # Extract which pasal is being amended
   245→   245→                    m_pasal = re.search(r'Pasal\s+(\d+|\w+)\s+(?:ayat\s*\(\d+\)\s+)?(?:diubah|diganti|dihapus)',
   246→   246→                                         text, re.IGNORECASE)
   247→   247→                    source_pasal = m_pasal.group(1) if m_pasal else "?"
   248→   248→                    self.in_quoted_amendment = True
   249→   249→                    self.quoted_amendment_source = source_pasal
   250→   250→                    self._flush_ayat()
   251→   251→                    # Store the trigger sentence as context for the quoted chunk
   252→   252→                    self.quoted_amendment_lines = [text]
   253→   253→                    continue
   254→   254→
   255→   255→            # Inside quoted amendment: accumulate text
   256→   256→            if self.in_quoted_amendment:
   257→   257→                self.quoted_amendment_lines.append(text)
   258→   258→                if self.RE_QUOTE_END.search(text):
   259→   259→                    self._flush_quoted_amendment()
   260→   260→                # Also check for end of quoted section without closing quote:
   261→   261→                # if we hit a new structural marker, the quote ended implicitly
   262→   262→                if (self.RE_BAB.match(text) or self.RE_PASAL.match(text) or
   263→   263→                    self.RE_PENUTUP.match(text)):
   264→   264→                    self._flush_quoted_amendment()
   265→   265→                    # Don't skip the structural marker - let it be processed below
   266→   266→                    self.in_quoted_amendment = False
   267→   267→                elif self.RE_QUOTE_END.search(text):
   268→   268→                    pass  # already flushed above
   269→   269→                else:
   270→   270→                    continue
   271→   271→
   272→   272→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   273→   273→            m = self.RE_BAB.match(text)
   274→   274→            if m:
   275→   275→                self._flush_ayat()
   276→   276→                self._flush_pasal()
   277→   277→                self.current_bab = m.group(1)
   278→   278→                label_match = re.match(r'^\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   279→   279→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   280→   280→                # Fix #2: if label is empty or just repeats "BAB X", set to None
   281→   281→                if raw_label and raw_label.upper() != f"BAB {m.group(1).upper()}":
   282→   282→                    self.current_bab_label = raw_label
   283→   283→                else:
   284→   284→                    self.current_bab_label = None
   285→   285→                self.current_bagian = None
   286→   286→                self.current_bagian_label = None
   287→   287→                self.current_paragraf = None
   288→   288→                self.current_paragraf_label = None
   289→   289→                # v4: reset pasal when BAB changes to prevent heading-as-chunk
   290→   290→                self.current_pasal = None
   291→   291→                continue
   292→   292→
   293→   293→            m = self.RE_BAGIAN.match(text)
   294→   294→            if m:
   295→   295→                self._flush_ayat()
   296→   296→                self._flush_pasal()
   297→   297→                self.current_bagian = m.group(1)
   298→   298→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   299→   299→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   300→   300→                if raw_label:
   301→   301→                    self.current_bagian_label = raw_label
   302→   302→                else:
   303→   303→                    self.current_bagian_label = None
   304→   304→                self.current_paragraf = None
   305→   305→                self.current_paragraf_label = None
   306→   306→                continue
   307→   307→
   308→   308→            m = self.RE_PARAGRAF.match(text)
   309→   309→            if m:
   310→   310→                self._flush_ayat()
   311→   311→                self._flush_pasal()
   312→   312→                self.current_paragraf = m.group(1)
   313→   313→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   314→   314→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   315→   315→                if raw_label:
   316→   316→                    self.current_paragraf_label = raw_label
   317→   317→                else:
   318→   318→                    self.current_paragraf_label = None
   319→   319→                continue
   320→   320→
   321→   321→            m = self.RE_PASAL.match(text)
   322→   322→            if m:
   323→   323→                self._flush_ayat()
   324→   324→                self._flush_pasal()
   325→   325→                raw_num = m.group(1)
   326→   326→                # v4: normalize space-separated digits ("1 1" -> "11", "1 10" -> "110")
   327→   327→                self.current_pasal = re.sub(r'\s+', '', raw_num) if re.match(r'^\d', raw_num) else raw_num
   328→   328→                self.current_ayat = None
   329→   329→                # v4: capture any remaining text on the same line as pasal heading
   330→   330→                remaining = text[m.end():].strip().lstrip('.').strip()
   331→   331→                if remaining and len(remaining) > 5:
   332→   332→                    self.current_pasal_lines.append(remaining)
   333→   333→                continue
   334→   334→
   335→   335→            m = self.RE_AYAT.match(text)
   336→   336→            if m and self.current_pasal:
   337→   337→                self._flush_ayat()
   338→   338→                self.current_ayat = m.group(1)
   339→   339→                ayat_text = self.RE_AYAT.sub('', text).strip()
   340→   340→                if ayat_text:
   341→   341→                    self.current_ayat_lines.append(ayat_text)
   342→   342→                continue
   343→   343→
   344→   344→            # If we're inside a pasal but no ayat detected, accumulate lines
   345→   345→            if self.current_pasal:
   346→   346→                if self.current_ayat:
   347→   347→                    self.current_ayat_lines.append(text)
   348→   348→                else:
   349→   349→                    self.current_pasal_lines.append(text)
   350→   350→
   351→   351→        # Flush remaining
   352→   352→        self._flush_ayat()
   353→   353→        self._flush_pasal()
   354→   354→        self._flush_quoted_amendment()
   355→   355→
   356→   356→        return self.chunks
   357→   357→
   358→   358→    def _flush_ayat(self):
   359→   359→        """Flush accumulated ayat lines as a chunk."""
   360→   360→        if not self.current_ayat or not self.current_ayat_lines:
   361→   361→            self.current_ayat_lines = []
   362→   362→            return
   363→   363→
   364→   364→        text = ' '.join(self.current_ayat_lines).strip()
   365→   365→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   366→   366→            self.current_ayat_lines = []
   367→   367→            return
   368→   368→
   369→   369→        headings = _make_bab_headings(
   370→   370→            self.current_bab, self.current_bab_label,
   371→   371→            self.current_bagian, self.current_bagian_label,
   372→   372→            self.current_paragraf, self.current_paragraf_label,
   373→   373→            pasal_str=f"Pasal {self.current_pasal}",
   374→   374→            ayat_str=f"Ayat ({self.current_ayat})"
   375→   375→        )
   376→   376→
   377→   377→        path = build_path(headings)
   378→   378→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   379→   379→                            pasal=self.current_pasal, ayat=self.current_ayat)
   380→   380→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   381→   381→
   382→   382→        self.chunks.append({
   383→   383→            "id": chunk_id,
   384→   384→            "text": text,
   385→   385→            "metadata": {
   386→   386→                "doc_type": self.doc_type,
   387→   387→                "issuer": self.issuer,
   388→   388→                "year": self.year,
   389→   389→                "bab": self.current_bab or None,
   390→   390→                "pasal": self.current_pasal,
   391→   391→                "ayat": self.current_ayat,
   392→   392→                "parent": parent_id,
   393→   393→                "path": path,
   394→   394→                "status": "active",
   395→   395→            }
   396→   396→        })
   397→   397→        self.current_ayat_lines = []
   398→   398→
   399→   399→    def _flush_pasal(self):
   400→   400→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   401→   401→        if not self.current_pasal or not self.current_pasal_lines:
   402→   402→            self.current_pasal_lines = []
   403→   403→            return
   404→   404→
   405→   405→        text = ' '.join(self.current_pasal_lines).strip()
   406→   406→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   407→   407→            self.current_pasal_lines = []
   408→   408→            return

   826→   826→def chunk_pdf(pdf_path: str, directory: str, family_map: Dict) -> List[Dict]:
   827→   827→    """
   828→   828→    Main entry point. Extract + parse a PDF into chunks.
   829→   829→    Returns list of chunk dicts with id, text, metadata.
   830→   830→    """
   831→   831→    dir_config = family_map.get(directory, {})

   837→   837→        lines, meta = extract_clean(pdf_path)
   838→   838→    except Exception as e:
   839→   839→        return [{"id": f"ERROR:{directory}", "text": str(e), "metadata": {"error": True}}]

   843→   843→
   844→   844→    # v4 fix: Detect doc_type from title block FIRST, directory as fallback.
   845→   845→    # Use page-1-only RAW lines (pre-boilerplate) since boilerplate stripping

   849→   849→    raw_title_block = " ".join(page1_raw[:15]) if page1_raw else ""
