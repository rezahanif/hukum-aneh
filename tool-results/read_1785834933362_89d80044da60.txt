    42→def build_id(doc_type, nomor, year, **parts):
    43→    """Build chunk ID. Handles both hierarchical and flat structures."""
    44→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    45→        nomor_perkara = parts.get("nomor_perkara", nomor or "?")
    46→        chunk_type = parts.get("chunk_type", "amar")
    47→        chunk_num = parts.get("chunk_num", "?")
    48→        return f"{doc_type}:{nomor_perkara}:{chunk_type}:{chunk_num}"
    49→    elif parts.get("chunk_type") in ("diktum", "pertimbangan"):
    50→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{parts['chunk_type']}:{parts['chunk_num']}"
    51→    else:
    52→        pasal = parts.get("pasal", "?")
    53→        ayat = parts.get("ayat", "?")
    54→        if ayat and ayat != "?":
    55→            return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}:{ayat}"
    56→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"

    58→
    59→def build_parent_id(doc_type, nomor, year, pasal):
    60→    """Build parent ID for ayat chunks."""
    61→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    62→        return None
    63→    return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"

    66→def build_path(headings):
    67→    """Build path string from heading stack. Filters out None and duplicates."""
    68→    seen = set()
    69→    parts = []
    70→    for h in headings:
    71→        if h and h not in seen:
    72→            seen.add(h)
    73→            parts.append(h)
    74→    return " > ".join(parts) if parts else ""

    77→def _make_bab_headings(bab_num, bab_label, bagian_num, bagian_label,
    78→                       paragraf_num, paragraf_label, pasal_str=None, ayat_str=None):
    79→    """Build heading list for path, avoiding duplicate BAB labels.
    80→    
    81→    Fix for bug #2: when BAB has no descriptive label (e.g., just "BAB I"),
    82→    the label defaults to "BAB I" which duplicates the number element.
    83→    Now: if label equals the number element, omit it.
    84→    """
    85→    headings = []
    86→    
    87→    bab_num_str = f"BAB {bab_num}" if bab_num else None
    88→    if bab_num_str:
    89→        headings.append(bab_num_str)
    90→    # Only add label if it's different from the number element and non-empty
    91→    if bab_label and bab_label != bab_num_str:
    92→        headings.append(bab_label)
    93→    
    94→    bagian_num_str = f"Bagian {bagian_num}" if bagian_num else None
    95→    if bagian_num_str:
    96→        headings.append(bagian_num_str)
    97→    if bagian_label and bagian_label != bagian_num_str:
    98→        headings.append(bagian_label)
    99→    
   100→    paragraf_num_str = f"Paragraf {paragraf_num}" if paragraf_num else None
   101→    if paragraf_num_str:
   102→        headings.append(paragraf_num_str)
   103→    if paragraf_label and paragraf_label != paragraf_num_str:
   104→        headings.append(paragraf_label)
   105→    
   106→    if pasal_str:
   107→        headings.append(pasal_str)
   108→    if ayat_str:
   109→        headings.append(ayat_str)
   110→    
   111→    return headings

   117→
   118→class FamilyAParser:
   119→    """
   120→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
   121→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
   122→    """
   123→
   124→    # Heading patterns (order matters: higher levels first)
   125→    RE_BAB = re.compile(r'^\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
   126→    RE_BAGIAN = re.compile(r'^\s*BAGIAN\s+(\w+)', re.IGNORECASE)
   127→    RE_PARAGRAF = re.compile(r'^\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
   128→    RE_PASAL = re.compile(r'^\s*pasal\s+([IVXLCDM]+|\d+[a-zA-Z]?)\s*\.?\s*.{0,5}$', re.IGNORECASE)
   129→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
   130→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
   131→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
   132→
   133→    # Quoted amendment detection
   134→    RE_AMENDMENT_TRIGGER = re.compile(
   135→        r'(?:diubah|diganti|dihapus)\s+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
   136→        re.IGNORECASE
   137→    )
   138→    RE_QUOTE_START = re.compile(r'^\s*["“]')
   139→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
   140→
   141→    # Penutup (closing section) marker
   142→    RE_PENUTUP = re.compile(r'^(?:PENUTUP|penutup)\b')
   143→
   144→    def __init__(self, lines, doc_type, nomor, year, issuer):
   145→        self.lines = lines
   146→        self.doc_type = doc_type
   147→        self.nomor = nomor
   148→        self.year = year
   149→        self.issuer = issuer
   150→        self.chunks = []
   151→
   152→        # Heading stack
   153→        self.current_bab = None
   154→        self.current_bab_label = None
   155→        self.current_bagian = None
   156→        self.current_bagian_label = None
   157→        self.current_paragraf = None
   158→        self.current_paragraf_label = None
   159→        self.current_pasal = None
   160→        self.current_ayat = None
   161→
   162→        # State
   163→        self.in_preamble = True
   164→        self.in_penutup = False
   165→        self.in_quoted_amendment = False
   166→        self.quoted_amendment_source = None  # tracks which pasal is being amended
   167→        self.current_ayat_lines = []
   168→        self.current_pasal_lines = []
   169→        self.quoted_amendment_lines = []  # accumulate quoted text for chunking
   170→
   171→    def parse(self) -> List[Dict]:
   172→        """Main parse loop. Returns list of chunks."""
   173→        for i, line in enumerate(self.lines):
   174→            text = line["text"].strip()
   175→            if not text:
   176→                continue
   177→
   178→            text_upper = text.upper()
   179→
   180→            # Detect preamble end / body start
   181→            if self.in_preamble:
   182→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   183→                    self.in_preamble = False
   184→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
   185→                    self.in_preamble = False
   186→                    continue
   187→                else:
   188→                    continue
   189→
   190→            # Detect penutup
   191→            if self.RE_PENUTUP.match(text):
   192→                self._flush_ayat()
   193→                self._flush_pasal()
   194→                self._flush_quoted_amendment()
   195→                self.in_penutup = True
   196→                continue
   197→
   198→            if self.in_penutup:
   199→                continue
   200→
   201→            # Check for quoted amendment TRIGGER ("Pasal X diubah sehingga berbunyi:")
   202→            if not self.in_quoted_amendment:
   203→                if self.RE_AMENDMENT_TRIGGER.search(text):
   204→                    # Extract which pasal is being amended
   205→                    m_pasal = re.search(r'Pasal\s+(\d+|\w+)\s+(?:ayat\s*\(\d+\)\s+)?(?:diubah|diganti|dihapus)',
   206→                                         text, re.IGNORECASE)
   207→                    source_pasal = m_pasal.group(1) if m_pasal else "?"
   208→                    self.in_quoted_amendment = True
   209→                    self.quoted_amendment_source = source_pasal
   210→                    self._flush_ayat()
   211→                    # Store the trigger sentence as context for the quoted chunk
   212→                    self.quoted_amendment_lines = [text]
   213→                    continue
   214→
   215→            # Inside quoted amendment: accumulate text
   216→            if self.in_quoted_amendment:
   217→                self.quoted_amendment_lines.append(text)
   218→                if self.RE_QUOTE_END.search(text):
   219→                    self._flush_quoted_amendment()
   220→                # Also check for end of quoted section without closing quote:
   221→                # if we hit a new structural marker, the quote ended implicitly
   222→                if (self.RE_BAB.match(text) or self.RE_PASAL.match(text) or
   223→                    self.RE_PENUTUP.match(text)):
   224→                    self._flush_quoted_amendment()
   225→                    # Don't skip the structural marker - let it be processed below
   226→                    self.in_quoted_amendment = False
   227→                elif self.RE_QUOTE_END.search(text):
   228→                    pass  # already flushed above
   229→                else:
   230→                    continue
   231→
   232→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   233→            m = self.RE_BAB.match(text)
   234→            if m:
   235→                self._flush_ayat()
   236→                self._flush_pasal()
   237→                self.current_bab = m.group(1)
   238→                label_match = re.match(r'^\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   239→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   240→                # Fix #2: if label is empty or just repeats "BAB X", set to None
   241→                if raw_label and raw_label.upper() != f"BAB {m.group(1).upper()}":
   242→                    self.current_bab_label = raw_label
   243→                else:
   244→                    self.current_bab_label = None
   245→                self.current_bagian = None
   246→                self.current_bagian_label = None
   247→                self.current_paragraf = None
   248→                self.current_paragraf_label = None
   249→                continue
   250→
   251→            m = self.RE_BAGIAN.match(text)
   252→            if m:
   253→                self._flush_ayat()
   254→                self._flush_pasal()
   255→                self.current_bagian = m.group(1)
   256→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   257→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   258→                if raw_label:
   259→                    self.current_bagian_label = raw_label
   260→                else:
   261→                    self.current_bagian_label = None
   262→                self.current_paragraf = None
   263→                self.current_paragraf_label = None
   264→                continue
   265→
   266→            m = self.RE_PARAGRAF.match(text)
   267→            if m:
   268→                self._flush_ayat()
   269→                self._flush_pasal()
   270→                self.current_paragraf = m.group(1)
   271→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   272→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   273→                if raw_label:
   274→                    self.current_paragraf_label = raw_label
   275→                else:
   276→                    self.current_paragraf_label = None
   277→                continue
   278→
   279→            m = self.RE_PASAL.match(text)
   280→            if m:
   281→                self._flush_ayat()
   282→                self._flush_pasal()
   283→                self.current_pasal = m.group(1)
   284→                self.current_ayat = None
   285→                continue
   286→
   287→            m = self.RE_AYAT.match(text)
   288→            if m and self.current_pasal:
   289→                self._flush_ayat()
   290→                self.current_ayat = m.group(1)
   291→                ayat_text = self.RE_AYAT.sub('', text).strip()
   292→                if ayat_text:
   293→                    self.current_ayat_lines.append(ayat_text)
   294→                continue
   295→
   296→            # If we're inside a pasal but no ayat detected, accumulate lines
   297→            if self.current_pasal:
   298→                if self.current_ayat:
   299→                    self.current_ayat_lines.append(text)
   300→                else:
   301→                    self.current_pasal_lines.append(text)
   302→
   303→        # Flush remaining
   304→        self._flush_ayat()
   305→        self._flush_pasal()
   306→        self._flush_quoted_amendment()
   307→
   308→        return self.chunks
   309→
   310→    def _flush_ayat(self):
   311→        """Flush accumulated ayat lines as a chunk."""
   312→        if not self.current_ayat or not self.current_ayat_lines:
   313→            self.current_ayat_lines = []
   314→            return
   315→
   316→        text = ' '.join(self.current_ayat_lines).strip()
   317→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   318→            self.current_ayat_lines = []
   319→            return
   320→
   321→        headings = _make_bab_headings(
   322→            self.current_bab, self.current_bab_label,
   323→            self.current_bagian, self.current_bagian_label,
   324→            self.current_paragraf, self.current_paragraf_label,
   325→            pasal_str=f"Pasal {self.current_pasal}",
   326→            ayat_str=f"Ayat ({self.current_ayat})"
   327→        )
   328→
   329→        path = build_path(headings)
   330→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   331→                            pasal=self.current_pasal, ayat=self.current_ayat)
   332→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   333→
   334→        self.chunks.append({
   335→            "id": chunk_id,
   336→            "text": text,
   337→            "metadata": {
   338→                "doc_type": self.doc_type,
   339→                "issuer": self.issuer,
   340→                "year": self.year,
   341→                "bab": self.current_bab or None,
   342→                "pasal": self.current_pasal,
   343→                "ayat": self.current_ayat,
   344→                "parent": parent_id,
   345→                "path": path,
   346→                "status": "active",
   347→            }
   348→        })
   349→        self.current_ayat_lines = []
   350→
   351→    def _flush_pasal(self):
   352→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   353→        if not self.current_pasal or not self.current_pasal_lines:
   354→            self.current_pasal_lines = []
   355→            return
   356→
   357→        text = ' '.join(self.current_pasal_lines).strip()
   358→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   359→            self.current_pasal_lines = []
   360→            return
   361→
   362→        headings = _make_bab_headings(
   363→            self.current_bab, self.current_bab_label,
   364→            self.current_bagian, self.current_bagian_label,
   365→            self.current_paragraf, self.current_paragraf_label,
   366→            pasal_str=f"Pasal {self.current_pasal}",
   367→        )
   368→        path = build_path(headings)
   369→        chunk_id = build_id(self.doc_type, self.nomor, self.year, pasal=self.current_pasal)
   370→
   371→        self.chunks.append({
   372→            "id": chunk_id,
   373→            "text": text,
   374→            "metadata": {
   375→                "doc_type": self.doc_type,
   376→                "issuer": self.issuer,
   377→                "year": self.year,
   378→                "bab": self.current_bab or None,
   379→                "pasal": self.current_pasal,
   380→                "ayat": None,
   381→                "parent": None,
   382→                "path": path,
   383→                "status": "active",
   384→            }
   385→        })
   386→        self.current_pasal_lines = []
   387→
   388→    def _flush_quoted_amendment(self):
   389→        """Flush quoted amendment text as a chunk with status='quoted_amendment'.
   390→        
   391→        Fix for bug #6: instead of silently skipping quoted amendments,
   392→        we now chunk them with a clear status tag so retrieval can
   393→        distinguish live law from historical amendment pointers.
   394→        """
   395→        if not self.quoted_amendment_lines:
   396→            self.in_quoted_amendment = False
   397→            self.quoted_amendment_source = None
   398→            return
   399→
   400→        text = ' '.join(self.quoted_amendment_lines).strip()
   401→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   402→            self.quoted_amendment_lines = []
   403→            self.in_quoted_amendment = False
   404→            self.quoted_amendment_source = None
   405→            return
   406→
   407→        source = self.quoted_amendment_source or "?"
   408→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   409→                            pasal=source, ayat="quoted")
   410→
   411→        self.chunks.append({
   412→            "id": chunk_id,
   413→            "text": text,
   414→            "metadata": {
   415→                "doc_type": self.doc_type,
   416→                "issuer": self.issuer,
   417→                "year": self.year,
   418→                "bab": self.current_bab or None,
   419→                "pasal": source,
   420→                "ayat": "quoted",
   421→                "parent": None,
   422→                "path": f"[AMANDEMEN Pasal {source}]",
   423→                "status": "quoted_amendment",
   424→                "amends_pasal": source,
   425→            }
   426→        })
   427→        self.quoted_amendment_lines = []
   428→        self.in_quoted_amendment = False
   429→        self.quoted_amendment_source = None

   433→# FAMILY B: DECREE/DECISION PARSER
   434→# ==========================================================================
   435→
   436→class FamilyBParser:
   437→    """
   438→    Parses decree/decision documents: Menimbang > Mengingat > MEMUTUSKAN/MENETAPKAN > Diktum.
   439→    Handles: Keppres, Inpres, Tap_MPR, Kepmen.
   440→    
   441→    v2 fix: Recognizes "Pasal X" as diktum divider (keppres pattern where
   442→    diktum items are organized under Pasal headings instead of numbered).
   443→    """
   444→
   445→    RE_MENIMBANG = re.compile(r'^\s*menimbang\s*:?\s*$', re.IGNORECASE)
   446→    RE_MENGINGAT = re.compile(r'^\s*mengingat\s*:?\s*$', re.IGNORECASE)
   447→    # Decision keywords: use $ anchor to match at END of line, allowing any prefix
   448→    RE_MEMUTUSKAN = re.compile(r'memutuskan\s*:?\s*$', re.IGNORECASE)
   449→    RE_MENETAPKAN = re.compile(r'menetapkan\s*:?\s*$', re.IGNORECASE)
   450→    RE_MENINSTRUKSIKAN = re.compile(r'menginstruksikan\s*:?\s*$', re.IGNORECASE)
   451→    RE_ITEM_LETTER = re.compile(r'^\s*([a-z])\s*\.\s*')
   452→    RE_ITEM_NUMBER = re.compile(r'^\s*(\d+)\s*\.\s*')
   453→    RE_ORDINAL = re.compile(
   454→        r'^\s*(KE(?:SATU|DUA|TIGA|EMPAT|LIMA|NAM|TUJUH|DELAPAN|SEMBILAN|SEPULUH|'
   455→        r'SEBELAS|DUA BELAS|TIGA BELAS|EMPAT BELAS|LIMA BELAS|ENAM BELAS|'
   456→        r'TUJUH BELAS|DELAPAN BELAS|SEMBILAN BELAS|DUA PULUH))\b',
   457→        re.IGNORECASE
   458→    )
   459→    # Fix #5: recognize "Pasal X" as diktum divider in Family B
   460→    RE_PASAL_DIKTUM = re.compile(r'^\s*Pasal\s+(\d+[a-zA-Z]?)\s*\.?\s*$', re.IGNORECASE)
   461→
   462→    def __init__(self, lines, doc_type, nomor, year, issuer):
   463→        self.lines = lines
   464→        self.doc_type = doc_type
   465→        self.nomor = nomor
   466→        self.year = year
   467→        self.issuer = issuer
   468→        self.chunks = []
   469→
   470→        self.section = "pre"  # pre, menimbang, mengingat, keputusan, done
   471→        self.current_item = None
   472→        self.current_item_lines = []
   473→        self.preamble_lines = []
   474→        self.decision_keyword = None
   475→
   476→    def parse(self) -> List[Dict]:
   477→        for line in self.lines:
   478→            text = line["text"].strip()
   479→            if not text:
   480→                continue
   481→
   482→            text_upper = text.upper()
   483→
   484→            # Section transitions
   485→            if self.RE_MENIMBANG.match(text):
   486→                self._flush_preamble()
   487→                self.section = "menimbang"
   488→                self.current_item = None
   489→                continue
   490→
   491→            if self.RE_MENGINGAT.match(text):
   492→                self._flush_preamble()
   493→                self.section = "mengingat"
   494→                self.current_item = None
   495→                continue
   496→
   497→            # Decision keyword triggers (use search() for end-anchored patterns)
   498→            if self.section not in ("keputusan", "done"):
   499→                if self.RE_MENINSTRUKSIKAN.search(text):
   500→                    self._flush_item()
   501→                    self.section = "keputusan"
   502→                    self.decision_keyword = "MENGINSTRUKSIKAN"
   503→                    self.current_item = None
   504→                    continue
   505→
   506→                if self.RE_MEMUTUSKAN.search(text):
   507→                    self._flush_item()
   508→                    self.section = "keputusan"
   509→                    self.decision_keyword = "MEMUTUSKAN"
   510→                    self.current_item = None
   511→                    continue
   512→
   513→                if self.RE_MENETAPKAN.search(text):
   514→                    self._flush_item()
   515→                    self.section = "keputusan"
   516→                    self.decision_keyword = "MENETAPKAN"
   517→                    self.current_item = None
   518→                    continue
   519→
   520→            if self.section == "pre":
   521→                continue
   522→
   523→            if self.section == "done":
   524→                continue
   525→
   526→            # Signature block detection
   527→            text_lower = text.lower()
   528→            if "ttd" in text_lower and len(text) < 20:
   529→                self._flush_item()
   530→                self.section = "done"
   531→                continue
   532→
   533→            # In keputusan section: check for item markers
   534→            if self.section == "keputusan":
   535→                m_letter = self.RE_ITEM_LETTER.match(text)
   536→                m_number = self.RE_ITEM_NUMBER.match(text)
   537→                m_ordinal = self.RE_ORDINAL.match(text)
   538→                m_pasal = self.RE_PASAL_DIKTUM.match(text)
   539→
   540→                if m_pasal:
   541→                    # Fix #5: "Pasal X" acts as a diktum divider in some keppres
   542→                    self._flush_item()
   543→                    self.current_item = f"Pasal {m_pasal.group(1)}"
   544→                    continue
   545→
   546→                if m_letter or m_number or m_ordinal:
   547→                    self._flush_item()
   548→                    if m_ordinal:
   549→                        self.current_item = m_ordinal.group(1).upper()
   550→                    elif m_letter:
   551→                        self.current_item = m_letter.group(1)
   552→                    else:
   553→                        self.current_item = m_number.group(1)
   554→                    m = m_letter or m_number or m_ordinal
   555→                    remaining = text[m.end():].strip()
   556→                    if remaining:
   557→                        self.current_item_lines.append(remaining)
   558→                    continue
   559→
   560→                # Sub-items: accumulate under current item
   561→                if self.current_item:
   562→                    self.current_item_lines.append(text)
   563→                continue
   564→
   565→        self._flush_item()
   566→        return self.chunks
   567→
   568→    def _flush_item(self):
   569→        if not self.current_item or not self.current_item_lines:
   570→            self.current_item_lines = []
   571→            return
   572→
   573→        text = ' '.join(self.current_item_lines).strip()
   574→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   575→            self.current_item_lines = []
   576→            return
   577→
   578→        kw = self.decision_keyword or "MEMUTUSKAN"
   579→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   580→                            chunk_type="diktum", chunk_num=self.current_item)
   581→
   582→        self.chunks.append({
   583→            "id": chunk_id,
   584→            "text": text,
   585→            "metadata": {
   586→                "doc_type": self.doc_type,
   587→                "issuer": self.issuer,
   588→                "year": self.year,
   589→                "nomor": self.nomor,
   590→                "section": "keputusan",
   591→                "decision_keyword": kw,
   592→                "diktum": self.current_item,
   593→                "parent": None,
   594→                "path": f"{kw} > Diktum {self.current_item}",
   595→                "status": "active",
   596→            }
   597→        })
   598→        self.current_item_lines = []
   599→
   600→    def _flush_preamble(self):
   601→        self.preamble_lines = []

   608→class FamilyCParser:
   609→    """
   610→    Parses court rulings: Duduk Perkara > Menimbang > MENGADILI/MENETAPKAN > Amar.
   611→    Handles: Putusan MK, Putusan MA.
   612→    """
   613→
   614→    RE_MENGADILI = re.compile(r'^\s*mengadili\s*:?\s*$', re.IGNORECASE)
   615→    RE_MENETAPKAN = re.compile(r'^\s*menetapkan\s*:?\s*$', re.IGNORECASE)
   616→    RE_AMAR = re.compile(r'^\s*(\d+)\s*\.\s+')
   617→    RE_PUTUSAN = re.compile(r'^\s*putusan\s*:?\s*$', re.IGNORECASE)
   618→
   619→    def __init__(self, lines, doc_type, nomor, year, issuer, nomor_perkara=None):
   620→        self.lines = lines
   621→        self.doc_type = doc_type
   622→        self.nomor = nomor
   623→        self.year = year
   624→        self.issuer = issuer
   625→        self.nomor_perkara = nomor_perkara
   626→        self.chunks = []
   627→
   628→        self.section = "pre"
   629→        self.current_amar = None
   630→        self.current_amar_lines = []
   631→        self.pertimbangan_lines = []
   632→
   633→    def parse(self) -> List[Dict]:
   634→        for line in self.lines:
   635→            text = line["text"].strip()
   636→            if not text:
   637→                continue
   638→
   639→            text_upper = text.upper()
   640→
   641→            if self.RE_MENGADILI.match(text):
   642→                self._flush_amar()
   643→                self.section = "amar"
   644→                self.decision_keyword = "MENGADILI"
   645→                continue
   646→
   647→            if self.RE_MENETAPKAN.match(text):
   648→                self._flush_amar()
   649→                self.section = "amar"
   650→                self.decision_keyword = "MENETAPKAN"
   651→                continue
   652→
   653→            if self.section == "pre":
   654→                continue
   655→
   656→            if self.section == "done":
   657→                continue
   658→
   659→            if self.section == "amar":
   660→                m = self.RE_AMAR.match(text)
   661→                if m:
   662→                    self._flush_amar()
   663→                    self.current_amar = m.group(1)
   664→                    remaining = text[m.end():].strip()
   665→                    if remaining:
   666→                        self.current_amar_lines.append(remaining)
   667→                    continue
   668→
   669→                if self.current_amar:
   670→                    self.current_amar_lines.append(text)
   671→
   672→                if re.search(r'\bttd\b|\bHAKIM\b|\bKETUA\b', text_upper) and len(text) < 30:
   673→                    self._flush_amar()
   674→                    self.section = "done"
   675→
   676→        self._flush_amar()
   677→        return self.chunks
   678→
   679→    def _flush_amar(self):
   680→        if not self.current_amar or not self.current_amar_lines:
   681→            self.current_amar_lines = []
   682→            return
   683→
   684→        text = ' '.join(self.current_amar_lines).strip()
   685→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   686→            self.current_amar_lines = []
   687→            return
   688→
   689→        kw = getattr(self, 'decision_keyword', 'MENGADILI')
   690→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   691→                            nomor_perkara=self.nomor_perkara,
   692→                            chunk_type="amar", chunk_num=self.current_amar)
   693→
   694→        self.chunks.append({
   695→            "id": chunk_id,
   696→            "text": text,
   697→            "metadata": {
   698→                "doc_type": self.doc_type,
   699→                "issuer": self.issuer,
   700→                "year": self.year,
   701→                "nomor_perkara": self.nomor_perkara,
   702→                "section": "amar",
   703→                "decision_keyword": kw,
   704→                "amar": self.current_amar,
   705→                "parent": None,
   706→                "path": f"{kw} > Amar {self.current_amar}",
   707→                "status": "active",
   708→            }
   709→        })
   710→        self.current_amar_lines = []

   717→def extract_nomor_perkara(lines):
   718→    """Extract case number (e.g. 96/PUU-XVI/2018) from court ruling lines."""
   719→    for line in lines[:30]:
   720→        m = re.search(r'(\d+/[A-Z]+-[A-Z]+/\d{4})', line["text"])
   721→        if m:
   722→            return m.group(1)
   723→    return None

   726→def chunk_pdf(pdf_path: str, directory: str, family_map: Dict) -> List[Dict]:
   727→    """
   728→    Main entry point. Extract + parse a PDF into chunks.
   729→    Returns list of chunk dicts with id, text, metadata.
   730→    """
   731→    dir_config = family_map.get(directory, {})
   732→    family = dir_config.get("family", "A")
   733→    default_doc_type = dir_config.get("doc_type", "Unknown")
   734→    default_issuer = dir_config.get("issuer", "Unknown")
   735→
   736→    try:
   737→        lines, meta = extract_clean(pdf_path)
   738→    except Exception as e:
   739→        return [{"id": f"ERROR:{directory}", "text": str(e), "metadata": {"error": True}}]
   740→
   741→    if not lines:
   742→        return [{"id": f"EMPTY:{directory}", "text": "No extractable text (possibly scanned PDF)", "metadata": {"error": True, "needs_ocr": True}}]
   743→
   744→    # Detect doc type: prefer directory config, use detection only as fallback.
   745→    # The directory mapping is authoritative (set by the pipeline operator).
   746→    title_block = extract_title_block(lines, max_first_n=5)
   747→    detected_type, detected_issuer = detect_doc_type_from_title(title_block)
   748→    
   749→    if default_doc_type != "Unknown":
   750→        doc_type = default_doc_type
   751→        issuer = default_issuer
   752→    elif detected_type != "Unknown":
   753→        doc_type = detected_type
   754→        issuer = detected_issuer
   755→    else:
   756→        doc_type = "Unknown"
   757→        issuer = "Unknown"
   758→    nomor = meta.get("nomor")
   759→    year = meta.get("year")
   760→
   761→    # Special case for JDIH_KPU: page 1 is image, detect from page 2+ content
   762→    if doc_type == "Unknown" and directory == "JDIH_KPU":
   763→        doc_type = "PKPU"
   764→        issuer = "Komisi Pemilihan Umum"
   765→
   766→    # Parse based on family
   767→    if family == "A":
   768→        parser = FamilyAParser(lines, doc_type, nomor, year, issuer)
   769→        chunks = parser.parse()
   770→    elif family == "B":
   771→        parser = FamilyBParser(lines, doc_type, nomor, year, issuer)
   772→        chunks = parser.parse()
   773→    elif family == "C":
   774→        nomor_perkara = extract_nomor_perkara(lines)
   775→        parser = FamilyCParser(lines, doc_type, nomor, year, issuer, nomor_perkara)
   776→        chunks = parser.parse()
   777→    else:
   778→        chunks = [{"id": f"UNKNOWN_FAMILY:{directory}", "text": "", "metadata": {"error": True}}]
   779→
   780→    # Add extraction metadata to all chunks
   781→    for chunk in chunks:
   782→        chunk["metadata"]["source_file"] = meta["file"]
   783→        chunk["metadata"]["directory"] = directory
   784→        chunk["metadata"]["family"] = family
   785→        if nomor:
   786→            chunk["metadata"]["nomor"] = nomor
   787→        if year:
   788→            chunk["metadata"]["year"] = year
   789→
   790→    return chunks
