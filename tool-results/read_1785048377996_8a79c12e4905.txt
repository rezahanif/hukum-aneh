    21→// calcTitleLayout for English
    22→function calcTitleLayout(title, maxWidthTwips, preferredPt = 40, minPt = 24) {
    23→  const charWidth = (pt) => pt * 11;
    24→  const charsPerLine = (pt) => Math.floor(maxWidthTwips / charWidth(pt));
    25→  let titlePt = preferredPt;
    26→  let lines;
    27→  while (titlePt >= minPt) {
    28→    const cpl = charsPerLine(titlePt);
    29→    if (cpl < 2) { titlePt -= 2; continue; }
    30→    lines = splitTitleLines(title, cpl);
    31→    if (lines.length <= 3) break;
    32→    titlePt -= 2;
    33→  }
    34→  if (!lines || lines.length > 3) {
    35→    lines = splitTitleLines(title, charsPerLine(minPt));
    36→    titlePt = minPt;
    37→  }
    38→  return { titlePt, titleLines: lines };
    39→}
    40→
    41→function splitTitleLines(title, charsPerLine) {
    42→  if (title.length <= charsPerLine) return [title];
    43→  const breakAfter = new Set([' ', '-', '_', '/', ':', '\u2014']);
    44→  const lines = [];
    45→  let remaining = title;
    46→  while (remaining.length > charsPerLine) {
    47→    let breakAt = -1;
    48→    for (let i = charsPerLine; i >= Math.floor(charsPerLine * 0.6); i--) {
    49→      if (i < remaining.length && breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    50→    }
    51→    if (breakAt === -1) breakAt = charsPerLine;
    52→    lines.push(remaining.slice(0, breakAt));
    53→    remaining = remaining.slice(breakAt);
    54→  }
    55→  if (remaining.length > 0) lines.push(remaining);
    56→  return lines;
    57→}

    59→function calcCoverSpacing(params) {
    60→  const { titleLineCount = 1, titlePt = 36, hasSubtitle = false, hasEnglishLabel = false, metaLineCount = 0, fixedHeight = 400, pageHeight = 16838, marginTop = 0, marginBottom = 0 } = params;
    61→  const SAFETY = 1200;
    62→  const usableHeight = pageHeight - marginTop - marginBottom - SAFETY;
    63→  const titleHeight = titleLineCount * (titlePt * 23 + 200);
    64→  const subtitleHeight = hasSubtitle ? (12 * 23 + 600) : 0;
    65→  const englishLabelHeight = hasEnglishLabel ? (9 * 23 + 600) : 0;
    66→  const metaHeight = metaLineCount * (10 * 23 + 100);
    67→  const implicitParaHeight = 3 * 300;
    68→  const contentHeight = titleHeight + subtitleHeight + englishLabelHeight + metaHeight + fixedHeight + implicitParaHeight;
    69→  const remainingSpace = Math.max(usableHeight - contentHeight, 400);
    70→  const FOOTER_MIN = 800;
    71→  const rawTop = Math.floor(remainingSpace * 0.45);
    72→  const rawBottom = Math.floor(remainingSpace * 0.45);
    73→  const bottomSpacing = Math.max(rawBottom, FOOTER_MIN);
    74→  return { topSpacing: rawTop, midSpacing: Math.floor(remainingSpace * 0.1), bottomSpacing };
    75→}

   185→    }))
   186→  });
   187→  const dataRows = rows.map((row, ri) => new TableRow({ cantSplit: true, children:
   188→    row.map(cell => new TableCell({ width: { size: colW, type: WidthType.PERCENTAGE },

   193→    }))
   194→  }));
   195→  return new Table({ width: { size: 100, type: WidthType.PERCENTAGE }, layout: TableLayoutType.FIXED, rows: [headerRow, ...dataRows] });
