    43→    43→def build_id(doc_type, nomor, year, **parts):
    44→    44→    """Build chunk ID. Handles both hierarchical and flat structures."""
    45→    45→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    46→    46→        nomor_perkara = parts.get("nomor_perkara", nomor or "?")
    47→    47→        chunk_type = parts.get("chunk_type", "amar")
    48→    48→        chunk_num = parts.get("chunk_num", "?")
    49→    49→        return f"{doc_type}:{nomor_perkara}:{chunk_type}:{chunk_num}"
    50→    50→    elif parts.get("chunk_type") in ("diktum", "pertimbangan"):
    51→    51→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{parts['chunk_type']}:{parts['chunk_num']}"
    52→    52→    else:
    53→    53→        pasal = parts.get("pasal", "?")
    54→    54→        ayat = parts.get("ayat", "?")
    55→    55→        if ayat and ayat != "?":
    56→    56→            return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}:{ayat}"
    57→    57→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"

    68→    68→    """Build path string from heading stack. Filters out None and duplicates."""
    69→    69→    seen = set()
    70→    70→    parts = []
    71→    71→    for h in headings:
    72→    72→        if h and h not in seen:
    73→    73→            seen.add(h)
    74→    74→            parts.append(h)
    75→    75→    return " > ".join(parts) if parts else ""

    78→    78→def _make_bab_headings(bab_num, bab_label, bagian_num, bagian_label,
    79→    79→                       paragraf_num, paragraf_label, pasal_str=None, ayat_str=None):
    80→    80→    """Build heading list for path, avoiding duplicate BAB labels.
    81→    81→    
    82→    82→    Fix for bug #2: when BAB has no descriptive label (e.g., just "BAB I"),
    83→    83→    the label defaults to "BAB I" which duplicates the number element.
    84→    84→    Now: if label equals the number element, omit it.
    85→    85→    """
    86→    86→    headings = []
    87→    87→    
    88→    88→    bab_num_str = f"BAB {bab_num}" if bab_num else None
    89→    89→    if bab_num_str:
    90→    90→        headings.append(bab_num_str)
    91→    91→    # Only add label if it's different from the number element and non-empty
    92→    92→    if bab_label and bab_label != bab_num_str:
    93→    93→        headings.append(bab_label)
    94→    94→    
    95→    95→    bagian_num_str = f"Bagian {bagian_num}" if bagian_num else None
    96→    96→    if bagian_num_str:
    97→    97→        headings.append(bagian_num_str)
    98→    98→    if bagian_label and bagian_label != bagian_num_str:
    99→    99→        headings.append(bagian_label)
   100→   100→    
   101→   101→    paragraf_num_str = f"Paragraf {paragraf_num}" if paragraf_num else None
   102→   102→    if paragraf_num_str:
   103→   103→        headings.append(paragraf_num_str)
   104→   104→    if paragraf_label and paragraf_label != paragraf_num_str:
   105→   105→        headings.append(paragraf_label)
   106→   106→    
   107→   107→    if pasal_str:
   108→   108→        headings.append(pasal_str)
   109→   109→    if ayat_str:
   110→   110→        headings.append(ayat_str)
   111→   111→    
   112→   112→    return headings

   115→   115→# ==========================================================================
   116→   116→# FAMILY A: HIERARCHICAL STATUTE PARSER
   117→   117→# ==========================================================================
   118→   118→
   119→   119→class FamilyAParser:
   120→   120→    """
   121→   121→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
   122→   122→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
   123→   123→    """
   124→   124→
   125→   125→    # Heading patterns (order matters: higher levels first)
   126→   126→    RE_BAB = re.compile(r'^\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
   127→   127→    RE_BAGIAN = re.compile(r'^\s*BAGIAN\s+(\w+)', re.IGNORECASE)
   128→   128→    RE_PARAGRAF = re.compile(r'^\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
   129→   129→    RE_PASAL = re.compile(r'^\s*pasal\s+([IVXLCDM]+|\d+[a-zA-Z]?)\s*\.?\s*.{0,5}$', re.IGNORECASE)
   130→   130→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
   131→   131→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
   132→   132→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
   133→   133→
   134→   134→    # Quoted amendment detection
   135→   135→    RE_AMENDMENT_TRIGGER = re.compile(
   136→   136→        r'(?:diubah|diganti|dihapus)\s+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
   137→   137→        re.IGNORECASE
   138→   138→    )
   139→   139→    RE_QUOTE_START = re.compile(r'^\s*["“]')
   140→   140→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
   141→   141→
   142→   142→    # Penutup (closing section) marker
   143→   143→    RE_PENUTUP = re.compile(r'^(?:PENUTUP|penutup)\b')
   144→   144→
   145→   145→    def __init__(self, lines, doc_type, nomor, year, issuer):
   146→   146→        self.lines = lines
   147→   147→        self.doc_type = doc_type
   148→   148→        self.nomor = nomor
   149→   149→        self.year = year
   150→   150→        self.issuer = issuer
   151→   151→        self.chunks = []
   152→   152→
   153→   153→        # Heading stack
   154→   154→        self.current_bab = None
   155→   155→        self.current_bab_label = None
   156→   156→        self.current_bagian = None
   157→   157→        self.current_bagian_label = None
   158→   158→        self.current_paragraf = None
   159→   159→        self.current_paragraf_label = None
   160→   160→        self.current_pasal = None
   161→   161→        self.current_ayat = None
   162→   162→
   163→   163→        # State
   164→   164→        self.in_preamble = True
   165→   165→        self.in_penutup = False
   166→   166→        self.in_quoted_amendment = False
   167→   167→        self.quoted_amendment_source = None  # tracks which pasal is being amended
   168→   168→        self.current_ayat_lines = []
   169→   169→        self.current_pasal_lines = []
   170→   170→        self.quoted_amendment_lines = []  # accumulate quoted text for chunking
   171→   171→
   172→   172→    def parse(self) -> List[Dict]:
   173→   173→        """Main parse loop. Returns list of chunks."""
   174→   174→        for i, line in enumerate(self.lines):
   175→   175→            text = line["text"].strip()
   176→   176→            if not text:
   177→   177→                continue
   178→   178→
   179→   179→            text_upper = text.upper()
   180→   180→
   181→   181→            # Detect preamble end / body start
   182→   182→            if self.in_preamble:
   183→   183→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   184→   184→                    self.in_preamble = False
   185→   185→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
   186→   186→                    self.in_preamble = False
   187→   187→                    continue
   188→   188→                else:
   189→   189→                    continue
   190→   190→
   191→   191→            # Detect penutup
   192→   192→            if self.RE_PENUTUP.match(text):
   193→   193→                self._flush_ayat()
   194→   194→                self._flush_pasal()
   195→   195→                self._flush_quoted_amendment()
   196→   196→                self.in_penutup = True
   197→   197→                continue
   198→   198→
   199→   199→            if self.in_penutup:
   200→   200→                continue
   201→   201→
   202→   202→            # Check for quoted amendment TRIGGER ("Pasal X diubah sehingga berbunyi:")
   203→   203→            if not self.in_quoted_amendment:
   204→   204→                if self.RE_AMENDMENT_TRIGGER.search(text):
   205→   205→                    # Extract which pasal is being amended
   206→   206→                    m_pasal = re.search(r'Pasal\s+(\d+|\w+)\s+(?:ayat\s*\(\d+\)\s+)?(?:diubah|diganti|dihapus)',
   207→   207→                                         text, re.IGNORECASE)
   208→   208→                    source_pasal = m_pasal.group(1) if m_pasal else "?"
   209→   209→                    self.in_quoted_amendment = True
   210→   210→                    self.quoted_amendment_source = source_pasal
   211→   211→                    self._flush_ayat()
   212→   212→                    # Store the trigger sentence as context for the quoted chunk
   213→   213→                    self.quoted_amendment_lines = [text]
   214→   214→                    continue
   215→   215→
   216→   216→            # Inside quoted amendment: accumulate text
   217→   217→            if self.in_quoted_amendment:
   218→   218→                self.quoted_amendment_lines.append(text)
   219→   219→                if self.RE_QUOTE_END.search(text):
   220→   220→                    self._flush_quoted_amendment()
   221→   221→                # Also check for end of quoted section without closing quote:
   222→   222→                # if we hit a new structural marker, the quote ended implicitly
   223→   223→                if (self.RE_BAB.match(text) or self.RE_PASAL.match(text) or
   224→   224→                    self.RE_PENUTUP.match(text)):
   225→   225→                    self._flush_quoted_amendment()
   226→   226→                    # Don't skip the structural marker - let it be processed below
   227→   227→                    self.in_quoted_amendment = False
   228→   228→                elif self.RE_QUOTE_END.search(text):
   229→   229→                    pass  # already flushed above
   230→   230→                else:
   231→   231→                    continue
   232→   232→
   233→   233→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   234→   234→            m = self.RE_BAB.match(text)
   235→   235→            if m:
   236→   236→                self._flush_ayat()
   237→   237→                self._flush_pasal()
   238→   238→                self.current_bab = m.group(1)
   239→   239→                label_match = re.match(r'^\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   240→   240→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   241→   241→                # Fix #2: if label is empty or just repeats "BAB X", set to None
   242→   242→                if raw_label and raw_label.upper() != f"BAB {m.group(1).upper()}":
   243→   243→                    self.current_bab_label = raw_label
   244→   244→                else:
   245→   245→                    self.current_bab_label = None
   246→   246→                self.current_bagian = None
   247→   247→                self.current_bagian_label = None
   248→   248→                self.current_paragraf = None
   249→   249→                self.current_paragraf_label = None
   250→   250→                continue
   251→   251→
   252→   252→            m = self.RE_BAGIAN.match(text)
   253→   253→            if m:
   254→   254→                self._flush_ayat()
   255→   255→                self._flush_pasal()
   256→   256→                self.current_bagian = m.group(1)
   257→   257→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   258→   258→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   259→   259→                if raw_label:
   260→   260→                    self.current_bagian_label = raw_label
   261→   261→                else:
   262→   262→                    self.current_bagian_label = None
   263→   263→                self.current_paragraf = None
   264→   264→                self.current_paragraf_label = None
   265→   265→                continue
   266→   266→
   267→   267→            m = self.RE_PARAGRAF.match(text)
   268→   268→            if m:
   269→   269→                self._flush_ayat()
   270→   270→                self._flush_pasal()
   271→   271→                self.current_paragraf = m.group(1)
   272→   272→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   273→   273→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   274→   274→                if raw_label:
   275→   275→                    self.current_paragraf_label = raw_label
   276→   276→                else:
   277→   277→                    self.current_paragraf_label = None
   278→   278→                continue
   279→   279→
   280→   280→            m = self.RE_PASAL.match(text)
   281→   281→            if m:
   282→   282→                self._flush_ayat()
   283→   283→                self._flush_pasal()
   284→   284→                self.current_pasal = m.group(1)
   285→   285→                self.current_ayat = None
   286→   286→                continue
   287→   287→
   288→   288→            m = self.RE_AYAT.match(text)
   289→   289→            if m and self.current_pasal:
   290→   290→                self._flush_ayat()
   291→   291→                self.current_ayat = m.group(1)
   292→   292→                ayat_text = self.RE_AYAT.sub('', text).strip()
   293→   293→                if ayat_text:
   294→   294→                    self.current_ayat_lines.append(ayat_text)
   295→   295→                continue
   296→   296→
   297→   297→            # If we're inside a pasal but no ayat detected, accumulate lines
   298→   298→            if self.current_pasal:
   299→   299→                if self.current_ayat:
   300→   300→                    self.current_ayat_lines.append(text)
   301→   301→                else:
   302→   302→                    self.current_pasal_lines.append(text)
   303→   303→
   304→   304→        # Flush remaining
   305→   305→        self._flush_ayat()
   306→   306→        self._flush_pasal()
   307→   307→        self._flush_quoted_amendment()
   308→   308→
   309→   309→        return self.chunks
   310→   310→
   311→   311→    def _flush_ayat(self):
   312→   312→        """Flush accumulated ayat lines as a chunk."""
   313→   313→        if not self.current_ayat or not self.current_ayat_lines:
   314→   314→            self.current_ayat_lines = []
   315→   315→            return
   316→   316→
   317→   317→        text = ' '.join(self.current_ayat_lines).strip()
   318→   318→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   319→   319→            self.current_ayat_lines = []
   320→   320→            return
   321→   321→
   322→   322→        headings = _make_bab_headings(
   323→   323→            self.current_bab, self.current_bab_label,
   324→   324→            self.current_bagian, self.current_bagian_label,
   325→   325→            self.current_paragraf, self.current_paragraf_label,
   326→   326→            pasal_str=f"Pasal {self.current_pasal}",
   327→   327→            ayat_str=f"Ayat ({self.current_ayat})"
   328→   328→        )
   329→   329→
   330→   330→        path = build_path(headings)
   331→   331→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   332→   332→                            pasal=self.current_pasal, ayat=self.current_ayat)
   333→   333→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   334→   334→
   335→   335→        self.chunks.append({
   336→   336→            "id": chunk_id,
   337→   337→            "text": text,
   338→   338→            "metadata": {
   339→   339→                "doc_type": self.doc_type,
   340→   340→                "issuer": self.issuer,
   341→   341→                "year": self.year,
   342→   342→                "bab": self.current_bab or None,
   343→   343→                "pasal": self.current_pasal,
   344→   344→                "ayat": self.current_ayat,
   345→   345→                "parent": parent_id,
   346→   346→                "path": path,
   347→   347→                "status": "active",
   348→   348→            }
   349→   349→        })
   350→   350→        self.current_ayat_lines = []
   351→   351→
   352→   352→    def _flush_pasal(self):
   353→   353→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   354→   354→        if not self.current_pasal or not self.current_pasal_lines:
   355→   355→            self.current_pasal_lines = []
   356→   356→            return
   357→   357→
   358→   358→        text = ' '.join(self.current_pasal_lines).strip()
   359→   359→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   360→   360→            self.current_pasal_lines = []
   361→   361→            return
   362→   362→
   363→   363→        headings = _make_bab_headings(
   364→   364→            self.current_bab, self.current_bab_label,
   365→   365→            self.current_bagian, self.current_bagian_label,
   366→   366→            self.current_paragraf, self.current_paragraf_label,
   367→   367→            pasal_str=f"Pasal {self.current_pasal}",
   368→   368→        )
   369→   369→        path = build_path(headings)
   370→   370→        chunk_id = build_id(self.doc_type, self.nomor, self.year, pasal=self.current_pasal)
   371→   371→
   372→   372→        self.chunks.append({
   373→   373→            "id": chunk_id,
   374→   374→            "text": text,
   375→   375→            "metadata": {
   376→   376→                "doc_type": self.doc_type,
   377→   377→                "issuer": self.issuer,
   378→   378→                "year": self.year,
   379→   379→                "bab": self.current_bab or None,
   380→   380→                "pasal": self.current_pasal,
   381→   381→                "ayat": None,
   382→   382→                "parent": None,
   383→   383→                "path": path,
   384→   384→                "status": "active",
   385→   385→            }
   386→   386→        })
   387→   387→        self.current_pasal_lines = []
   388→   388→
   389→   389→    def _flush_quoted_amendment(self):
   390→   390→        """Flush quoted amendment text as a chunk with status='quoted_amendment'.
   391→   391→        
   392→   392→        Fix for bug #6: instead of silently skipping quoted amendments,
   393→   393→        we now chunk them with a clear status tag so retrieval can
   394→   394→        distinguish live law from historical amendment pointers.
   395→   395→        """
   396→   396→        if not self.quoted_amendment_lines:
   397→   397→            self.in_quoted_amendment = False
   398→   398→            self.quoted_amendment_source = None
   399→   399→            return
   400→   400→
   401→   401→        text = ' '.join(self.quoted_amendment_lines).strip()
   402→   402→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   403→   403→            self.quoted_amendment_lines = []
   404→   404→            self.in_quoted_amendment = False
   405→   405→            self.quoted_amendment_source = None
   406→   406→            return
   407→   407→
   408→   408→        source = self.quoted_amendment_source or "?"
   409→   409→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   410→   410→                            pasal=source, ayat="quoted")
   411→   411→
   412→   412→        self.chunks.append({
   413→   413→            "id": chunk_id,
   414→   414→            "text": text,
   415→   415→            "metadata": {
   416→   416→                "doc_type": self.doc_type,
   417→   417→                "issuer": self.issuer,
   418→   418→                "year": self.year,
   419→   419→                "bab": self.current_bab or None,
   420→   420→                "pasal": source,
   421→   421→                "ayat": "quoted",
   422→   422→                "parent": None,
   423→   423→                "path": f"[AMANDEMEN Pasal {source}]",
   424→   424→                "status": "quoted_amendment",
   425→   425→                "amends_pasal": source,
   426→   426→            }
   427→   427→        })

   437→   437→class FamilyBParser:
   438→   438→    """
   439→   439→    Parses decree/decision documents: Menimbang > Mengingat > MEMUTUSKAN/MENETAPKAN > Diktum.
   440→   440→    Handles: Keppres, Inpres, Tap_MPR, Kepmen.
   441→   441→    
   442→   442→    v2 fix: Recognizes "Pasal X" as diktum divider (keppres pattern where
   443→   443→    diktum items are organized under Pasal headings instead of numbered).
   444→   444→    """
   445→   445→
   446→   446→    RE_MENIMBANG = re.compile(r'^\s*menimbang\s*:?\s*$', re.IGNORECASE)
   447→   447→    RE_MENGINGAT = re.compile(r'^\s*mengingat\s*:?\s*$', re.IGNORECASE)
   448→   448→    # Decision keywords: use $ anchor to match at END of line, allowing any prefix
   449→   449→    RE_MEMUTUSKAN = re.compile(r'memutuskan\s*:?\s*$', re.IGNORECASE)
   450→   450→    RE_MENETAPKAN = re.compile(r'menetapkan\s*:?\s*$', re.IGNORECASE)
   451→   451→    RE_MENINSTRUKSIKAN = re.compile(r'menginstruksikan\s*:?\s*$', re.IGNORECASE)
   452→   452→    RE_ITEM_LETTER = re.compile(r'^\s*([a-z])\s*\.\s*')
   453→   453→    RE_ITEM_NUMBER = re.compile(r'^\s*(\d+)\s*\.\s*')
   454→   454→    RE_ORDINAL = re.compile(
   455→   455→        r'^\s*(KE(?:SATU|DUA|TIGA|EMPAT|LIMA|NAM|TUJUH|DELAPAN|SEMBILAN|SEPULUH|'
   456→   456→        r'SEBELAS|DUA BELAS|TIGA BELAS|EMPAT BELAS|LIMA BELAS|ENAM BELAS|'
   457→   457→        r'TUJUH BELAS|DELAPAN BELAS|SEMBILAN BELAS|DUA PULUH))\b',
   458→   458→        re.IGNORECASE
   459→   459→    )
   460→   460→    # Fix #5: recognize "Pasal X" as diktum divider in Family B
   461→   461→    RE_PASAL_DIKTUM = re.compile(r'^\s*Pasal\s+(\d+[a-zA-Z]?)\s*\.?\s*$', re.IGNORECASE)
   462→   462→
   463→   463→    def __init__(self, lines, doc_type, nomor, year, issuer):
   464→   464→        self.lines = lines
   465→   465→        self.doc_type = doc_type
   466→   466→        self.nomor = nomor
   467→   467→        self.year = year
   468→   468→        self.issuer = issuer
   469→   469→        self.chunks = []
   470→   470→
   471→   471→        self.section = "pre"  # pre, menimbang, mengingat, keputusan, done
   472→   472→        self.current_item = None
   473→   473→        self.current_item_lines = []
   474→   474→        self.preamble_lines = []
   475→   475→        self.decision_keyword = None

   477→   477→    def parse(self) -> List[Dict]:
   478→   478→        for line in self.lines:
   479→   479→            text = line["text"].strip()
   480→   480→            if not text:
   481→   481→                continue
   482→   482→
   483→   483→            text_upper = text.upper()
   484→   484→
   485→   485→            # Section transitions
   486→   486→            if self.RE_MENIMBANG.match(text):
   487→   487→                self._flush_preamble()
   488→   488→                self.section = "menimbang"
   489→   489→                self.current_item = None
   490→   490→                continue
   491→   491→
   492→   492→            if self.RE_MENGINGAT.match(text):
   493→   493→                self._flush_preamble()
   494→   494→                self.section = "mengingat"
   495→   495→                self.current_item = None
   496→   496→                continue
   497→   497→
   498→   498→            # Decision keyword triggers (use search() for end-anchored patterns)
   499→   499→            if self.section not in ("keputusan", "done"):
   500→   500→                if self.RE_MENINSTRUKSIKAN.search(text):
   501→   501→                    self._flush_item()
   502→   502→                    self.section = "keputusan"
   503→   503→                    self.decision_keyword = "MENGINSTRUKSIKAN"
   504→   504→                    self.current_item = None
   505→   505→                    continue
   506→   506→
   507→   507→                if self.RE_MEMUTUSKAN.search(text):
   508→   508→                    self._flush_item()
   509→   509→                    self.section = "keputusan"
   510→   510→                    self.decision_keyword = "MEMUTUSKAN"
   511→   511→                    self.current_item = None
   512→   512→                    continue
   513→   513→
   514→   514→                if self.RE_MENETAPKAN.search(text):
   515→   515→                    self._flush_item()
   516→   516→                    self.section = "keputusan"
   517→   517→                    self.decision_keyword = "MENETAPKAN"
   518→   518→                    self.current_item = None
   519→   519→                    continue
   520→   520→
   521→   521→            if self.section == "pre":
   522→   522→                continue
   523→   523→
   524→   524→            if self.section == "done":
   525→   525→                continue
   526→   526→
   527→   527→            # Signature block detection
   528→   528→            text_lower = text.lower()
   529→   529→            if "ttd" in text_lower and len(text) < 20:
   530→   530→                self._flush_item()
   531→   531→                self.section = "done"
   532→   532→                continue
   533→   533→
   534→   534→            # In keputusan section: check for item markers
   535→   535→            if self.section == "keputusan":
   536→   536→                m_letter = self.RE_ITEM_LETTER.match(text)
   537→   537→                m_number = self.RE_ITEM_NUMBER.match(text)
   538→   538→                m_ordinal = self.RE_ORDINAL.match(text)
   539→   539→                m_pasal = self.RE_PASAL_DIKTUM.match(text)
   540→   540→
   541→   541→                if m_pasal:
   542→   542→                    # Fix #5: "Pasal X" acts as a diktum divider in some keppres
   543→   543→                    self._flush_item()
   544→   544→                    self.current_item = f"Pasal {m_pasal.group(1)}"
   545→   545→                    continue
   546→   546→
   547→   547→                if m_letter or m_number or m_ordinal:
   548→   548→                    self._flush_item()
   549→   549→                    if m_ordinal:
   550→   550→                        self.current_item = m_ordinal.group(1).upper()
   551→   551→                    elif m_letter:
   552→   552→                        self.current_item = m_letter.group(1)
   553→   553→                    else:
   554→   554→                        self.current_item = m_number.group(1)
   555→   555→                    m = m_letter or m_number or m_ordinal
   556→   556→                    remaining = text[m.end():].strip()
   557→   557→                    if remaining:
   558→   558→                        self.current_item_lines.append(remaining)
   559→   559→                    continue
   560→   560→
   561→   561→                # Sub-items: accumulate under current item
   562→   562→                if self.current_item:
   563→   563→                    self.current_item_lines.append(text)
   564→   564→                continue
   565→   565→
   566→   566→        self._flush_item()
   567→   567→        return self.chunks
   568→   568→
   569→   569→    def _flush_item(self):
   570→   570→        if not self.current_item or not self.current_item_lines:
   571→   571→            self.current_item_lines = []
   572→   572→            return
   573→   573→
   574→   574→        text = ' '.join(self.current_item_lines).strip()
   575→   575→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   576→   576→            self.current_item_lines = []
   577→   577→            return
   578→   578→
   579→   579→        kw = self.decision_keyword or "MEMUTUSKAN"
   580→   580→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   581→   581→                            chunk_type="diktum", chunk_num=self.current_item)
   582→   582→
   583→   583→        self.chunks.append({
   584→   584→            "id": chunk_id,
   585→   585→            "text": text,
   586→   586→            "metadata": {
   587→   587→                "doc_type": self.doc_type,
   588→   588→                "issuer": self.issuer,
   589→   589→                "year": self.year,
   590→   590→                "nomor": self.nomor,
   591→   591→                "section": "keputusan",
   592→   592→                "decision_keyword": kw,
   593→   593→                "diktum": self.current_item,
   594→   594→                "parent": None,
   595→   595→                "path": f"{kw} > Diktum {self.current_item}",
   596→   596→                "status": "active",
   597→   597→            }
   598→   598→        })
   599→   599→        self.current_item_lines = []

   602→   602→        self.preamble_lines = []

   719→   719→    """Extract case number (e.g. 96/PUU-XVI/2018) from court ruling lines."""
   720→   720→    for line in lines[:30]:

   728→   728→    """
