    42→def build_id(doc_type, nomor, year, **parts):
    43→    """Build chunk ID. Handles both hierarchical and flat structures."""
    44→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    45→        nomor_perkara = parts.get("nomor_perkara", nomor or "?")
    46→        chunk_type = parts.get("chunk_type", "amar")
    47→        chunk_num = parts.get("chunk_num", "?")
    48→        return f"{doc_type}:{nomor_perkara}:{chunk_type}:{chunk_num}"
    49→    elif parts.get("chunk_type") in ("diktum", "pertimbangan"):
    50→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{parts['chunk_type']}:{parts['chunk_num']}"
    51→    else:
    52→        pasal = parts.get("pasal", "?")
    53→        ayat = parts.get("ayat", "?")
    54→        if ayat and ayat != "?":
    55→            return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}:{ayat}"
    56→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"
    57→
    58→
    59→def build_parent_id(doc_type, nomor, year, pasal):
    60→    """Build parent ID for ayat chunks."""
    61→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    62→        return None
    63→    return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"

    66→def build_path(headings):
    67→    """Build path string from heading stack. Filters out None and duplicates."""
    68→    seen = set()
    69→    parts = []
    70→    for h in headings:
    71→        if h and h not in seen:
    72→            seen.add(h)
    73→            parts.append(h)
    74→    return " > ".join(parts) if parts else ""

    77→def _make_bab_headings(bab_num, bab_label, bagian_num, bagian_label,
    78→                       paragraf_num, paragraf_label, pasal_str=None, ayat_str=None):
    79→    """Build heading list for path, avoiding duplicate BAB labels.
    80→    
    81→    Fix for bug #2: when BAB has no descriptive label (e.g., just "BAB I"),
    82→    the label defaults to "BAB I" which duplicates the number element.
    83→    Now: if label equals the number element, omit it.
    84→    """
    85→    headings = []
    86→    
    87→    bab_num_str = f"BAB {bab_num}" if bab_num else None
    88→    if bab_num_str:
    89→        headings.append(bab_num_str)
    90→    # Only add label if it's different from the number element and non-empty
    91→    if bab_label and bab_label != bab_num_str:
    92→        headings.append(bab_label)
    93→    
    94→    bagian_num_str = f"Bagian {bagian_num}" if bagian_num else None
    95→    if bagian_num_str:
    96→        headings.append(bagian_num_str)
    97→    if bagian_label and bagian_label != bagian_num_str:
    98→        headings.append(bagian_label)
    99→    
   100→    paragraf_num_str = f"Paragraf {paragraf_num}" if paragraf_num else None
   101→    if paragraf_num_str:
   102→        headings.append(paragraf_num_str)
   103→    if paragraf_label and paragraf_label != paragraf_num_str:
   104→        headings.append(paragraf_label)
   105→    
   106→    if pasal_str:
   107→        headings.append(pasal_str)
   108→    if ayat_str:
   109→        headings.append(ayat_str)
   110→    
   111→    return headings

   114→# ==========================================================================
   115→# FAMILY A: HIERARCHICAL STATUTE PARSER
   116→# ==========================================================================
   117→
   118→class FamilyAParser:
   119→    """
   120→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
   121→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
   122→    """
   123→
   124→    # Heading patterns (order matters: higher levels first)
   125→    # v6: allow '* ' prefix (UUD 1945 amendment markers)
   126→    RE_BAB = re.compile(r'^\*?\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
   127→    RE_BAGIAN = re.compile(r'^\*?\s*BAGIAN\s+(\w+)', re.IGNORECASE)
   128→    RE_PARAGRAF = re.compile(r'^\*?\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
   129→    # v6: capture alpha suffixes AND allow '* ' prefix (UUD 1945 amendment markers)
   130→    RE_PASAL = re.compile(r'^\*?\s*pasal\s+([IVXLCDM]+|\d+(?:\s+\d+)*[a-zA-Z]?)\s*[.}\s]*$', re.IGNORECASE)
   131→    # v6: Match pasal embedded mid-line (e.g., 'BAB VIIA ... * Pasal 22C')
   132→    RE_PASAL_EMBEDDED = re.compile(r'\*\s*Pasal\s+(\d+[a-zA-Z]*)', re.IGNORECASE)
   133→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
   134→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
   135→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
   136→
   137→    # Quoted amendment detection
   138→    # v7: allow optional comma between verb and "sehingga" (PKPU uses "diubah, sehingga")
   139→    RE_AMENDMENT_TRIGGER = re.compile(
   140→        r'(?:diubah|diganti|dihapus)[,\s]+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
   141→        re.IGNORECASE
   142→    )
   143→    RE_QUOTE_START = re.compile(r'^\s*["“]')
   144→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
   145→
   146→    # Penutup (closing section) marker
   147→    RE_PENUTUP = re.compile(r'^(?:KETENTUAN\s+PENUTUP|PENUTUP|penutup)\b')
   148→    # v3: Penjelasan (elucidation) marker — can appear after penutup or directly after last pasal
   149→    RE_PENJELASAN = re.compile(r'^(?:PENJELASAN|penjelasan)\b')
   150→
   151→    def __init__(self, lines, doc_type, nomor, year, issuer):
   152→        self.lines = lines
   153→        self.doc_type = doc_type
   154→        self.nomor = nomor
   155→        self.year = year
   156→        self.issuer = issuer
   157→        self.chunks = []
   158→
   159→        # Heading stack
   160→        self.current_bab = None
   161→        self.current_bab_label = None
   162→        self.current_bagian = None
   163→        self.current_bagian_label = None
   164→        self.current_paragraf = None
   165→        self.current_paragraf_label = None
   166→        self.current_pasal = None
   167→        self.current_ayat = None
   168→
   169→        # State
   170→        self.in_preamble = True
   171→        self.in_penutup = False
   172→        self.in_penjelasan = False  # v3: track elucidation section
   173→        self.in_quoted_amendment = False
   174→        self.quoted_amendment_source = None  # tracks which pasal is being amended
   175→        self.current_ayat_lines = []
   176→        self.current_pasal_lines = []
   177→        self.quoted_amendment_lines = []  # accumulate quoted text for chunking
   178→        self._penjelasan_lines = []  # v3: accumulate Penjelasan text
   179→        self._penjelasan_pasal = None
   180→
   181→    def parse(self) -> List[Dict]:
   182→        """Main parse loop. Returns list of chunks."""
   183→        for i, line in enumerate(self.lines):
   184→            text = line["text"].strip()
   185→            if not text:
   186→                continue
   187→
   188→            text_upper = text.upper()
   189→
   190→            # Detect preamble end / body start
   191→            if self.in_preamble:
   192→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   193→                    self.in_preamble = False
   194→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
   195→                    self.in_preamble = False
   196→                    continue
   197→                else:
   198→                    continue
   199→
   200→            # Detect penutup
   201→            if self.RE_PENUTUP.match(text):
   202→                self._flush_ayat()
   203→                self._flush_pasal()
   204→                self._flush_quoted_amendment()
   205→                self.in_penutup = True
   206→                continue
   207→
   208→            # v3: Detect PENJELASAN even without penutup (many docs go straight to penjelasan)
   209→            if self.RE_PENJELASAN.match(text) and not self.in_penjelasan:
   210→                self._flush_ayat()
   211→                self._flush_pasal()
   212→                self._flush_quoted_amendment()
   213→                self.in_penutup = True  # treat as post-operative
   214→                self.in_penjelasan = True
   215→                self._penjelasan_pasal = None
   216→                continue
   217→
   218→            if self.in_penutup:
   219→                # v3: Detect signature block and Penjelasan as hard boundaries
   220→                text_lower = text.lower()
   221→                # Signature block: lines like "Ditetapkan di Jakarta", "pada tanggal", or ttd/
   222→                # Also detect by very short lines after penutup start
   223→                if (re.search(r'ttd|disahkan|ditetapkan', text_lower) or
   224→                    (len(text) < 15 and text_upper == text and len(text) > 2)):
   225→                    self._flush_penjelasan()  # flush any accumulated penjelasan
   226→                    continue
   227→                # PENJELASAN section header
   228→                if text_upper == "PENJELASAN" or text_upper.startswith("PENJELASAN "):
   229→                    self._flush_penjelasan()
   230→                    self.in_penjelasan = True
   231→                    self._penjelasan_pasal = None
   232→                    continue
   233→                # Inside Penjelasan: track pasal references for IDs
   234→                if self.in_penjelasan:
   235→                    # v5: also capture alpha suffixes in Penjelasan pasal references
   236→                    m_pj = re.match(r'^\s*Pasal\s+(\d+[a-zA-Z]*)', text, re.IGNORECASE)
   237→                    if m_pj:
   238→                        self._flush_penjelasan()
   239→                        self._penjelasan_pasal = m_pj.group(1)
   240→                    else:
   241→                        self._penjelasan_lines.append(text)
   242→                continue
   243→
   244→            # Check for quoted amendment TRIGGER ("Pasal X diubah sehingga berbunyi:")
   245→            if not self.in_quoted_amendment:
   246→                if self.RE_AMENDMENT_TRIGGER.search(text):
   247→                    # Extract which pasal is being amended
   248→                    # v8: allow trailing comma and "huruf X" between ayat ref and verb
   249→                    # e.g. "Pasal 21 ayat (1), diubah" or "Pasal 28 ayat (4) huruf d diubah"
   250→                    m_pasal = re.search(
   251→                        r'Pasal\s+(\d+|\w+)\s+'
   252→                        r'(?:ayat\s*\(\d+\)\s*'?'
   253→                        r'(?:,\s*|\s+)?'
   254→                        r'(?:huruf\s+\w\s+)?'
   255→                        r'(?:diubah|diganti|dihapus)',
   256→                        text, re.IGNORECASE)
   257→                    source_pasal = m_pasal.group(1) if m_pasal else "?"
   258→                    self.in_quoted_amendment = True
   259→                    self.quoted_amendment_source = source_pasal
   260→                    self._flush_ayat()
   261→                    # Store the trigger sentence as context for the quoted chunk
   262→                    self.quoted_amendment_lines = [text]
   263→                    continue
   264→
   265→            # Inside quoted amendment: accumulate text
   266→            if self.in_quoted_amendment:
   267→                self.quoted_amendment_lines.append(text)
   268→                if self.RE_QUOTE_END.search(text):
   269→                    self._flush_quoted_amendment()
   270→                # Also check for end of quoted section without closing quote:
   271→                # if we hit a new structural marker, the quote ended implicitly
   272→                if (self.RE_BAB.match(text) or self.RE_PASAL.match(text) or
   273→                    self.RE_PENUTUP.match(text)):
   274→                    self._flush_quoted_amendment()
   275→                    # Don't skip the structural marker - let it be processed below
   276→                    self.in_quoted_amendment = False
   277→                elif self.RE_QUOTE_END.search(text):
   278→                    pass  # already flushed above
   279→                else:
   280→                    continue
   281→
   282→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   283→            m = self.RE_BAB.match(text)
   284→            if m:
   285→                self._flush_ayat()
   286→                self._flush_pasal()
   287→                self.current_bab = m.group(1)
   288→                label_match = re.match(r'^\*?\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   289→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   290→                # Fix #2: if label is empty or just repeats "BAB X", set to None
   291→                if raw_label and raw_label.upper() != f"BAB {m.group(1).upper()}":
   292→                    self.current_bab_label = raw_label
   293→                else:
   294→                    self.current_bab_label = None
   295→                self.current_bagian = None
   296→                self.current_bagian_label = None
   297→                self.current_paragraf = None
   298→                self.current_paragraf_label = None
   299→                # v4: reset pasal when BAB changes to prevent heading-as-chunk
   300→                self.current_pasal = None
   301→                # v6: Check if a Pasal reference is glued to the BAB line
   302→                # (e.g., 'BAB VIIA DEWAN PERWAKILAN DAERAH * Pasal 22C')
   303→                m_glued = self.RE_PASAL_EMBEDDED.search(text)
   304→                if m_glued:
   305→                    self.current_pasal = m_glued.group(1)
   306→                    self.current_ayat = None
   307→                continue
   308→
   309→            m = self.RE_BAGIAN.match(text)
   310→            if m:
   311→                self._flush_ayat()
   312→                self._flush_pasal()
   313→                self.current_bagian = m.group(1)
   314→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   315→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   316→                if raw_label:
   317→                    self.current_bagian_label = raw_label
   318→                else:
   319→                    self.current_bagian_label = None
   320→                self.current_paragraf = None
   321→                self.current_paragraf_label = None
   322→                continue
   323→
   324→            m = self.RE_PARAGRAF.match(text)
   325→            if m:
   326→                self._flush_ayat()
   327→                self._flush_pasal()
   328→                self.current_paragraf = m.group(1)
   329→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   330→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   331→                if raw_label:
   332→                    self.current_paragraf_label = raw_label
   333→                else:
   334→                    self.current_paragraf_label = None
   335→                continue
   336→
   337→            m = self.RE_PASAL.match(text)
   338→            if m:
   339→                self._flush_ayat()
   340→                self._flush_pasal()
   341→                raw_num = m.group(1)
   342→                # v4: normalize space-separated digits ("1 1" -> "11", "1 10" -> "110")
   343→                self.current_pasal = re.sub(r'\s+', '', raw_num) if re.match(r'^\d', raw_num) else raw_num
   344→                self.current_ayat = None
   345→                # v4: capture any remaining text on the same line as pasal heading
   346→                remaining = text[m.end():].strip().lstrip('.').strip()
   347→                if remaining and len(remaining) > 5:
   348→                    self.current_pasal_lines.append(remaining)
   349→                continue
   350→
   351→            m = self.RE_AYAT.match(text)
   352→            if m and self.current_pasal:
   353→                self._flush_ayat()
   354→                self.current_ayat = m.group(1)
   355→                ayat_text = self.RE_AYAT.sub('', text).strip()
   356→                if ayat_text:
   357→                    self.current_ayat_lines.append(ayat_text)
   358→                continue
   359→
   360→            # v6: Check for embedded pasal in non-structural line
   361→            # (e.g., 'PEMILIHAN UMUM * Pasal 22E' — BAB label with glued pasal)
   362→            m_emb = self.RE_PASAL_EMBEDDED.search(text)
   363→            if m_emb and self.current_pasal:
   364→                self._flush_ayat()
   365→                self._flush_pasal()
   366→                self.current_pasal = m_emb.group(1)
   367→                self.current_ayat = None
   368→                continue
   369→
   370→            # If we're inside a pasal but no ayat detected, accumulate lines
   371→            if self.current_pasal:
   372→                if self.current_ayat:
   373→                    self.current_ayat_lines.append(text)
   374→                else:
   375→                    self.current_pasal_lines.append(text)
   376→
   377→        # Flush remaining
   378→        self._flush_ayat()
   379→        self._flush_pasal()
   380→        self._flush_quoted_amendment()
   381→
   382→        # v6: Deduplicate truncated page-break fragments
   383→        self._dedup_truncated_fragments()
   384→
   385→        # v7: Flag B↔8 OCR confusion in pasal numbers
   386→        self._flag_ocr_pasal_numbers()
   387→
   388→        return self.chunks
   389→
   390→    def _flush_ayat(self):
   391→        """Flush accumulated ayat lines as a chunk."""
   392→        if not self.current_ayat or not self.current_ayat_lines:
   393→            self.current_ayat_lines = []
   394→            return
   395→
   396→        text = ' '.join(self.current_ayat_lines).strip()
   397→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   398→            self.current_ayat_lines = []
   399→            return
   400→
   401→        headings = _make_bab_headings(
   402→            self.current_bab, self.current_bab_label,
   403→            self.current_bagian, self.current_bagian_label,
   404→            self.current_paragraf, self.current_paragraf_label,
   405→            pasal_str=f"Pasal {self.current_pasal}",
   406→            ayat_str=f"Ayat ({self.current_ayat})"
   407→        )
   408→
   409→        path = build_path(headings)
   410→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   411→                            pasal=self.current_pasal, ayat=self.current_ayat)
   412→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   413→
   414→        self.chunks.append({
   415→            "id": chunk_id,
   416→            "text": text,
   417→            "metadata": {
   418→                "doc_type": self.doc_type,
   419→                "issuer": self.issuer,
   420→                "year": self.year,
   421→                "bab": self.current_bab or None,
   422→                "pasal": self.current_pasal,
   423→                "ayat": self.current_ayat,
   424→                "parent": parent_id,
   425→                "path": path,
   426→                "status": "active",
   427→            }
   428→        })
   429→        self.current_ayat_lines = []
   430→
   431→    def _flush_pasal(self):
   432→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   433→        if not self.current_pasal or not self.current_pasal_lines:
   434→            self.current_pasal_lines = []
   435→            return
   436→
   437→        text = ' '.join(self.current_pasal_lines).strip()
   438→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   439→            self.current_pasal_lines = []
   440→            return
   441→
   442→        headings = _make_bab_headings(
   443→            self.current_bab, self.current_bab_label,
   444→            self.current_bagian, self.current_bagian_label,
   445→            self.current_paragraf, self.current_paragraf_label,
   446→            pasal_str=f"Pasal {self.current_pasal}",
   447→        )
   448→        path = build_path(headings)
   449→        chunk_id = build_id(self.doc_type, self.nomor, self.year, pasal=self.current_pasal)
   450→
   451→        self.chunks.append({
   452→            "id": chunk_id,
   453→            "text": text,
   454→            "metadata": {
   455→                "doc_type": self.doc_type,
   456→                "issuer": self.issuer,
   457→                "year": self.year,
   458→                "bab": self.current_bab or None,
   459→                "pasal": self.current_pasal,
   460→                "ayat": None,
   461→                "parent": None,
   462→                "path": path,
   463→                "status": "active",
   464→            }
   465→        })
   466→        self.current_pasal_lines = []
   467→
   468→    def _flush_quoted_amendment(self):
   469→        """Flush quoted amendment text as a chunk with status='quoted_amendment'.
   470→        
   471→        Fix for bug #6: instead of silently skipping quoted amendments,
   472→        we now chunk them with a clear status tag so retrieval can
   473→        distinguish live law from historical amendment pointers.
   474→        """
   475→        if not self.quoted_amendment_lines:
   476→            self.in_quoted_amendment = False
   477→            self.quoted_amendment_source = None
   478→            return
   479→
   480→        text = ' '.join(self.quoted_amendment_lines).strip()

   547→    def _dedup_truncated_fragments(self):
   548→        """v6+v7: Remove page-break artifacts that share an ID with a real chunk.
   549→        
   550→        v6: When text spans a page break, the bottom-of-page fragment (10-50 chars)
   551→        gets the same pasal:ayat ID as the continuation on the next page.
   552→        Keep the longer chunk, drop the fragment (if it's a substring).
   553→        
   554→        v7: Also catch page-break label chunks (organizational unit names like
   555→        "Sekretariat Presiden", "Deputi Bidang ..." that appear at the top of
   556→        a new page and inherit the current pasal:ayat context). These are NOT
   557→        substrings of the real content, but they are very short (< 80 chars)
   558→        and collide with the real chunk's ID.
   559→        
   560→        Also catches OCR fragments like "Proses . . . A" (< 50 chars).
   561→        """
   562→        from collections import defaultdict
   563→        id_chunks = defaultdict(list)
   564→        for i, chunk in enumerate(self.chunks):
   565→            id_chunks[chunk["id"]].append(i)
   566→        
   567→        to_remove = set()
   568→        for chunk_id, indices in id_chunks.items():
   569→            if len(indices) < 2:
   570→                continue
   571→            pairs = [(i, len(self.chunks[i]["text"])) for i in indices]
   572→            pairs.sort(key=lambda x: -x[1])
   573→            longest_idx, longest_len = pairs[0]
   574→            
   575→            for idx, text_len in pairs[1:]:
   576→                shorter = self.chunks[idx]["text"].strip()
   577→                longer = self.chunks[longest_idx]["text"].strip()
   578→                # v6: substring-based dedup (truncated page-break tail)
   579→                if text_len < longest_len * 0.5:
   580→                    if longer.endswith(shorter) or shorter in longer:
   581→                        to_remove.add(idx)
   582→                # v7: short page-break label/artifact (< 80 chars, not substring)
   583→                # Checked independently (not elif) to catch non-substring fragments
   584→                if text_len < 80 and text_len < longest_len * 0.4:
   585→                    to_remove.add(idx)
   586→        
   587→        if to_remove:
   588→            self.chunks = [c for i, c in enumerate(self.chunks) if i not in to_remove]
   589→
   590→    def _flag_ocr_pasal_numbers(self):
   591→        """v7: Detect and flag pasal numbers likely corrupted by B↔8 OCR confusion.
   592→        
   593→        Pattern: In amendment-heavy documents (like UU 41/2014), pasal numbers
   594→        with letter suffixes (36A, 36B, 36C...) are common. If OCR reads 'B' as '8',
   595→        '36B' becomes '368' — a plausible-looking but wrong pasal number.
   596→        
   597→        Detection heuristic: If a numeric pasal number ends with a digit that
   598→        could be an OCR-confused letter, AND the same prefix has OTHER alpha
   599→        variants but NOT this one, it's likely OCR corruption.
   600→        
   601→        Example: 36A, 36C, 36D, 36E exist, but 368 (likely 36B) fills the gap.
   602→        
   603→        This adds metadata._ocr_flag to affected chunks for downstream QA.
   604→        """
   605→        # Collect all pasal numbers from chunks
   606→        pasal_map = {}  # pasal_str -> [chunk_indices]
   607→        for i, chunk in enumerate(self.chunks):
   608→            p = chunk.get('metadata', {}).get('pasal')
   609→            if p and p != '?':
   610→                pasal_map.setdefault(str(p), []).append(i)
   611→
   612→        OCR_DIGIT_TO_LETTER = {'8': 'B', '0': 'O', '1': 'I', '5': 'S', '6': 'G'}
   613→        flagged = set()
   614→        
   615→        for p_str in pasal_map:
   616→            # Only check all-digit pasal numbers with 2+ digits
   617→            if not re.match(r'^\d+$', p_str):
   618→                continue
   619→            if len(p_str) < 2:
   620→                continue
   621→            last_char = p_str[-1]
   622→            if last_char not in OCR_DIGIT_TO_LETTER:
   623→                continue
   624→            
   625→            prefix = p_str[:-1]
   626→            letter = OCR_DIGIT_TO_LETTER[last_char]
   627→            candidate = prefix + letter
   628→            
   629→            # The alpha variant does NOT exist in the data (it's the OCR-corrupted one)
   630→            if candidate in pasal_map:
   631→                continue
   632→            
   633→            # Check if the prefix has ANY other alpha-suffixed variants
   634→            # (e.g., 36A, 36C exist → 368 is likely 36B)
   635→            has_alpha_siblings = any(
   636→                k.startswith(prefix) and len(k) == len(prefix) + 1 and k[-1].isalpha()
   637→                for k in pasal_map
   638→            )
   639→            
   640→            if has_alpha_siblings:
   641→                flagged.add(p_str)
   642→                for idx in pasal_map[p_str]:
   643→                    self.chunks[idx]['metadata']['_ocr_flag'] = (
   644→                        f'Possible OCR: pasal "{p_str}" may be "{candidate}" '
   645→                        f'(sibling alpha pasals exist for prefix {prefix})'
   646→                    )
   647→
   648→        if flagged:
   649→            print(f"  [WARN] OCR pasal flag: {flagged} (trailing digit may be letter)")
   650→

   656→class FamilyBParser:
   657→    """
   658→    Parses decree/decision documents: Menimbang > Mengingat > MEMUTUSKAN/MENETAPKAN > Diktum.
   659→    Handles: Keppres, Inpres, Tap_MPR, Kepmen.
   660→    
   661→    v2 fix: Recognizes "Pasal X" as diktum divider (keppres pattern where
   662→    diktum items are organized under Pasal headings instead of numbered).
   663→    """
   664→
   665→    RE_MENIMBANG = re.compile(r'^\s*menimbang\s*:?\s*$', re.IGNORECASE)
   666→    RE_MENGINGAT = re.compile(r'^\s*mengingat\s*:?\s*$', re.IGNORECASE)
   667→    # Decision keywords: use $ anchor to match at END of line, allowing any prefix
   668→    RE_MEMUTUSKAN = re.compile(r'memutuskan\s*:?\s*$', re.IGNORECASE)
   669→    RE_MENETAPKAN = re.compile(r'menetapkan\s*:?\s*$', re.IGNORECASE)
   670→    RE_MENINSTRUKSIKAN = re.compile(r'menginstruksikan\s*:?\s*$', re.IGNORECASE)
   671→    RE_ITEM_LETTER = re.compile(r'^\s*([a-z])\s*\.\s*')
   672→    RE_ITEM_NUMBER = re.compile(r'^\s*(\d+)\s*\.\s*')
   673→    RE_ORDINAL = re.compile(
   674→        r'^\s*(KE(?:SATU|DUA|TIGA|EMPAT|LIMA|NAM|TUJUH|DELAPAN|SEMBILAN|SEPULUH|'
   675→        r'SEBELAS|DUA BELAS|TIGA BELAS|EMPAT BELAS|LIMA BELAS|ENAM BELAS|'
   676→        r'TUJUH BELAS|DELAPAN BELAS|SEMBILAN BELAS|DUA PULUH))\b',
   677→        re.IGNORECASE
   678→    )
   679→    # Fix #5: recognize "Pasal X" as diktum divider in Family B
   680→    # v5: also handle alpha suffixed pasal numbers (e.g. Pasal 68A)
   681→    RE_PASAL_DIKTUM = re.compile(r'^\s*Pasal\s+(\d+[a-zA-Z]?)\s*\.?\s*$', re.IGNORECASE)
   682→
   683→    def __init__(self, lines, doc_type, nomor, year, issuer):
   684→        self.lines = lines
   685→        self.doc_type = doc_type
   686→        self.nomor = nomor
   687→        self.year = year
   688→        self.issuer = issuer
   689→        self.chunks = []
   690→
   691→        self.section = "pre"  # pre, menimbang, mengingat, keputusan, done
   692→        self.current_item = None
   693→        self.current_item_lines = []
   694→        self.preamble_lines = []
   695→        self.decision_keyword = None
   696→        self.parent_section = None  # v3: track parent lettered section for inpres diktum IDs
   697→
   698→    def parse(self) -> List[Dict]:
   699→        for line in self.lines:
   700→            text = line["text"].strip()
   701→            if not text:
   702→                continue
   703→
   704→            text_upper = text.upper()
   705→
   706→            # Section transitions
   707→            if self.RE_MENIMBANG.match(text):
   708→                self._flush_preamble()
   709→                self.section = "menimbang"
   710→                self.current_item = None
   711→                continue
   712→
   713→            if self.RE_MENGINGAT.match(text):
   714→                self._flush_preamble()
   715→                self.section = "mengingat"
   716→                self.current_item = None
   717→                continue
   718→
   719→            # Decision keyword triggers (use search() for end-anchored patterns)
   720→            if self.section not in ("keputusan", "done"):
   721→                if self.RE_MENINSTRUKSIKAN.search(text):
   722→                    self._flush_item()
   723→                    self.section = "keputusan"
   724→                    self.decision_keyword = "MENGINSTRUKSIKAN"
   725→                    self.current_item = None
   726→                    continue
   727→
   728→                if self.RE_MEMUTUSKAN.search(text):
   729→                    self._flush_item()
   730→                    self.section = "keputusan"
   731→                    self.decision_keyword = "MEMUTUSKAN"
   732→                    self.current_item = None
   733→                    continue
   734→
   735→                if self.RE_MENETAPKAN.search(text):
   736→                    self._flush_item()
   737→                    self.section = "keputusan"
   738→                    self.decision_keyword = "MENETAPKAN"
   739→                    self.current_item = None
   740→                    continue
   741→
   742→            if self.section == "pre":
   743→                continue
   744→
   745→            if self.section == "done":
   746→                continue
   747→
   748→            # v3: Signature block detection (broader)
   749→            text_lower = text.lower()
   750→            if ("ttd" in text_lower and len(text) < 30):
   751→                self._flush_item()
   752→                self.section = "done"
   753→                continue
   754→            # Also stop at formal signing language
   755→            if re.search(r'ditetapkan\s+di|dilantik\s+di', text_lower):
   756→                self._flush_item()
   757→                self.section = "done"
   758→                continue
   759→            # Stop at Penjelasan in Family B documents
   760→            if text_upper == "PENJELASAN":
   761→                self._flush_item()
   762→                self.section = "done"
   763→                continue
   764→
   765→            # In keputusan section: check for item markers
   766→            if self.section == "keputusan":
   767→                m_letter = self.RE_ITEM_LETTER.match(text)
   768→                m_number = self.RE_ITEM_NUMBER.match(text)
   769→                m_ordinal = self.RE_ORDINAL.match(text)
   770→                m_pasal = self.RE_PASAL_DIKTUM.match(text)
   771→
   772→                if m_pasal:
   773→                    # Fix #5: "Pasal X" acts as a diktum divider in some keppres
   774→                    self._flush_item()
   775→                    self.current_item = f"Pasal {m_pasal.group(1)}"
   776→                    continue
   777→
   778→                if m_letter or m_number or m_ordinal:
   779→                    self._flush_item()
   780→                    if m_ordinal:
   781→                        self.current_item = m_ordinal.group(1).upper()
   782→                    elif m_letter:
   783→                        self.current_item = m_letter.group(1)
   784→                    else:
   785→                        self.current_item = m_number.group(1)
   786→                    m = m_letter or m_number or m_ordinal
   787→                    remaining = text[m.end():].strip()
   788→                    if remaining:
   789→                        self.current_item_lines.append(remaining)
   790→                    continue
   791→
   792→                # Sub-items: accumulate under current item
   793→                if self.current_item:
   794→                    self.current_item_lines.append(text)
   795→                continue
   796→
   797→        self._flush_item()
   798→
   799→        # v7: Deduplicate short page-break artifacts (same as Family A)
   800→        self._dedup_short_artifacts()
   801→
   802→        return self.chunks

   804→    def _dedup_short_artifacts(self):

   806→        from collections import defaultdict
   807→        id_chunks = defaultdict(list)
   808→        for i, chunk in enumerate(self.chunks):
   809→            id_chunks[chunk["id"]].append(i)
   810→        
   811→        to_remove = set()
   812→        for chunk_id, indices in id_chunks.items():
   813→            if len(indices) < 2:
   814→                continue
   815→            pairs = [(i, len(self.chunks[i]["text"])) for i in indices]
   816→            pairs.sort(key=lambda x: -x[1])
   817→            longest_idx, longest_len = pairs[0]
   818→            
   819→            for idx, text_len in pairs[1:]:
   820→                shorter = self.chunks[idx]["text"].strip()
   821→                longer = self.chunks[longest_idx]["text"].strip()
   822→                if text_len < longest_len * 0.5:
   823→                    if longer.endswith(shorter) or shorter in longer:
   824→                        to_remove.add(idx)
   825→                if text_len < 80 and text_len < longest_len * 0.4:
   826→                    to_remove.add(idx)
   827→        
   828→        if to_remove:
   829→            self.chunks = [c for i, c in enumerate(self.chunks) if i not in to_remove]

   831→    def _flush_item(self):
   832→        if not self.current_item or not self.current_item_lines:
   833→            self.current_item_lines = []
   834→            return
   835→
   836→        text = ' '.join(self.current_item_lines).strip()
   837→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   838→            self.current_item_lines = []
   839→            return
   840→
   841→        kw = self.decision_keyword or "MEMUTUSKAN"
   842→        # v3: Include parent section in diktum ID for inpres (prevents numbering collision)
   843→        diktum_num = self.current_item

   845→            diktum_num = f"{self.parent_section}.{diktum_num}"

   847→                            chunk_type="diktum", chunk_num=diktum_num)
   848→
   849→        self.chunks.append({
   850→            "id": chunk_id,
   851→            "text": text,
   852→            "metadata": {
   853→                "doc_type": self.doc_type,
   854→                "issuer": self.issuer,
   855→                "year": self.year,
   856→                "nomor": self.nomor,
   857→                "section": "keputusan",
   858→                "decision_keyword": kw,
   859→                "diktum": self.current_item,
   860→                "parent": None,
   861→                "path": f"{kw} > Diktum {self.current_item}",
   862→                "status": "active",
   863→            }
   864→        })
   865→        self.current_item_lines = []
   866→
   867→    def _flush_preamble(self):
   868→        self.preamble_lines = []

   984→def extract_nomor_perkara(lines):

   993→def chunk_pdf(pdf_path: str, directory: str, family_map: Dict) -> List[Dict]:
   994→    """
   995→    Main entry point. Extract + parse a PDF into chunks.
   996→    Returns list of chunk dicts with id, text, metadata.
   997→    """
   998→    dir_config = family_map.get(directory, {})
