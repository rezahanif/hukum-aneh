   408→   437→   437→   536→   536→   536→    def _dedup_truncated_fragments(self):
   409→   438→   438→   537→   537→   537→        """v6: Remove truncated page-break fragments that duplicate a longer chunk.
   410→   439→   439→   538→   538→   538→        
   411→   440→   440→   539→   539→   539→        When text spans a page break, the bottom-of-page fragment (10-50 chars)
   412→   441→   441→   540→   540→   540→        gets the same pasal:ayat ID as the continuation on the next page.
   413→   442→   442→   541→   541→   541→        Keep the longer chunk, drop the fragment.
   414→   443→   443→   542→   542→   542→        """

   416→   445→   445→   544→   544→   544→        id_chunks = defaultdict(list)
   417→   446→   446→   545→   545→   545→        for i, chunk in enumerate(self.chunks):
   418→   447→   447→   546→   546→   546→            id_chunks[chunk["id"]].append(i)
   419→   448→   448→   547→   547→   547→        
   420→   449→   449→   548→   548→   548→        to_remove = set()
   421→   450→   450→   549→   549→   549→        for chunk_id, indices in id_chunks.items():
   422→   451→   451→   550→   550→   550→            if len(indices) < 2:
   423→   452→   452→   551→   551→   551→                continue
   424→   453→   453→   552→   552→   552→            pairs = [(i, len(self.chunks[i]["text"])) for i in indices]
   425→   454→   454→   553→   553→   553→            pairs.sort(key=lambda x: -x[1])
   426→   455→   455→   554→   554→   554→            longest_idx, longest_len = pairs[0]
   427→   456→   456→   555→   555→   555→            
   428→   457→   457→   556→   556→   556→            for idx, text_len in pairs[1:]:
   429→   458→   458→   557→   557→   557→                if text_len < longest_len * 0.5:
   430→   459→   459→   558→   558→   558→                    shorter = self.chunks[idx]["text"].strip()
   431→   460→   460→   559→   559→   559→                    longer = self.chunks[longest_idx]["text"].strip()
   432→   461→   461→   560→   560→   560→                    if longer.endswith(shorter) or shorter in longer:
   433→   462→   462→   561→   561→   561→                        to_remove.add(idx)
   434→   463→   463→   562→   562→   562→        
   435→   464→   464→   563→   563→   563→        if to_remove:
   436→   465→   465→   564→   564→   564→            self.chunks = [c for i, c in enumerate(self.chunks) if i not in to_remove]
   437→   466→   466→   565→   565→   565→
