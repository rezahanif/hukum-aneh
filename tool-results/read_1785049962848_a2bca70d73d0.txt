    17→function calcTitleLayout(title, maxWidthTwips, preferredPt, minPt) {
    18→  preferredPt = preferredPt || 40; minPt = minPt || 24;
    19→  const charWidth = (pt) => pt * 20;
    20→  const charsPerLine = (pt) => Math.floor(maxWidthTwips / charWidth(pt));
    21→  let titlePt = preferredPt, lines;
    22→  while (titlePt >= minPt) {
    23→    const cpl = charsPerLine(titlePt);
    24→    if (cpl < 2) { titlePt -= 2; continue; }
    25→    lines = splitTitleLines(title, cpl);
    26→    if (lines.length <= 3) break;
    27→    titlePt -= 2;
    28→  }
    29→  if (!lines || lines.length > 3) {
    30→    lines = splitTitleLines(title, charsPerLine(minPt));
    31→    titlePt = minPt;
    32→  }
    33→  return { titlePt, titleLines: lines };
    34→}
    35→
    36→function splitTitleLines(title, charsPerLine) {
    37→  if (title.length <= charsPerLine) return [title];
    38→  const breakAfter = new Set([' ', '-', '/', ':', '(', ')', '.']);
    39→  const lines = []; let remaining = title;
    40→  while (remaining.length > charsPerLine) {
    41→    let breakAt = -1;
    42→    for (let i = charsPerLine; i >= Math.floor(charsPerLine * 0.6); i--) {
    43→      if (i < remaining.length && breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    44→    }
    45→    if (breakAt === -1) breakAt = charsPerLine;
    46→    lines.push(remaining.slice(0, breakAt));
    47→    remaining = remaining.slice(breakAt);
    48→  }
    49→  if (remaining.length > 0) lines.push(remaining);
    50→  if (lines.length > 0 && lines[lines.length - 1].length <= 2 && lines.length > 1) {
    51→    lines[lines.length - 2] += " " + lines.pop();
    52→  }
    53→  return lines;
    54→}
