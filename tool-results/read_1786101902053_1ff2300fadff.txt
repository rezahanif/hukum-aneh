     1→   100→   100→   100→    paragraf_num_str = f"Paragraf {paragraf_num}" if paragraf_num else None
     2→   101→   101→   101→    if paragraf_num_str:
     3→   102→   102→   102→        headings.append(paragraf_num_str)
     4→   103→   103→   103→    if paragraf_label and paragraf_label != paragraf_num_str:
     5→   104→   104→   104→        headings.append(paragraf_label)
     6→   105→   105→   105→    
     7→   106→   106→   106→    if pasal_str:
     8→   107→   107→   107→        headings.append(pasal_str)
     9→   108→   108→   108→    if ayat_str:
    10→   109→   109→   109→        headings.append(ayat_str)
    11→   110→   110→   110→    
    12→   111→   111→   111→    return headings

    19→   118→   118→   118→class FamilyAParser:
    20→   119→   119→   119→    """
    21→   120→   120→   120→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
    22→   121→   121→   121→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
    23→   122→   122→   122→    """
    24→   123→   123→   123→
    25→   124→   124→   124→    # Heading patterns (order matters: higher levels first)
    26→   125→   125→   125→    # v6: allow '* ' prefix (UUD 1945 amendment markers)
    27→   126→   126→   126→    RE_BAB = re.compile(r'^\*?\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
    28→   127→   127→   127→    RE_BAGIAN = re.compile(r'^\*?\s*BAGIAN\s+(\w+)', re.IGNORECASE)
    29→   128→   128→   128→    RE_PARAGRAF = re.compile(r'^\*?\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
    30→   129→   129→   129→    # v6: capture alpha suffixes AND allow '* ' prefix (UUD 1945 amendment markers)
    31→   130→   130→   130→    RE_PASAL = re.compile(r'^\*?\s*pasal\s+([IVXLCDM]+|\d+(?:\s+\d+)*[a-zA-Z]?)\s*[.}\s]*$', re.IGNORECASE)
    32→   131→   131→   131→    # v6: Match pasal embedded mid-line (e.g., 'BAB VIIA ... * Pasal 22C')
    33→   132→   132→   132→    RE_PASAL_EMBEDDED = re.compile(r'\*\s*Pasal\s+(\d+[a-zA-Z]*)', re.IGNORECASE)
    34→   133→   133→   133→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
    35→   134→   134→   134→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
    36→   135→   135→   135→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
    37→   136→   136→   136→
    38→   137→   137→   137→    # Quoted amendment detection
    39→   138→   138→   138→    RE_AMENDMENT_TRIGGER = re.compile(
    40→   139→   139→   139→        r'(?:diubah|diganti|dihapus)\s+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
    41→   140→   140→   140→        re.IGNORECASE
    42→   141→   141→   141→    )
    43→   142→   142→   142→    RE_QUOTE_START = re.compile(r'^\s*["“]')
    44→   143→   143→   143→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
    45→   144→   144→   144→
    46→   145→   145→   145→    # Penutup (closing section) marker
    47→   146→   146→   146→    RE_PENUTUP = re.compile(r'^(?:KETENTUAN\s+PENUTUP|PENUTUP|penutup)\b')
    48→   147→   147→   147→    # v3: Penjelasan (elucidation) marker — can appear after penutup or directly after last pasal
    49→   148→   148→   148→    RE_PENJELASAN = re.compile(r'^(?:PENJELASAN|penjelasan)\b')
    50→   149→   149→   149→
    51→   150→   150→   150→    def __init__(self, lines, doc_type, nomor, year, issuer):
    52→   151→   151→   151→        self.lines = lines
    53→   152→   152→   152→        self.doc_type = doc_type
    54→   153→   153→   153→        self.nomor = nomor
    55→   154→   154→   154→        self.year = year
    56→   155→   155→   155→        self.issuer = issuer
    57→   156→   156→   156→        self.chunks = []
    58→   157→   157→   157→
    59→   158→   158→   158→        # Heading stack
    60→   159→   159→   159→        self.current_bab = None
    61→   160→   160→   160→        self.current_bab_label = None
    62→   161→   161→   161→        self.current_bagian = None
    63→   162→   162→   162→        self.current_bagian_label = None
    64→   163→   163→   163→        self.current_paragraf = None
    65→   164→   164→   164→        self.current_paragraf_label = None
    66→   165→   165→   165→        self.current_pasal = None
    67→   166→   166→   166→        self.current_ayat = None
    68→   167→   167→   167→
    69→   168→   168→   168→        # State
    70→   169→   169→   169→        self.in_preamble = True
    71→   170→   170→   170→        self.in_penutup = False
    72→   171→   171→   171→        self.in_penjelasan = False  # v3: track elucidation section
    73→   172→   172→   172→        self.in_quoted_amendment = False
    74→   173→   173→   173→        self.quoted_amendment_source = None  # tracks which pasal is being amended
    75→   174→   174→   174→        self.current_ayat_lines = []
    76→   175→   175→   175→        self.current_pasal_lines = []
    77→   176→   176→   176→        self.quoted_amendment_lines = []  # accumulate quoted text for chunking
    78→   177→   177→   177→        self._penjelasan_lines = []  # v3: accumulate Penjelasan text
    79→   178→   178→   178→        self._penjelasan_pasal = None
    80→   179→   179→   179→
    81→   180→   180→   180→    def parse(self) -> List[Dict]:
    82→   181→   181→   181→        """Main parse loop. Returns list of chunks."""
    83→   182→   182→   182→        for i, line in enumerate(self.lines):
    84→   183→   183→   183→            text = line["text"].strip()
    85→   184→   184→   184→            if not text:
    86→   185→   185→   185→                continue
    87→   186→   186→   186→
    88→   187→   187→   187→            text_upper = text.upper()
    89→   188→   188→   188→
    90→   189→   189→   189→            # Detect preamble end / body start
    91→   190→   190→   190→            if self.in_preamble:
    92→   191→   191→   191→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
    93→   192→   192→   192→                    self.in_preamble = False
    94→   193→   193→   193→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
    95→   194→   194→   194→                    self.in_preamble = False
    96→   195→   195→   195→                    continue
    97→   196→   196→   196→                else:
    98→   197→   197→   197→                    continue
    99→   198→   198→   198→
   100→   199→   199→   199→            # Detect penutup
   101→   200→   200→   200→            if self.RE_PENUTUP.match(text):
   102→   201→   201→   201→                self._flush_ayat()
   103→   202→   202→   202→                self._flush_pasal()
   104→   203→   203→   203→                self._flush_quoted_amendment()
   105→   204→   204→   204→                self.in_penutup = True
   106→   205→   205→   205→                continue
   107→   206→   206→   206→
   108→   207→   207→   207→            # v3: Detect PENJELASAN even without penutup (many docs go straight to penjelasan)
   109→   208→   208→   208→            if self.RE_PENJELASAN.match(text) and not self.in_penjelasan:
   110→   209→   209→   209→                self._flush_ayat()
   111→   210→   210→   210→                self._flush_pasal()
   112→   211→   211→   211→                self._flush_quoted_amendment()
   113→   212→   212→   212→                self.in_penutup = True  # treat as post-operative
   114→   213→   213→   213→                self.in_penjelasan = True
   115→   214→   214→   214→                self._penjelasan_pasal = None
   116→   215→   215→   215→                continue
   117→   216→   216→   216→
   118→   217→   217→   217→            if self.in_penutup:
   119→   218→   218→   218→                # v3: Detect signature block and Penjelasan as hard boundaries
   120→   219→   219→   219→                text_lower = text.lower()
   121→   220→   220→   220→                # Signature block: lines like "Ditetapkan di Jakarta", "pada tanggal", or ttd/
   122→   221→   221→   221→                # Also detect by very short lines after penutup start
   123→   222→   222→   222→                if (re.search(r'ttd|disahkan|ditetapkan', text_lower) or
   124→   223→   223→   223→                    (len(text) < 15 and text_upper == text and len(text) > 2)):
   125→   224→   224→   224→                    self._flush_penjelasan()  # flush any accumulated penjelasan
   126→   225→   225→   225→                    continue
   127→   226→   226→   226→                # PENJELASAN section header
   128→   227→   227→   227→                if text_upper == "PENJELASAN" or text_upper.startswith("PENJELASAN "):
   129→   228→   228→   228→                    self._flush_penjelasan()
   130→   229→   229→   229→                    self.in_penjelasan = True
   131→   230→   230→   230→                    self._penjelasan_pasal = None
   132→   231→   231→   231→                    continue
   133→   232→   232→   232→                # Inside Penjelasan: track pasal references for IDs
   134→   233→   233→   233→                if self.in_penjelasan:
   135→   234→   234→   234→                    # v5: also capture alpha suffixes in Penjelasan pasal references
   136→   235→   235→   235→                    m_pj = re.match(r'^\s*Pasal\s+(\d+[a-zA-Z]*)', text, re.IGNORECASE)
   137→   236→   236→   236→                    if m_pj:
   138→   237→   237→   237→                        self._flush_penjelasan()
   139→   238→   238→   238→                        self._penjelasan_pasal = m_pj.group(1)
   140→   239→   239→   239→                    else:
   141→   240→   240→   240→                        self._penjelasan_lines.append(text)
   142→   241→   241→   241→                continue
   143→   242→   242→   242→
   144→   243→   243→   243→            # Check for quoted amendment TRIGGER ("Pasal X diubah sehingga berbunyi:")
   145→   244→   244→   244→            if not self.in_quoted_amendment:
   146→   245→   245→   245→                if self.RE_AMENDMENT_TRIGGER.search(text):
   147→   246→   246→   246→                    # Extract which pasal is being amended
   148→   247→   247→   247→                    m_pasal = re.search(r'Pasal\s+(\d+|\w+)\s+(?:ayat\s*\(\d+\)\s+)?(?:diubah|diganti|dihapus)',
   149→   248→   248→   248→                                         text, re.IGNORECASE)
   150→   249→   249→   249→                    source_pasal = m_pasal.group(1) if m_pasal else "?"
   151→   250→   250→   250→                    self.in_quoted_amendment = True
   152→   251→   251→   251→                    self.quoted_amendment_source = source_pasal
   153→   252→   252→   252→                    self._flush_ayat()
   154→   253→   253→   253→                    # Store the trigger sentence as context for the quoted chunk
   155→   254→   254→   254→                    self.quoted_amendment_lines = [text]
   156→   255→   255→   255→                    continue
   157→   256→   256→   256→
   158→   257→   257→   257→            # Inside quoted amendment: accumulate text
   159→   258→   258→   258→            if self.in_quoted_amendment:
   160→   259→   259→   259→                self.quoted_amendment_lines.append(text)
   161→   260→   260→   260→                if self.RE_QUOTE_END.search(text):
   162→   261→   261→   261→                    self._flush_quoted_amendment()
   163→   262→   262→   262→                # Also check for end of quoted section without closing quote:
   164→   263→   263→   263→                # if we hit a new structural marker, the quote ended implicitly
   165→   264→   264→   264→                if (self.RE_BAB.match(text) or self.RE_PASAL.match(text) or
   166→   265→   265→   265→                    self.RE_PENUTUP.match(text)):
   167→   266→   266→   266→                    self._flush_quoted_amendment()
   168→   267→   267→   267→                    # Don't skip the structural marker - let it be processed below
   169→   268→   268→   268→                    self.in_quoted_amendment = False
   170→   269→   269→   269→                elif self.RE_QUOTE_END.search(text):
   171→   270→   270→   270→                    pass  # already flushed above
   172→   271→   271→   271→                else:
   173→   272→   272→   272→                    continue
   174→   273→   273→   273→
   175→   274→   274→   274→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   176→   275→   275→   275→            m = self.RE_BAB.match(text)
   177→   276→   276→   276→            if m:
   178→   277→   277→   277→                self._flush_ayat()
   179→   278→   278→   278→                self._flush_pasal()
   180→   279→   279→   279→                self.current_bab = m.group(1)
   181→   280→   280→   280→                label_match = re.match(r'^\*?\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   182→   281→   281→   281→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   183→   282→   282→   282→                # Fix #2: if label is empty or just repeats "BAB X", set to None
   184→   283→   283→   283→                if raw_label and raw_label.upper() != f"BAB {m.group(1).upper()}":
   185→   284→   284→   284→                    self.current_bab_label = raw_label
   186→   285→   285→   285→                else:
   187→   286→   286→   286→                    self.current_bab_label = None
   188→   287→   287→   287→                self.current_bagian = None
   189→   288→   288→   288→                self.current_bagian_label = None
   190→   289→   289→   289→                self.current_paragraf = None
   191→   290→   290→   290→                self.current_paragraf_label = None
   192→   291→   291→   291→                # v4: reset pasal when BAB changes to prevent heading-as-chunk
   193→   292→   292→   292→                self.current_pasal = None
   194→   293→   293→   293→                # v6: Check if a Pasal reference is glued to the BAB line
   195→   294→   294→   294→                # (e.g., 'BAB VIIA DEWAN PERWAKILAN DAERAH * Pasal 22C')
   196→   295→   295→   295→                m_glued = self.RE_PASAL_EMBEDDED.search(text)
   197→   296→   296→   296→                if m_glued:
   198→   297→   297→   297→                    self.current_pasal = m_glued.group(1)
   199→   298→   298→   298→                    self.current_ayat = None
   200→   299→   299→   299→                continue
   201→   300→   300→   300→
   202→   301→   301→   301→            m = self.RE_BAGIAN.match(text)
   203→   302→   302→   302→            if m:
   204→   303→   303→   303→                self._flush_ayat()
   205→   304→   304→   304→                self._flush_pasal()
   206→   305→   305→   305→                self.current_bagian = m.group(1)
   207→   306→   306→   306→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)

   209→   308→   308→   308→                if raw_label:
   210→   309→   309→   309→                    self.current_bagian_label = raw_label
   211→   310→   310→   310→                else:
   212→   311→   311→   311→                    self.current_bagian_label = None
   213→   312→   312→   312→                self.current_paragraf = None
   214→   313→   313→   313→                self.current_paragraf_label = None
   215→   314→   314→   314→                continue
   216→   315→   315→   315→
   217→   316→   316→   316→            m = self.RE_PARAGRAF.match(text)
   218→   317→   317→   317→            if m:
   219→   318→   318→   318→                self._flush_ayat()
   220→   319→   319→   319→                self._flush_pasal()
   221→   320→   320→   320→                self.current_paragraf = m.group(1)
   222→   321→   321→   321→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   223→   322→   322→   322→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   224→   323→   323→   323→                if raw_label:
   225→   324→   324→   324→                    self.current_paragraf_label = raw_label
   226→   325→   325→   325→                else:
   227→   326→   326→   326→                    self.current_paragraf_label = None
   228→   327→   327→   327→                continue
   229→   328→   328→   328→
   230→   329→   329→   329→            m = self.RE_PASAL.match(text)
   231→   330→   330→   330→            if m:
   232→   331→   331→   331→                self._flush_ayat()
   233→   332→   332→   332→                self._flush_pasal()
   234→   333→   333→   333→                raw_num = m.group(1)
   235→   334→   334→   334→                # v4: normalize space-separated digits ("1 1" -> "11", "1 10" -> "110")
   236→   335→   335→   335→                self.current_pasal = re.sub(r'\s+', '', raw_num) if re.match(r'^\d', raw_num) else raw_num
   237→   336→   336→   336→                self.current_ayat = None

   240→   339→   339→   339→                if remaining and len(remaining) > 5:
   241→   340→   340→   340→                    self.current_pasal_lines.append(remaining)
   242→   341→   341→   341→                continue

   244→   343→   343→   343→            m = self.RE_AYAT.match(text)

   246→   345→   345→   345→                self._flush_ayat()
   247→   346→   346→   346→                self.current_ayat = m.group(1)
   248→   347→   347→   347→                ayat_text = self.RE_AYAT.sub('', text).strip()
   249→   348→   348→   348→                if ayat_text:
   250→   349→   349→   349→                    self.current_ayat_lines.append(ayat_text)
   251→   350→   350→   350→                continue
   252→   351→   351→   351→
   253→   352→   352→   352→            # v6: Check for embedded pasal in non-structural line
   254→   353→   353→   353→            # (e.g., 'PEMILIHAN UMUM * Pasal 22E' — BAB label with glued pasal)
   255→   354→   354→   354→            m_emb = self.RE_PASAL_EMBEDDED.search(text)
   256→   355→   355→   355→            if m_emb and self.current_pasal:
   257→   356→   356→   356→                self._flush_ayat()
   258→   357→   357→   357→                self._flush_pasal()
   259→   358→   358→   358→                self.current_pasal = m_emb.group(1)
   260→   359→   359→   359→                self.current_ayat = None
   261→   360→   360→   360→                continue
   262→   361→   361→   361→
   263→   362→   362→   362→            # If we're inside a pasal but no ayat detected, accumulate lines
   264→   363→   363→   363→            if self.current_pasal:
   265→   364→   364→   364→                if self.current_ayat:
   266→   365→   365→   365→                    self.current_ayat_lines.append(text)
   267→   366→   366→   366→                else:
   268→   367→   367→   367→                    self.current_pasal_lines.append(text)
   269→   368→   368→   368→
   270→   369→   369→   369→        # Flush remaining
   271→   370→   370→   370→        self._flush_ayat()
   272→   371→   371→   371→        self._flush_pasal()
   273→   372→   372→   372→        self._flush_quoted_amendment()
   274→   373→   373→   373→
   275→   374→   374→   374→        # v6: Deduplicate truncated page-break fragments
   276→   375→   375→   375→        self._dedup_truncated_fragments()
   277→   376→   376→   376→
   278→   377→   377→   377→        return self.chunks
   279→   378→   378→   378→
   280→   379→   379→   379→    def _flush_ayat(self):
   281→   380→   380→   380→        """Flush accumulated ayat lines as a chunk."""
   282→   381→   381→   381→        if not self.current_ayat or not self.current_ayat_lines:
   283→   382→   382→   382→            self.current_ayat_lines = []
   284→   383→   383→   383→            return
   285→   384→   384→   384→
   286→   385→   385→   385→        text = ' '.join(self.current_ayat_lines).strip()
   287→   386→   386→   386→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:

   290→   389→   389→   389→

   303→   402→   402→   402→

   305→   404→   404→   404→            "id": chunk_id,
   306→   405→   405→   405→            "text": text,
   307→   406→   406→   406→            "metadata": {

   437→   536→   536→   536→    def _dedup_truncated_fragments(self):
   438→   537→   537→   537→        """v6: Remove truncated page-break fragments that duplicate a longer chunk.
   439→   538→   538→   538→        
   440→   539→   539→   539→        When text spans a page break, the bottom-of-page fragment (10-50 chars)
   441→   540→   540→   540→        gets the same pasal:ayat ID as the continuation on the next page.
   442→   541→   541→   541→        Keep the longer chunk, drop the fragment.
   443→   542→   542→   542→        """
   444→   543→   543→   543→        from collections import defaultdict
   445→   544→   544→   544→        id_chunks = defaultdict(list)
   446→   545→   545→   545→        for i, chunk in enumerate(self.chunks):
   447→   546→   546→   546→            id_chunks[chunk["id"]].append(i)
   448→   547→   547→   547→        
   449→   548→   548→   548→        to_remove = set()
   450→   549→   549→   549→        for chunk_id, indices in id_chunks.items():
   451→   550→   550→   550→            if len(indices) < 2:
   452→   551→   551→   551→                continue
   453→   552→   552→   552→            pairs = [(i, len(self.chunks[i]["text"])) for i in indices]
   454→   553→   553→   553→            pairs.sort(key=lambda x: -x[1])
   455→   554→   554→   554→            longest_idx, longest_len = pairs[0]
   456→   555→   555→   555→            
   457→   556→   556→   556→            for idx, text_len in pairs[1:]:
   458→   557→   557→   557→                if text_len < longest_len * 0.5:
   459→   558→   558→   558→                    shorter = self.chunks[idx]["text"].strip()
   460→   559→   559→   559→                    longer = self.chunks[longest_idx]["text"].strip()
   461→   560→   560→   560→                    if longer.endswith(shorter) or shorter in longer:
   462→   561→   561→   561→                        to_remove.add(idx)
   463→   562→   562→   562→        
   464→   563→   563→   563→        if to_remove:
   465→   564→   564→   564→            self.chunks = [c for i, c in enumerate(self.chunks) if i not in to_remove]
   466→   565→   565→   565→
