    43→    43→    """Build chunk ID. Handles both hierarchical and flat structures."""
    44→    44→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    45→    45→        nomor_perkara = parts.get("nomor_perkara", nomor or "?")

    77→    77→def _make_bab_headings(bab_num, bab_label, bagian_num, bagian_label,

    79→    79→    """Build heading list for path, avoiding duplicate BAB labels.
    80→    80→    
    81→    81→    Fix for bug #2: when BAB has no descriptive label (e.g., just "BAB I"),
    82→    82→    the label defaults to "BAB I" which duplicates the number element.
    83→    83→    Now: if label equals the number element, omit it.
    84→    84→    """
    85→    85→    headings = []
    86→    86→    
    87→    87→    bab_num_str = f"BAB {bab_num}" if bab_num else None
    88→    88→    if bab_num_str:
    89→    89→        headings.append(bab_num_str)
    90→    90→    # Only add label if it's different from the number element and non-empty
    91→    91→    if bab_label and bab_label != bab_num_str:
    92→    92→        headings.append(bab_label)
    93→    93→    
    94→    94→    bagian_num_str = f"Bagian {bagian_num}" if bagian_num else None
    95→    95→    if bagian_num_str:
    96→    96→        headings.append(bagian_num_str)
    97→    97→    if bagian_label and bagian_label != bagian_num_str:
    98→    98→        headings.append(bagian_label)
    99→    99→    
   100→   100→    paragraf_num_str = f"Paragraf {paragraf_num}" if paragraf_num else None
   101→   101→    if paragraf_num_str:
   102→   102→        headings.append(paragraf_num_str)
   103→   103→    if paragraf_label and paragraf_label != paragraf_num_str:
   104→   104→        headings.append(paragraf_label)
   105→   105→    
   106→   106→    if pasal_str:
   107→   107→        headings.append(pasal_str)
   108→   108→    if ayat_str:
   109→   109→        headings.append(ayat_str)
   110→   110→    
   111→   111→    return headings

   114→   114→# ==========================================================================
   115→   115→# FAMILY A: HIERARCHICAL STATUTE PARSER
   116→   116→# ==========================================================================
   117→   117→
   118→   118→class FamilyAParser:
   119→   119→    """
   120→   120→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
   121→   121→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
   122→   122→    """
   123→   123→
   124→   124→    # Heading patterns (order matters: higher levels first)
   125→   125→    RE_BAB = re.compile(r'^\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
   126→   126→    RE_BAGIAN = re.compile(r'^\s*BAGIAN\s+(\w+)', re.IGNORECASE)
   127→   127→    RE_PARAGRAF = re.compile(r'^\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
   128→   128→    RE_PASAL = re.compile(r'^\s*pasal\s+([IVXLCDM]+|\d+[a-zA-Z]?)\s*\.?\s*.{0,5}$', re.IGNORECASE)
   129→   129→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
   130→   130→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
   131→   131→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
   132→   132→
   133→   133→    # Quoted amendment detection
   134→   134→    RE_AMENDMENT_TRIGGER = re.compile(
   135→   135→        r'(?:diubah|diganti|dihapus)\s+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
   136→   136→        re.IGNORECASE
   137→   137→    )
   138→   138→    RE_QUOTE_START = re.compile(r'^\s*["“]')
   139→   139→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
   140→   140→
   141→   141→    # Penutup (closing section) marker
   142→   142→    RE_PENUTUP = re.compile(r'^(?:PENUTUP|penutup)\b')
   143→   143→
   144→   144→    def __init__(self, lines, doc_type, nomor, year, issuer):
   145→   145→        self.lines = lines
   146→   146→        self.doc_type = doc_type
   147→   147→        self.nomor = nomor
   148→   148→        self.year = year
   149→   149→        self.issuer = issuer
   150→   150→        self.chunks = []
   151→   151→
   152→   152→        # Heading stack
   153→   153→        self.current_bab = None
   154→   154→        self.current_bab_label = None
   155→   155→        self.current_bagian = None
   156→   156→        self.current_bagian_label = None
   157→   157→        self.current_paragraf = None
   158→   158→        self.current_paragraf_label = None
   159→   159→        self.current_pasal = None
   160→   160→        self.current_ayat = None
   161→   161→
   162→   162→        # State
   163→   163→        self.in_preamble = True
   164→   164→        self.in_penutup = False
   165→   165→        self.in_quoted_amendment = False
   166→   166→        self.quoted_amendment_source = None  # tracks which pasal is being amended
   167→   167→        self.current_ayat_lines = []
   168→   168→        self.current_pasal_lines = []
   169→   169→        self.quoted_amendment_lines = []  # accumulate quoted text for chunking
   170→   170→
   171→   171→    def parse(self) -> List[Dict]:
   172→   172→        """Main parse loop. Returns list of chunks."""
   173→   173→        for i, line in enumerate(self.lines):
   174→   174→            text = line["text"].strip()
   175→   175→            if not text:
   176→   176→                continue
   177→   177→
   178→   178→            text_upper = text.upper()
   179→   179→
   180→   180→            # Detect preamble end / body start
   181→   181→            if self.in_preamble:
   182→   182→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   183→   183→                    self.in_preamble = False
   184→   184→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
   185→   185→                    self.in_preamble = False
   186→   186→                    continue
   187→   187→                else:
   188→   188→                    continue
   189→   189→
   190→   190→            # Detect penutup
   191→   191→            if self.RE_PENUTUP.match(text):
   192→   192→                self._flush_ayat()
   193→   193→                self._flush_pasal()
   194→   194→                self._flush_quoted_amendment()
   195→   195→                self.in_penutup = True
   196→   196→                continue
   197→   197→
   198→   198→            if self.in_penutup:
   199→   199→                continue
   200→   200→
   201→   201→            # Check for quoted amendment TRIGGER ("Pasal X diubah sehingga berbunyi:")
   202→   202→            if not self.in_quoted_amendment:
   203→   203→                if self.RE_AMENDMENT_TRIGGER.search(text):
   204→   204→                    # Extract which pasal is being amended
   205→   205→                    m_pasal = re.search(r'Pasal\s+(\d+|\w+)\s+(?:ayat\s*\(\d+\)\s+)?(?:diubah|diganti|dihapus)',
   206→   206→                                         text, re.IGNORECASE)
   207→   207→                    source_pasal = m_pasal.group(1) if m_pasal else "?"
   208→   208→                    self.in_quoted_amendment = True
   209→   209→                    self.quoted_amendment_source = source_pasal
   210→   210→                    self._flush_ayat()
   211→   211→                    # Store the trigger sentence as context for the quoted chunk
   212→   212→                    self.quoted_amendment_lines = [text]
   213→   213→                    continue
   214→   214→
   215→   215→            # Inside quoted amendment: accumulate text
   216→   216→            if self.in_quoted_amendment:
   217→   217→                self.quoted_amendment_lines.append(text)
   218→   218→                if self.RE_QUOTE_END.search(text):
   219→   219→                    self._flush_quoted_amendment()
   220→   220→                # Also check for end of quoted section without closing quote:
   221→   221→                # if we hit a new structural marker, the quote ended implicitly
   222→   222→                if (self.RE_BAB.match(text) or self.RE_PASAL.match(text) or
   223→   223→                    self.RE_PENUTUP.match(text)):
   224→   224→                    self._flush_quoted_amendment()
   225→   225→                    # Don't skip the structural marker - let it be processed below
   226→   226→                    self.in_quoted_amendment = False
   227→   227→                elif self.RE_QUOTE_END.search(text):
   228→   228→                    pass  # already flushed above
   229→   229→                else:
   230→   230→                    continue
   231→   231→
   232→   232→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   233→   233→            m = self.RE_BAB.match(text)
   234→   234→            if m:
   235→   235→                self._flush_ayat()
   236→   236→                self._flush_pasal()
   237→   237→                self.current_bab = m.group(1)
   238→   238→                label_match = re.match(r'^\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   239→   239→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   240→   240→                # Fix #2: if label is empty or just repeats "BAB X", set to None
   241→   241→                if raw_label and raw_label.upper() != f"BAB {m.group(1).upper()}":
   242→   242→                    self.current_bab_label = raw_label
   243→   243→                else:
   244→   244→                    self.current_bab_label = None
   245→   245→                self.current_bagian = None
   246→   246→                self.current_bagian_label = None
   247→   247→                self.current_paragraf = None
   248→   248→                self.current_paragraf_label = None
   249→   249→                continue
   250→   250→
   251→   251→            m = self.RE_BAGIAN.match(text)
   252→   252→            if m:
   253→   253→                self._flush_ayat()
   254→   254→                self._flush_pasal()
   255→   255→                self.current_bagian = m.group(1)
   256→   256→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   257→   257→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   258→   258→                if raw_label:
   259→   259→                    self.current_bagian_label = raw_label
   260→   260→                else:
   261→   261→                    self.current_bagian_label = None
   262→   262→                self.current_paragraf = None
   263→   263→                self.current_paragraf_label = None
   264→   264→                continue
   265→   265→
   266→   266→            m = self.RE_PARAGRAF.match(text)
   267→   267→            if m:
   268→   268→                self._flush_ayat()
   269→   269→                self._flush_pasal()
   270→   270→                self.current_paragraf = m.group(1)
   271→   271→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   272→   272→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   273→   273→                if raw_label:
   274→   274→                    self.current_paragraf_label = raw_label
   275→   275→                else:
   276→   276→                    self.current_paragraf_label = None
   277→   277→                continue
   278→   278→
   279→   279→            m = self.RE_PASAL.match(text)
   280→   280→            if m:
   281→   281→                self._flush_ayat()
   282→   282→                self._flush_pasal()
   283→   283→                self.current_pasal = m.group(1)
   284→   284→                self.current_ayat = None
   285→   285→                continue
   286→   286→
   287→   287→            m = self.RE_AYAT.match(text)
   288→   288→            if m and self.current_pasal:
   289→   289→                self._flush_ayat()
   290→   290→                self.current_ayat = m.group(1)
   291→   291→                ayat_text = self.RE_AYAT.sub('', text).strip()
   292→   292→                if ayat_text:
   293→   293→                    self.current_ayat_lines.append(ayat_text)
   294→   294→                continue
   295→   295→
   296→   296→            # If we're inside a pasal but no ayat detected, accumulate lines
   297→   297→            if self.current_pasal:
   298→   298→                if self.current_ayat:
   299→   299→                    self.current_ayat_lines.append(text)
   300→   300→                else:
   301→   301→                    self.current_pasal_lines.append(text)
   302→   302→
   303→   303→        # Flush remaining
   304→   304→        self._flush_ayat()
   305→   305→        self._flush_pasal()
   306→   306→        self._flush_quoted_amendment()
   307→   307→
   308→   308→        return self.chunks
   309→   309→
   310→   310→    def _flush_ayat(self):
   311→   311→        """Flush accumulated ayat lines as a chunk."""
   312→   312→        if not self.current_ayat or not self.current_ayat_lines:
   313→   313→            self.current_ayat_lines = []
   314→   314→            return
   315→   315→
   316→   316→        text = ' '.join(self.current_ayat_lines).strip()
   317→   317→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   318→   318→            self.current_ayat_lines = []
   319→   319→            return
   320→   320→
   321→   321→        headings = _make_bab_headings(
   322→   322→            self.current_bab, self.current_bab_label,
   323→   323→            self.current_bagian, self.current_bagian_label,
   324→   324→            self.current_paragraf, self.current_paragraf_label,
   325→   325→            pasal_str=f"Pasal {self.current_pasal}",
   326→   326→            ayat_str=f"Ayat ({self.current_ayat})"
   327→   327→        )
   328→   328→
   329→   329→        path = build_path(headings)
   330→   330→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   331→   331→                            pasal=self.current_pasal, ayat=self.current_ayat)
   332→   332→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   333→   333→
   334→   334→        self.chunks.append({
   335→   335→            "id": chunk_id,
   336→   336→            "text": text,
   337→   337→            "metadata": {
   338→   338→                "doc_type": self.doc_type,
   339→   339→                "issuer": self.issuer,
   340→   340→                "year": self.year,
   341→   341→                "bab": self.current_bab or None,
   342→   342→                "pasal": self.current_pasal,
   343→   343→                "ayat": self.current_ayat,
   344→   344→                "parent": parent_id,
   345→   345→                "path": path,
   346→   346→                "status": "active",
   347→   347→            }
   348→   348→        })
   349→   349→        self.current_ayat_lines = []
   350→   350→
   351→   351→    def _flush_pasal(self):
   352→   352→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   353→   353→        if not self.current_pasal or not self.current_pasal_lines:
   354→   354→            self.current_pasal_lines = []
   355→   355→            return
   356→   356→
   357→   357→        text = ' '.join(self.current_pasal_lines).strip()
   358→   358→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   359→   359→            self.current_pasal_lines = []
   360→   360→            return
   361→   361→
   362→   362→        headings = _make_bab_headings(
   363→   363→            self.current_bab, self.current_bab_label,
   364→   364→            self.current_bagian, self.current_bagian_label,
   365→   365→            self.current_paragraf, self.current_paragraf_label,
   366→   366→            pasal_str=f"Pasal {self.current_pasal}",
   367→   367→        )
   368→   368→        path = build_path(headings)
   369→   369→        chunk_id = build_id(self.doc_type, self.nomor, self.year, pasal=self.current_pasal)
   370→   370→
   371→   371→        self.chunks.append({
   372→   372→            "id": chunk_id,
   373→   373→            "text": text,
   374→   374→            "metadata": {
   375→   375→                "doc_type": self.doc_type,
   376→   376→                "issuer": self.issuer,
   377→   377→                "year": self.year,
   378→   378→                "bab": self.current_bab or None,
   379→   379→                "pasal": self.current_pasal,
   380→   380→                "ayat": None,
   381→   381→                "parent": None,
   382→   382→                "path": path,
   383→   383→                "status": "active",
   384→   384→            }
   385→   385→        })
   386→   386→        self.current_pasal_lines = []
   387→   387→
   388→   388→    def _flush_quoted_amendment(self):
   389→   389→        """Flush quoted amendment text as a chunk with status='quoted_amendment'.
   390→   390→        
   391→   391→        Fix for bug #6: instead of silently skipping quoted amendments,
   392→   392→        we now chunk them with a clear status tag so retrieval can
   393→   393→        distinguish live law from historical amendment pointers.
   394→   394→        """
   395→   395→        if not self.quoted_amendment_lines:
   396→   396→            self.in_quoted_amendment = False
   397→   397→            self.quoted_amendment_source = None
   398→   398→            return
   399→   399→
   400→   400→        text = ' '.join(self.quoted_amendment_lines).strip()
   401→   401→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   402→   402→            self.quoted_amendment_lines = []
   403→   403→            self.in_quoted_amendment = False
   404→   404→            self.quoted_amendment_source = None
   405→   405→            return
   406→   406→
   407→   407→        source = self.quoted_amendment_source or "?"
   408→   408→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   409→   409→                            pasal=source, ayat="quoted")
   410→   410→
   411→   411→        self.chunks.append({
   412→   412→            "id": chunk_id,
   413→   413→            "text": text,
   414→   414→            "metadata": {
   415→   415→                "doc_type": self.doc_type,
   416→   416→                "issuer": self.issuer,
   417→   417→                "year": self.year,
   418→   418→                "bab": self.current_bab or None,
   419→   419→                "pasal": source,
   420→   420→                "ayat": "quoted",
   421→   421→                "parent": None,
   422→   422→                "path": f"[AMANDEMEN Pasal {source}]",
   423→   423→                "status": "quoted_amendment",
   424→   424→                "amends_pasal": source,
   425→   425→            }
   426→   426→        })
   427→   427→        self.quoted_amendment_lines = []
   428→   428→        self.in_quoted_amendment = False

   436→   436→class FamilyBParser:
   437→   437→    """
   438→   438→    Parses decree/decision documents: Menimbang > Mengingat > MEMUTUSKAN/MENETAPKAN > Diktum.
   439→   439→    Handles: Keppres, Inpres, Tap_MPR, Kepmen.
   440→   440→    
   441→   441→    v2 fix: Recognizes "Pasal X" as diktum divider (keppres pattern where
   442→   442→    diktum items are organized under Pasal headings instead of numbered).
   443→   443→    """
   444→   444→
   445→   445→    RE_MENIMBANG = re.compile(r'^\s*menimbang\s*:?\s*$', re.IGNORECASE)
   446→   446→    RE_MENGINGAT = re.compile(r'^\s*mengingat\s*:?\s*$', re.IGNORECASE)
   447→   447→    # Decision keywords: use $ anchor to match at END of line, allowing any prefix
   448→   448→    RE_MEMUTUSKAN = re.compile(r'memutuskan\s*:?\s*$', re.IGNORECASE)
   449→   449→    RE_MENETAPKAN = re.compile(r'menetapkan\s*:?\s*$', re.IGNORECASE)
   450→   450→    RE_MENINSTRUKSIKAN = re.compile(r'menginstruksikan\s*:?\s*$', re.IGNORECASE)
   451→   451→    RE_ITEM_LETTER = re.compile(r'^\s*([a-z])\s*\.\s*')
   452→   452→    RE_ITEM_NUMBER = re.compile(r'^\s*(\d+)\s*\.\s*')
   453→   453→    RE_ORDINAL = re.compile(
   454→   454→        r'^\s*(KE(?:SATU|DUA|TIGA|EMPAT|LIMA|NAM|TUJUH|DELAPAN|SEMBILAN|SEPULUH|'
   455→   455→        r'SEBELAS|DUA BELAS|TIGA BELAS|EMPAT BELAS|LIMA BELAS|ENAM BELAS|'
   456→   456→        r'TUJUH BELAS|DELAPAN BELAS|SEMBILAN BELAS|DUA PULUH))\b',
   457→   457→        re.IGNORECASE
   458→   458→    )
   459→   459→    # Fix #5: recognize "Pasal X" as diktum divider in Family B
   460→   460→    RE_PASAL_DIKTUM = re.compile(r'^\s*Pasal\s+(\d+[a-zA-Z]?)\s*\.?\s*$', re.IGNORECASE)
   461→   461→
   462→   462→    def __init__(self, lines, doc_type, nomor, year, issuer):
   463→   463→        self.lines = lines
   464→   464→        self.doc_type = doc_type
   465→   465→        self.nomor = nomor
   466→   466→        self.year = year
   467→   467→        self.issuer = issuer
   468→   468→        self.chunks = []
   469→   469→
   470→   470→        self.section = "pre"  # pre, menimbang, mengingat, keputusan, done
   471→   471→        self.current_item = None
   472→   472→        self.current_item_lines = []
   473→   473→        self.preamble_lines = []
   474→   474→        self.decision_keyword = None

   476→   476→    def parse(self) -> List[Dict]:
   477→   477→        for line in self.lines:
   478→   478→            text = line["text"].strip()
   479→   479→            if not text:
   480→   480→                continue
   481→   481→
   482→   482→            text_upper = text.upper()
   483→   483→
   484→   484→            # Section transitions
   485→   485→            if self.RE_MENIMBANG.match(text):
   486→   486→                self._flush_preamble()
   487→   487→                self.section = "menimbang"
   488→   488→                self.current_item = None
   489→   489→                continue
   490→   490→
   491→   491→            if self.RE_MENGINGAT.match(text):
   492→   492→                self._flush_preamble()
   493→   493→                self.section = "mengingat"
   494→   494→                self.current_item = None
   495→   495→                continue
   496→   496→
   497→   497→            # Decision keyword triggers (use search() for end-anchored patterns)
   498→   498→            if self.section not in ("keputusan", "done"):
   499→   499→                if self.RE_MENINSTRUKSIKAN.search(text):
   500→   500→                    self._flush_item()
   501→   501→                    self.section = "keputusan"
   502→   502→                    self.decision_keyword = "MENGINSTRUKSIKAN"
   503→   503→                    self.current_item = None
   504→   504→                    continue
   505→   505→
   506→   506→                if self.RE_MEMUTUSKAN.search(text):
   507→   507→                    self._flush_item()
   508→   508→                    self.section = "keputusan"
   509→   509→                    self.decision_keyword = "MEMUTUSKAN"
   510→   510→                    self.current_item = None
   511→   511→                    continue
   512→   512→
   513→   513→                if self.RE_MENETAPKAN.search(text):
   514→   514→                    self._flush_item()
   515→   515→                    self.section = "keputusan"
   516→   516→                    self.decision_keyword = "MENETAPKAN"
   517→   517→                    self.current_item = None
   518→   518→                    continue
   519→   519→
   520→   520→            if self.section == "pre":
   521→   521→                continue
   522→   522→
   523→   523→            if self.section == "done":
   524→   524→                continue
   525→   525→
   526→   526→            # Signature block detection
   527→   527→            text_lower = text.lower()
   528→   528→            if "ttd" in text_lower and len(text) < 20:
   529→   529→                self._flush_item()
   530→   530→                self.section = "done"
   531→   531→                continue
   532→   532→
   533→   533→            # In keputusan section: check for item markers
   534→   534→            if self.section == "keputusan":
   535→   535→                m_letter = self.RE_ITEM_LETTER.match(text)
   536→   536→                m_number = self.RE_ITEM_NUMBER.match(text)
   537→   537→                m_ordinal = self.RE_ORDINAL.match(text)
   538→   538→                m_pasal = self.RE_PASAL_DIKTUM.match(text)
   539→   539→
   540→   540→                if m_pasal:
   541→   541→                    # Fix #5: "Pasal X" acts as a diktum divider in some keppres
   542→   542→                    self._flush_item()
   543→   543→                    self.current_item = f"Pasal {m_pasal.group(1)}"
   544→   544→                    continue
   545→   545→
   546→   546→                if m_letter or m_number or m_ordinal:
   547→   547→                    self._flush_item()
   548→   548→                    if m_ordinal:
   549→   549→                        self.current_item = m_ordinal.group(1).upper()
   550→   550→                    elif m_letter:
   551→   551→                        self.current_item = m_letter.group(1)
   552→   552→                    else:
   553→   553→                        self.current_item = m_number.group(1)
   554→   554→                    m = m_letter or m_number or m_ordinal
   555→   555→                    remaining = text[m.end():].strip()
   556→   556→                    if remaining:
   557→   557→                        self.current_item_lines.append(remaining)
   558→   558→                    continue
   559→   559→
   560→   560→                # Sub-items: accumulate under current item
   561→   561→                if self.current_item:
   562→   562→                    self.current_item_lines.append(text)
   563→   563→                continue
   564→   564→
   565→   565→        self._flush_item()
   566→   566→        return self.chunks
   567→   567→
   568→   568→    def _flush_item(self):
   569→   569→        if not self.current_item or not self.current_item_lines:
   570→   570→            self.current_item_lines = []
   571→   571→            return
   572→   572→
   573→   573→        text = ' '.join(self.current_item_lines).strip()
   574→   574→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   575→   575→            self.current_item_lines = []
   576→   576→            return
   577→   577→
   578→   578→        kw = self.decision_keyword or "MEMUTUSKAN"
   579→   579→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   580→   580→                            chunk_type="diktum", chunk_num=self.current_item)
   581→   581→
   582→   582→        self.chunks.append({
   583→   583→            "id": chunk_id,
   584→   584→            "text": text,
   585→   585→            "metadata": {
   586→   586→                "doc_type": self.doc_type,
   587→   587→                "issuer": self.issuer,
   588→   588→                "year": self.year,
   589→   589→                "nomor": self.nomor,
   590→   590→                "section": "keputusan",
   591→   591→                "decision_keyword": kw,
   592→   592→                "diktum": self.current_item,
   593→   593→                "parent": None,
   594→   594→                "path": f"{kw} > Diktum {self.current_item}",
   595→   595→                "status": "active",
   596→   596→            }
   597→   597→        })
   598→   598→        self.current_item_lines = []

   601→   601→        self.preamble_lines = []

   719→   719→    for line in lines[:30]:

   727→   727→    """
