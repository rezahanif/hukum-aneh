    42→def build_id(doc_type, nomor, year, **parts):
    43→    """Build chunk ID. Handles both hierarchical and flat structures."""
    44→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    45→        nomor_perkara = parts.get("nomor_perkara", nomor or "?")
    46→        chunk_type = parts.get("chunk_type", "amar")
    47→        chunk_num = parts.get("chunk_num", "?")
    48→        return f"{doc_type}:{nomor_perkara}:{chunk_type}:{chunk_num}"
    49→    elif parts.get("chunk_type") in ("diktum", "pertimbangan"):
    50→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{parts['chunk_type']}:{parts['chunk_num']}"
    51→    else:
    52→        pasal = parts.get("pasal", "?")
    53→        ayat = parts.get("ayat", "?")
    54→        if ayat and ayat != "?":
    55→            return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}:{ayat}"
    56→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"

    59→def build_parent_id(doc_type, nomor, year, pasal):
    60→    """Build parent ID for ayat chunks."""
    61→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    62→        return None
    63→    return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"

    66→def build_path(headings):
    67→    """Build path string from heading stack. Filters out None and duplicates."""
    68→    seen = set()
    69→    parts = []
    70→    for h in headings:
    71→        if h and h not in seen:
    72→            seen.add(h)
    73→            parts.append(h)
    74→    return " > ".join(parts) if parts else ""

    77→def _make_bab_headings(bab_num, bab_label, bagian_num, bagian_label,
    78→                       paragraf_num, paragraf_label, pasal_str=None, ayat_str=None):
    79→    """Build heading list for path, avoiding duplicate BAB labels.
    80→    
    81→    Fix for bug #2: when BAB has no descriptive label (e.g., just "BAB I"),
    82→    the label defaults to "BAB I" which duplicates the number element.
    83→    Now: if label equals the number element, omit it.
    84→    """
    85→    headings = []
    86→    
    87→    bab_num_str = f"BAB {bab_num}" if bab_num else None
    88→    if bab_num_str:
    89→        headings.append(bab_num_str)
    90→    # Only add label if it's different from the number element and non-empty
    91→    if bab_label and bab_label != bab_num_str:
    92→        headings.append(bab_label)
    93→    
    94→    bagian_num_str = f"Bagian {bagian_num}" if bagian_num else None
    95→    if bagian_num_str:
    96→        headings.append(bagian_num_str)
    97→    if bagian_label and bagian_label != bagian_num_str:
    98→        headings.append(bagian_label)
    99→    
   100→    paragraf_num_str = f"Paragraf {paragraf_num}" if paragraf_num else None
   101→    if paragraf_num_str:
   102→        headings.append(paragraf_num_str)
   103→    if paragraf_label and paragraf_label != paragraf_num_str:
   104→        headings.append(paragraf_label)
   105→    
   106→    if pasal_str:
   107→        headings.append(pasal_str)
   108→    if ayat_str:
   109→        headings.append(ayat_str)
   110→    
   111→    return headings

   114→# ==========================================================================
   115→# FAMILY A: HIERARCHICAL STATUTE PARSER
   116→# ==========================================================================
   117→
   118→class FamilyAParser:
   119→    """
   120→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
   121→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
   122→    """
   123→
   124→    # Heading patterns (order matters: higher levels first)
   125→    # v6: allow '* ' prefix (UUD 1945 amendment markers)
   126→    RE_BAB = re.compile(r'^\*?\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
   127→    RE_BAGIAN = re.compile(r'^\*?\s*BAGIAN\s+(\w+)', re.IGNORECASE)
   128→    RE_PARAGRAF = re.compile(r'^\*?\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
   129→    # v6: capture alpha suffixes AND allow '* ' prefix (UUD 1945 amendment markers)
   130→    RE_PASAL = re.compile(r'^\*?\s*pasal\s+([IVXLCDM]+|\d+(?:\s+\d+)*[a-zA-Z]?)\s*[.}\s]*$', re.IGNORECASE)
   131→    # v6: Match pasal embedded mid-line (e.g., 'BAB VIIA ... * Pasal 22C')
   132→    RE_PASAL_EMBEDDED = re.compile(r'\*\s*Pasal\s+(\d+[a-zA-Z]*)', re.IGNORECASE)
   133→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
   134→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
   135→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
   136→
   137→    # Quoted amendment detection
   138→    RE_AMENDMENT_TRIGGER = re.compile(
   139→        r'(?:diubah|diganti|dihapus)\s+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
   140→        re.IGNORECASE
   141→    )
   142→    RE_QUOTE_START = re.compile(r'^\s*["“]')
   143→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
   144→
   145→    # Penutup (closing section) marker
   146→    RE_PENUTUP = re.compile(r'^(?:KETENTUAN\s+PENUTUP|PENUTUP|penutup)\b')
   147→    # v3: Penjelasan (elucidation) marker — can appear after penutup or directly after last pasal
   148→    RE_PENJELASAN = re.compile(r'^(?:PENJELASAN|penjelasan)\b')
   149→
   150→    def __init__(self, lines, doc_type, nomor, year, issuer):
   151→        self.lines = lines
   152→        self.doc_type = doc_type
   153→        self.nomor = nomor
   154→        self.year = year
   155→        self.issuer = issuer
   156→        self.chunks = []
   157→
   158→        # Heading stack
   159→        self.current_bab = None
   160→        self.current_bab_label = None
   161→        self.current_bagian = None
   162→        self.current_bagian_label = None
   163→        self.current_paragraf = None
   164→        self.current_paragraf_label = None
   165→        self.current_pasal = None
   166→        self.current_ayat = None
   167→
   168→        # State
   169→        self.in_preamble = True
   170→        self.in_penutup = False
   171→        self.in_penjelasan = False  # v3: track elucidation section
   172→        self.in_quoted_amendment = False
   173→        self.quoted_amendment_source = None  # tracks which pasal is being amended
   174→        self.current_ayat_lines = []
   175→        self.current_pasal_lines = []
   176→        self.quoted_amendment_lines = []  # accumulate quoted text for chunking
   177→        self._penjelasan_lines = []  # v3: accumulate Penjelasan text
   178→        self._penjelasan_pasal = None
   179→
   180→    def parse(self) -> List[Dict]:
   181→        """Main parse loop. Returns list of chunks."""
   182→        for i, line in enumerate(self.lines):
   183→            text = line["text"].strip()
   184→            if not text:
   185→                continue
   186→
   187→            text_upper = text.upper()
   188→
   189→            # Detect preamble end / body start
   190→            if self.in_preamble:
   191→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   192→                    self.in_preamble = False
   193→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
   194→                    self.in_preamble = False
   195→                    continue
   196→                else:
   197→                    continue
   198→
   199→            # Detect penutup
   200→            if self.RE_PENUTUP.match(text):
   201→                self._flush_ayat()
   202→                self._flush_pasal()
   203→                self._flush_quoted_amendment()
   204→                self.in_penutup = True
   205→                continue
   206→
   207→            # v3: Detect PENJELASAN even without penutup (many docs go straight to penjelasan)
   208→            if self.RE_PENJELASAN.match(text) and not self.in_penjelasan:
   209→                self._flush_ayat()
   210→                self._flush_pasal()
   211→                self._flush_quoted_amendment()
   212→                self.in_penutup = True  # treat as post-operative
   213→                self.in_penjelasan = True
   214→                self._penjelasan_pasal = None
   215→                continue
   216→
   217→            if self.in_penutup:
   218→                # v3: Detect signature block and Penjelasan as hard boundaries
   219→                text_lower = text.lower()
   220→                # Signature block: lines like "Ditetapkan di Jakarta", "pada tanggal", or ttd/
   221→                # Also detect by very short lines after penutup start
   222→                if (re.search(r'ttd|disahkan|ditetapkan', text_lower) or
   223→                    (len(text) < 15 and text_upper == text and len(text) > 2)):
   224→                    self._flush_penjelasan()  # flush any accumulated penjelasan
   225→                    continue
   226→                # PENJELASAN section header
   227→                if text_upper == "PENJELASAN" or text_upper.startswith("PENJELASAN "):
   228→                    self._flush_penjelasan()
   229→                    self.in_penjelasan = True
   230→                    self._penjelasan_pasal = None
   231→                    continue
   232→                # Inside Penjelasan: track pasal references for IDs
   233→                if self.in_penjelasan:
   234→                    # v5: also capture alpha suffixes in Penjelasan pasal references
   235→                    m_pj = re.match(r'^\s*Pasal\s+(\d+[a-zA-Z]*)', text, re.IGNORECASE)
   236→                    if m_pj:
   237→                        self._flush_penjelasan()
   238→                        self._penjelasan_pasal = m_pj.group(1)
   239→                    else:
   240→                        self._penjelasan_lines.append(text)
   241→                continue
   242→
   243→            # Check for quoted amendment TRIGGER ("Pasal X diubah sehingga berbunyi:")
   244→            if not self.in_quoted_amendment:
   245→                if self.RE_AMENDMENT_TRIGGER.search(text):
   246→                    # Extract which pasal is being amended
   247→                    m_pasal = re.search(r'Pasal\s+(\d+|\w+)\s+(?:ayat\s*\(\d+\)\s+)?(?:diubah|diganti|dihapus)',
   248→                                         text, re.IGNORECASE)
   249→                    source_pasal = m_pasal.group(1) if m_pasal else "?"
   250→                    self.in_quoted_amendment = True
   251→                    self.quoted_amendment_source = source_pasal
   252→                    self._flush_ayat()
   253→                    # Store the trigger sentence as context for the quoted chunk
   254→                    self.quoted_amendment_lines = [text]
   255→                    continue
   256→
   257→            # Inside quoted amendment: accumulate text
   258→            if self.in_quoted_amendment:
   259→                self.quoted_amendment_lines.append(text)
   260→                if self.RE_QUOTE_END.search(text):
   261→                    self._flush_quoted_amendment()
   262→                # Also check for end of quoted section without closing quote:
   263→                # if we hit a new structural marker, the quote ended implicitly
   264→                if (self.RE_BAB.match(text) or self.RE_PASAL.match(text) or
   265→                    self.RE_PENUTUP.match(text)):
   266→                    self._flush_quoted_amendment()
   267→                    # Don't skip the structural marker - let it be processed below
   268→                    self.in_quoted_amendment = False
   269→                elif self.RE_QUOTE_END.search(text):
   270→                    pass  # already flushed above
   271→                else:
   272→                    continue
   273→
   274→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   275→            m = self.RE_BAB.match(text)
   276→            if m:
   277→                self._flush_ayat()
   278→                self._flush_pasal()
   279→                self.current_bab = m.group(1)
   280→                label_match = re.match(r'^\*?\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   281→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   282→                # Fix #2: if label is empty or just repeats "BAB X", set to None
   283→                if raw_label and raw_label.upper() != f"BAB {m.group(1).upper()}":
   284→                    self.current_bab_label = raw_label
   285→                else:
   286→                    self.current_bab_label = None
   287→                self.current_bagian = None
   288→                self.current_bagian_label = None
   289→                self.current_paragraf = None
   290→                self.current_paragraf_label = None
   291→                # v4: reset pasal when BAB changes to prevent heading-as-chunk
   292→                self.current_pasal = None
   293→                # v6: Check if a Pasal reference is glued to the BAB line
   294→                # (e.g., 'BAB VIIA DEWAN PERWAKILAN DAERAH * Pasal 22C')
   295→                m_glued = self.RE_PASAL_EMBEDDED.search(text)
   296→                if m_glued:
   297→                    self.current_pasal = m_glued.group(1)
   298→                    self.current_ayat = None
   299→                continue
   300→
   301→            m = self.RE_BAGIAN.match(text)
   302→            if m:
   303→                self._flush_ayat()
   304→                self._flush_pasal()
   305→                self.current_bagian = m.group(1)
   306→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   307→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   308→                if raw_label:
   309→                    self.current_bagian_label = raw_label
   310→                else:
   311→                    self.current_bagian_label = None
   312→                self.current_paragraf = None
   313→                self.current_paragraf_label = None
   314→                continue
   315→
   316→            m = self.RE_PARAGRAF.match(text)
   317→            if m:
   318→                self._flush_ayat()
   319→                self._flush_pasal()
   320→                self.current_paragraf = m.group(1)
   321→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   322→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   323→                if raw_label:
   324→                    self.current_paragraf_label = raw_label
   325→                else:
   326→                    self.current_paragraf_label = None
   327→                continue
   328→
   329→            m = self.RE_PASAL.match(text)
   330→            if m:
   331→                self._flush_ayat()
   332→                self._flush_pasal()
   333→                raw_num = m.group(1)
   334→                # v4: normalize space-separated digits ("1 1" -> "11", "1 10" -> "110")
   335→                self.current_pasal = re.sub(r'\s+', '', raw_num) if re.match(r'^\d', raw_num) else raw_num
   336→                self.current_ayat = None
   337→                # v4: capture any remaining text on the same line as pasal heading
   338→                remaining = text[m.end():].strip().lstrip('.').strip()
   339→                if remaining and len(remaining) > 5:
   340→                    self.current_pasal_lines.append(remaining)
   341→                continue
   342→
   343→            m = self.RE_AYAT.match(text)
   344→            if m and self.current_pasal:
   345→                self._flush_ayat()
   346→                self.current_ayat = m.group(1)
   347→                ayat_text = self.RE_AYAT.sub('', text).strip()
   348→                if ayat_text:
   349→                    self.current_ayat_lines.append(ayat_text)
   350→                continue
   351→
   352→            # v6: Check for embedded pasal in non-structural line
   353→            # (e.g., 'PEMILIHAN UMUM * Pasal 22E' — BAB label with glued pasal)
   354→            m_emb = self.RE_PASAL_EMBEDDED.search(text)
   355→            if m_emb and self.current_pasal:
   356→                self._flush_ayat()
   357→                self._flush_pasal()
   358→                self.current_pasal = m_emb.group(1)
   359→                self.current_ayat = None
   360→                continue
   361→
   362→            # If we're inside a pasal but no ayat detected, accumulate lines
   363→            if self.current_pasal:
   364→                if self.current_ayat:
   365→                    self.current_ayat_lines.append(text)
   366→                else:
   367→                    self.current_pasal_lines.append(text)
   368→
   369→        # Flush remaining
   370→        self._flush_ayat()
   371→        self._flush_pasal()
   372→        self._flush_quoted_amendment()
   373→
   374→        # v6: Deduplicate truncated page-break fragments
   375→        self._dedup_truncated_fragments()
   376→
   377→        return self.chunks
   378→
   379→    def _flush_ayat(self):
   380→        """Flush accumulated ayat lines as a chunk."""
   381→        if not self.current_ayat or not self.current_ayat_lines:
   382→            self.current_ayat_lines = []
   383→            return
   384→
   385→        text = ' '.join(self.current_ayat_lines).strip()
   386→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   387→            self.current_ayat_lines = []
   388→            return
   389→
   390→        headings = _make_bab_headings(
   391→            self.current_bab, self.current_bab_label,
   392→            self.current_bagian, self.current_bagian_label,
   393→            self.current_paragraf, self.current_paragraf_label,
   394→            pasal_str=f"Pasal {self.current_pasal}",
   395→            ayat_str=f"Ayat ({self.current_ayat})"
   396→        )
   397→
   398→        path = build_path(headings)
   399→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   400→                            pasal=self.current_pasal, ayat=self.current_ayat)
   401→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   402→
   403→        self.chunks.append({
   404→            "id": chunk_id,
   405→            "text": text,
   406→            "metadata": {
   407→                "doc_type": self.doc_type,
   408→                "issuer": self.issuer,
   409→                "year": self.year,
   410→                "bab": self.current_bab or None,
   411→                "pasal": self.current_pasal,
   412→                "ayat": self.current_ayat,
   413→                "parent": parent_id,
   414→                "path": path,
   415→                "status": "active",
   416→            }
   417→        })
   418→        self.current_ayat_lines = []
   419→
   420→    def _flush_pasal(self):
   421→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   422→        if not self.current_pasal or not self.current_pasal_lines:
   423→            self.current_pasal_lines = []
   424→            return
   425→
   426→        text = ' '.join(self.current_pasal_lines).strip()
   427→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   428→            self.current_pasal_lines = []
   429→            return
   430→
   431→        headings = _make_bab_headings(
   432→            self.current_bab, self.current_bab_label,
   433→            self.current_bagian, self.current_bagian_label,
   434→            self.current_paragraf, self.current_paragraf_label,
   435→            pasal_str=f"Pasal {self.current_pasal}",
   436→        )
   437→        path = build_path(headings)
   438→        chunk_id = build_id(self.doc_type, self.nomor, self.year, pasal=self.current_pasal)
   439→
   440→        self.chunks.append({
   441→            "id": chunk_id,
   442→            "text": text,
   443→            "metadata": {
   444→                "doc_type": self.doc_type,
   445→                "issuer": self.issuer,
   446→                "year": self.year,
   447→                "bab": self.current_bab or None,
   448→                "pasal": self.current_pasal,
   449→                "ayat": None,
   450→                "parent": None,
   451→                "path": path,
   452→                "status": "active",
   453→            }
   454→        })
   455→        self.current_pasal_lines = []
   456→
   457→    def _flush_quoted_amendment(self):
   458→        """Flush quoted amendment text as a chunk with status='quoted_amendment'.
   459→        
   460→        Fix for bug #6: instead of silently skipping quoted amendments,
   461→        we now chunk them with a clear status tag so retrieval can
   462→        distinguish live law from historical amendment pointers.
   463→        """
   464→        if not self.quoted_amendment_lines:
   465→            self.in_quoted_amendment = False
   466→            self.quoted_amendment_source = None
   467→            return
   468→
   469→        text = ' '.join(self.quoted_amendment_lines).strip()
   470→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   471→            self.quoted_amendment_lines = []
   472→            self.in_quoted_amendment = False
   473→            self.quoted_amendment_source = None
   474→            return
   475→
   476→        source = self.quoted_amendment_source or "?"
   477→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   478→                            pasal=source, ayat="quoted")
   479→
   480→        self.chunks.append({
   481→            "id": chunk_id,
   482→            "text": text,
   483→            "metadata": {
   484→                "doc_type": self.doc_type,

   506→        if not self._penjelasan_lines:

   515→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   516→                            pasal=pasal_ref, ayat="penjelasan")
   517→
   518→        self.chunks.append({
   519→            "id": chunk_id,
   520→            "text": text,
   521→            "metadata": {
   522→                "doc_type": self.doc_type,
   523→                "issuer": self.issuer,
   524→                "year": self.year,
   525→                "bab": self.current_bab or None,
   526→                "pasal": pasal_ref,
   527→                "ayat": "penjelasan",
   528→                "parent": None,
   529→                "path": f"Penjelasan > Pasal {pasal_ref}",

   531→            }

   536→    def _dedup_truncated_fragments(self):
   537→        """v6: Remove truncated page-break fragments that duplicate a longer chunk.
   538→        
   539→        When text spans a page break, the bottom-of-page fragment (10-50 chars)
   540→        gets the same pasal:ayat ID as the continuation on the next page.
   541→        Keep the longer chunk, drop the fragment.
   542→        """
   543→        from collections import defaultdict
   544→        id_chunks = defaultdict(list)
   545→        for i, chunk in enumerate(self.chunks):
   546→            id_chunks[chunk["id"]].append(i)
   547→        
   548→        to_remove = set()
   549→        for chunk_id, indices in id_chunks.items():
   550→            if len(indices) < 2:
   551→                continue
   552→            pairs = [(i, len(self.chunks[i]["text"])) for i in indices]
   553→            pairs.sort(key=lambda x: -x[1])
   554→            longest_idx, longest_len = pairs[0]
   555→            
   556→            for idx, text_len in pairs[1:]:
   557→                if text_len < longest_len * 0.5:
   558→                    shorter = self.chunks[idx]["text"].strip()
   559→                    longer = self.chunks[longest_idx]["text"].strip()
   560→                    if longer.endswith(shorter) or shorter in longer:
   561→                        to_remove.add(idx)
   562→        
   563→        if to_remove:
   564→            self.chunks = [c for i, c in enumerate(self.chunks) if i not in to_remove]

   571→class FamilyBParser:
   572→    """
   573→    Parses decree/decision documents: Menimbang > Mengingat > MEMUTUSKAN/MENETAPKAN > Diktum.
   574→    Handles: Keppres, Inpres, Tap_MPR, Kepmen.
   575→    
   576→    v2 fix: Recognizes "Pasal X" as diktum divider (keppres pattern where
   577→    diktum items are organized under Pasal headings instead of numbered).
   578→    """
   579→
   580→    RE_MENIMBANG = re.compile(r'^\s*menimbang\s*:?\s*$', re.IGNORECASE)
   581→    RE_MENGINGAT = re.compile(r'^\s*mengingat\s*:?\s*$', re.IGNORECASE)
   582→    # Decision keywords: use $ anchor to match at END of line, allowing any prefix
   583→    RE_MEMUTUSKAN = re.compile(r'memutuskan\s*:?\s*$', re.IGNORECASE)
   584→    RE_MENETAPKAN = re.compile(r'menetapkan\s*:?\s*$', re.IGNORECASE)
   585→    RE_MENINSTRUKSIKAN = re.compile(r'menginstruksikan\s*:?\s*$', re.IGNORECASE)
   586→    RE_ITEM_LETTER = re.compile(r'^\s*([a-z])\s*\.\s*')
   587→    RE_ITEM_NUMBER = re.compile(r'^\s*(\d+)\s*\.\s*')
   588→    RE_ORDINAL = re.compile(
   589→        r'^\s*(KE(?:SATU|DUA|TIGA|EMPAT|LIMA|NAM|TUJUH|DELAPAN|SEMBILAN|SEPULUH|'
   590→        r'SEBELAS|DUA BELAS|TIGA BELAS|EMPAT BELAS|LIMA BELAS|ENAM BELAS|'
   591→        r'TUJUH BELAS|DELAPAN BELAS|SEMBILAN BELAS|DUA PULUH))\b',
   592→        re.IGNORECASE
   593→    )
   594→    # Fix #5: recognize "Pasal X" as diktum divider in Family B
   595→    # v5: also handle alpha suffixed pasal numbers (e.g. Pasal 68A)
   596→    RE_PASAL_DIKTUM = re.compile(r'^\s*Pasal\s+(\d+[a-zA-Z]?)\s*\.?\s*$', re.IGNORECASE)
   597→
   598→    def __init__(self, lines, doc_type, nomor, year, issuer):
   599→        self.lines = lines
   600→        self.doc_type = doc_type
   601→        self.nomor = nomor
   602→        self.year = year
   603→        self.issuer = issuer
   604→        self.chunks = []
   605→
   606→        self.section = "pre"  # pre, menimbang, mengingat, keputusan, done
   607→        self.current_item = None
   608→        self.current_item_lines = []
   609→        self.preamble_lines = []
   610→        self.decision_keyword = None
   611→        self.parent_section = None  # v3: track parent lettered section for inpres diktum IDs
   612→
   613→    def parse(self) -> List[Dict]:
   614→        for line in self.lines:
   615→            text = line["text"].strip()
   616→            if not text:
   617→                continue
   618→
   619→            text_upper = text.upper()
   620→
   621→            # Section transitions
   622→            if self.RE_MENIMBANG.match(text):
   623→                self._flush_preamble()
   624→                self.section = "menimbang"
   625→                self.current_item = None
   626→                continue
   627→
   628→            if self.RE_MENGINGAT.match(text):
   629→                self._flush_preamble()
   630→                self.section = "mengingat"
   631→                self.current_item = None
   632→                continue
   633→
   634→            # Decision keyword triggers (use search() for end-anchored patterns)
   635→            if self.section not in ("keputusan", "done"):
   636→                if self.RE_MENINSTRUKSIKAN.search(text):
   637→                    self._flush_item()
   638→                    self.section = "keputusan"
   639→                    self.decision_keyword = "MENGINSTRUKSIKAN"
   640→                    self.current_item = None
   641→                    continue
   642→
   643→                if self.RE_MEMUTUSKAN.search(text):
   644→                    self._flush_item()
   645→                    self.section = "keputusan"
   646→                    self.decision_keyword = "MEMUTUSKAN"
   647→                    self.current_item = None
   648→                    continue
   649→
   650→                if self.RE_MENETAPKAN.search(text):
   651→                    self._flush_item()
   652→                    self.section = "keputusan"
   653→                    self.decision_keyword = "MENETAPKAN"
   654→                    self.current_item = None
   655→                    continue
   656→
   657→            if self.section == "pre":
   658→                continue
   659→
   660→            if self.section == "done":
   661→                continue
   662→
   663→            # v3: Signature block detection (broader)
   664→            text_lower = text.lower()
   665→            if ("ttd" in text_lower and len(text) < 30):
   666→                self._flush_item()
   667→                self.section = "done"
   668→                continue
   669→            # Also stop at formal signing language
   670→            if re.search(r'ditetapkan\s+di|dilantik\s+di', text_lower):
   671→                self._flush_item()
   672→                self.section = "done"
   673→                continue
   674→            # Stop at Penjelasan in Family B documents
   675→            if text_upper == "PENJELASAN":
   676→                self._flush_item()
   677→                self.section = "done"
   678→                continue
   679→
   680→            # In keputusan section: check for item markers
   681→            if self.section == "keputusan":
   682→                m_letter = self.RE_ITEM_LETTER.match(text)
   683→                m_number = self.RE_ITEM_NUMBER.match(text)
   684→                m_ordinal = self.RE_ORDINAL.match(text)
   685→                m_pasal = self.RE_PASAL_DIKTUM.match(text)
   686→
   687→                if m_pasal:
   688→                    # Fix #5: "Pasal X" acts as a diktum divider in some keppres
   689→                    self._flush_item()
   690→                    self.current_item = f"Pasal {m_pasal.group(1)}"
   691→                    continue
   692→
   693→                if m_letter or m_number or m_ordinal:
   694→                    self._flush_item()
   695→                    if m_ordinal:
   696→                        self.current_item = m_ordinal.group(1).upper()
   697→                    elif m_letter:
   698→                        self.current_item = m_letter.group(1)
   699→                    else:
   700→                        self.current_item = m_number.group(1)
   701→                    m = m_letter or m_number or m_ordinal
   702→                    remaining = text[m.end():].strip()
   703→                    if remaining:
   704→                        self.current_item_lines.append(remaining)
   705→                    continue
   706→
   707→                # Sub-items: accumulate under current item
   708→                if self.current_item:
   709→                    self.current_item_lines.append(text)
   710→                continue
   711→
   712→        self._flush_item()
   713→        return self.chunks
   714→
   715→    def _flush_item(self):
   716→        if not self.current_item or not self.current_item_lines:
   717→            self.current_item_lines = []
   718→            return
   719→
   720→        text = ' '.join(self.current_item_lines).strip()
   721→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   722→            self.current_item_lines = []
   723→            return
   724→
   725→        kw = self.decision_keyword or "MEMUTUSKAN"
   726→        # v3: Include parent section in diktum ID for inpres (prevents numbering collision)
   727→        diktum_num = self.current_item
   728→        if self.parent_section and diktum != self.parent_section:
   729→            diktum_num = f"{self.parent_section}.{diktum_num}"
   730→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   731→                            chunk_type="diktum", chunk_num=diktum_num)
   732→
   733→        self.chunks.append({
   734→            "id": chunk_id,
   735→            "text": text,
   736→            "metadata": {
   737→                "doc_type": self.doc_type,
   738→                "issuer": self.issuer,
   739→                "year": self.year,
   740→                "nomor": self.nomor,
   741→                "section": "keputusan",
   742→                "decision_keyword": kw,
   743→                "diktum": self.current_item,
   744→                "parent": None,
   745→                "path": f"{kw} > Diktum {self.current_item}",
   746→                "status": "active",
   747→            }
   748→        })
   749→        self.current_item_lines = []
   750→
   751→    def _flush_preamble(self):
   752→        self.preamble_lines = []

   784→    def parse(self) -> List[Dict]:

   819→
   820→                if self.current_amar:

   868→def extract_nomor_perkara(lines):

   877→def chunk_pdf(pdf_path: str, directory: str, family_map: Dict) -> List[Dict]:
   878→    """
   879→    Main entry point. Extract + parse a PDF into chunks.
   880→    Returns list of chunk dicts with id, text, metadata.
   881→    """
   882→    dir_config = family_map.get(directory, {})
   883→    family = dir_config.get("family", "A")
   884→    default_doc_type = dir_config.get("doc_type", "Unknown")
   885→    default_issuer = dir_config.get("issuer", "Unknown")
   886→
   887→    try:
   888→        lines, meta = extract_clean(pdf_path)
   889→    except Exception as e:
   890→        return [{"id": f"ERROR:{directory}", "text": str(e), "metadata": {"error": True}}]
   891→
   892→    if not lines:
   893→        return [{"id": f"EMPTY:{directory}", "text": "No extractable text (possibly scanned PDF)", "metadata": {"error": True, "needs_ocr": True}}]
   894→
   895→    # v4 fix: Detect doc_type from title block FIRST, directory as fallback.
   896→    # Use page-1-only RAW lines (pre-boilerplate) since boilerplate stripping
   897→    # removes the actual document title (e.g. "PERATURAN PRESIDEN").
   898→    # Only page 1 to avoid body citations polluting the detection.
   899→    page1_raw = [l["text"].strip() for l in lines if l.get("page") == 1]
   900→    raw_title_block = " ".join(page1_raw[:15]) if page1_raw else ""
   901→    detected_type, detected_issuer = detect_doc_type_from_title(raw_title_block)
   902→    
   903→    # If page 1 was empty (image) or detection failed, try cleaned lines
   904→    if detected_type == "Unknown":
   905→        clean_title = extract_title_block(lines, max_first_n=5)
   906→        detected_type, detected_issuer = detect_doc_type_from_title(clean_title)
   907→    
   908→    # Title block wins when it returns a real type; directory is fallback only
   909→    if detected_type != "Unknown":
   910→        doc_type = detected_type
   911→        issuer = detected_issuer
   912→        if default_doc_type != "Unknown" and detected_type != default_doc_type:
   913→            print(f"  [WARN] doc_type: title={detected_type} overrides dir={default_doc_type} for {directory}")
   914→    elif default_doc_type != "Unknown":
   915→        doc_type = default_doc_type
   916→        issuer = default_issuer
   917→    else:
   918→        doc_type = "Unknown"
   919→        issuer = "Unknown"
   920→    nomor = meta.get("nomor")
   921→    year = meta.get("year")
   922→
   923→    # v4: If page 1 was empty (image), always re-extract nomor/year from post-decision text
   924→    # since the pre-decision text contains citations, not the document's own number.
   925→    page1_has_text = any(l.get("page") == 1 for l in lines)
   926→    if not page1_has_text and meta.get("total_pages", 0) > 1:
   927→        post_nomor, post_year = extract_nomor_year_post_decision(lines)
   928→        if post_nomor and post_nomor != nomor:
   929→            nomor = post_nomor
   930→            if post_year:
   931→                year = post_year
   932→            print(f"  [INFO] Post-decision: nomor={nomor}, year={year}")
   933→        # v4: Filename fallback when page 1 is image and text extraction found wrong number
   934→        if (not post_nomor) or (nomor == meta.get('nomor') and re.search(r'(\d+)[_.](\d{4})', os.path.basename(pdf_path))):
   935→            fn = os.path.basename(pdf_path)
   936→            fn_match = re.search(r'(\d+)[_.](\d{4})', fn)
   937→            if fn_match:
   938→                nomor = fn_match.group(1)
   939→                year = fn_match.group(2)
   940→                print(f"  [INFO] Filename fallback: nomor={nomor}, year={year}")
   941→    
   942→    # Directory-specific overrides for image-cover documents where title can't be read
   943→    if directory == "JDIH_KPU" and doc_type in ("Peraturan", "Unknown"):
   944→        doc_type = "PKPU"
   945→        issuer = "Komisi Pemilihan Umum"
   946→
   947→    # Parse based on family
   948→    if family == "A":
   949→        parser = FamilyAParser(lines, doc_type, nomor, year, issuer)
   950→        chunks = parser.parse()
   951→    elif family == "B":
   952→        parser = FamilyBParser(lines, doc_type, nomor, year, issuer)
   953→        chunks = parser.parse()
   954→    elif family == "C":
   955→        nomor_perkara = extract_nomor_perkara(lines)
   956→        parser = FamilyCParser(lines, doc_type, nomor, year, issuer, nomor_perkara)
   957→        chunks = parser.parse()
   958→    else:
   959→        chunks = [{"id": f"UNKNOWN_FAMILY:{directory}", "text": "", "metadata": {"error": True}}]
   960→
   961→    # Add extraction metadata to all chunks
   962→    for chunk in chunks:
   963→        chunk["metadata"]["source_file"] = meta["file"]
   964→        chunk["metadata"]["directory"] = directory
   965→        chunk["metadata"]["family"] = family
   966→        if nomor:
   967→            chunk["metadata"]["nomor"] = nomor
   968→        if year:
   969→            chunk["metadata"]["year"] = year
   970→
   971→    return chunks
