   118→   118→   118→class FamilyAParser:
   119→   119→   119→    """
   120→   120→   120→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
   121→   121→   121→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
   122→   122→   122→    """
   123→   123→   123→
   124→   124→   124→    # Heading patterns (order matters: higher levels first)
   125→   125→   125→    # v6: allow '* ' prefix (UUD 1945 amendment markers)
   126→   126→   126→    RE_BAB = re.compile(r'^\*?\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
   127→   127→   127→    RE_BAGIAN = re.compile(r'^\*?\s*BAGIAN\s+(\w+)', re.IGNORECASE)
   128→   128→   128→    RE_PARAGRAF = re.compile(r'^\*?\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
   129→   129→   129→    # v6: capture alpha suffixes AND allow '* ' prefix (UUD 1945 amendment markers)
   130→   130→   130→    RE_PASAL = re.compile(r'^\*?\s*pasal\s+([IVXLCDM]+|\d+(?:\s+\d+)*[a-zA-Z]?)\s*[.}\s]*$', re.IGNORECASE)
   131→   131→   131→    # v6: Match pasal embedded mid-line (e.g., 'BAB VIIA ... * Pasal 22C')
   132→   132→   132→    RE_PASAL_EMBEDDED = re.compile(r'\*\s*Pasal\s+(\d+[a-zA-Z]*)', re.IGNORECASE)
   133→   133→   133→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
   134→   134→   134→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
   135→   135→   135→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
   136→   136→   136→
   137→   137→   137→    # Quoted amendment detection
   138→   138→   138→    RE_AMENDMENT_TRIGGER = re.compile(
   139→   139→   139→        r'(?:diubah|diganti|dihapus)\s+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
   140→   140→   140→        re.IGNORECASE
   141→   141→   141→    )
   142→   142→   142→    RE_QUOTE_START = re.compile(r'^\s*["“]')
   143→   143→   143→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
   144→   144→   144→
   145→   145→   145→    # Penutup (closing section) marker
   146→   146→   146→    RE_PENUTUP = re.compile(r'^(?:KETENTUAN\s+PENUTUP|PENUTUP|penutup)\b')
   147→   147→   147→    # v3: Penjelasan (elucidation) marker — can appear after penutup or directly after last pasal
   148→   148→   148→    RE_PENJELASAN = re.compile(r'^(?:PENJELASAN|penjelasan)\b')
   149→   149→   149→
   150→   150→   150→    def __init__(self, lines, doc_type, nomor, year, issuer):
   151→   151→   151→        self.lines = lines
   152→   152→   152→        self.doc_type = doc_type
   153→   153→   153→        self.nomor = nomor
   154→   154→   154→        self.year = year
   155→   155→   155→        self.issuer = issuer
   156→   156→   156→        self.chunks = []
   157→   157→   157→
   158→   158→   158→        # Heading stack
   159→   159→   159→        self.current_bab = None
   160→   160→   160→        self.current_bab_label = None
   161→   161→   161→        self.current_bagian = None
   162→   162→   162→        self.current_bagian_label = None
   163→   163→   163→        self.current_paragraf = None
   164→   164→   164→        self.current_paragraf_label = None
   165→   165→   165→        self.current_pasal = None
   166→   166→   166→        self.current_ayat = None
   167→   167→   167→
   168→   168→   168→        # State
   169→   169→   169→        self.in_preamble = True
   170→   170→   170→        self.in_penutup = False
   171→   171→   171→        self.in_penjelasan = False  # v3: track elucidation section
   172→   172→   172→        self.in_quoted_amendment = False
   173→   173→   173→        self.quoted_amendment_source = None  # tracks which pasal is being amended
   174→   174→   174→        self.current_ayat_lines = []
   175→   175→   175→        self.current_pasal_lines = []
   176→   176→   176→        self.quoted_amendment_lines = []  # accumulate quoted text for chunking
   177→   177→   177→        self._penjelasan_lines = []  # v3: accumulate Penjelasan text
   178→   178→   178→        self._penjelasan_pasal = None
   179→   179→   179→
   180→   180→   180→    def parse(self) -> List[Dict]:
   181→   181→   181→        """Main parse loop. Returns list of chunks."""
   182→   182→   182→        for i, line in enumerate(self.lines):
   183→   183→   183→            text = line["text"].strip()
   184→   184→   184→            if not text:
   185→   185→   185→                continue
   186→   186→   186→
   187→   187→   187→            text_upper = text.upper()
   188→   188→   188→
   189→   189→   189→            # Detect preamble end / body start
   190→   190→   190→            if self.in_preamble:
   191→   191→   191→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   192→   192→   192→                    self.in_preamble = False
   193→   193→   193→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
   194→   194→   194→                    self.in_preamble = False
   195→   195→   195→                    continue
   196→   196→   196→                else:
   197→   197→   197→                    continue
   198→   198→   198→
   199→   199→   199→            # Detect penutup
   200→   200→   200→            if self.RE_PENUTUP.match(text):
   201→   201→   201→                self._flush_ayat()
   202→   202→   202→                self._flush_pasal()
   203→   203→   203→                self._flush_quoted_amendment()
   204→   204→   204→                self.in_penutup = True
   205→   205→   205→                continue
   206→   206→   206→
   207→   207→   207→            # v3: Detect PENJELASAN even without penutup (many docs go straight to penjelasan)
   208→   208→   208→            if self.RE_PENJELASAN.match(text) and not self.in_penjelasan:
   209→   209→   209→                self._flush_ayat()
   210→   210→   210→                self._flush_pasal()
   211→   211→   211→                self._flush_quoted_amendment()
   212→   212→   212→                self.in_penutup = True  # treat as post-operative
   213→   213→   213→                self.in_penjelasan = True
   214→   214→   214→                self._penjelasan_pasal = None
   215→   215→   215→                continue
   216→   216→   216→
   217→   217→   217→            if self.in_penutup:
   218→   218→   218→                # v3: Detect signature block and Penjelasan as hard boundaries
   219→   219→   219→                text_lower = text.lower()
   220→   220→   220→                # Signature block: lines like "Ditetapkan di Jakarta", "pada tanggal", or ttd/
   221→   221→   221→                # Also detect by very short lines after penutup start
   222→   222→   222→                if (re.search(r'ttd|disahkan|ditetapkan', text_lower) or
   223→   223→   223→                    (len(text) < 15 and text_upper == text and len(text) > 2)):
   224→   224→   224→                    self._flush_penjelasan()  # flush any accumulated penjelasan
   225→   225→   225→                    continue
   226→   226→   226→                # PENJELASAN section header
   227→   227→   227→                if text_upper == "PENJELASAN" or text_upper.startswith("PENJELASAN "):
   228→   228→   228→                    self._flush_penjelasan()
   229→   229→   229→                    self.in_penjelasan = True
   230→   230→   230→                    self._penjelasan_pasal = None
   231→   231→   231→                    continue
   232→   232→   232→                # Inside Penjelasan: track pasal references for IDs
   233→   233→   233→                if self.in_penjelasan:
   234→   234→   234→                    # v5: also capture alpha suffixes in Penjelasan pasal references
   235→   235→   235→                    m_pj = re.match(r'^\s*Pasal\s+(\d+[a-zA-Z]*)', text, re.IGNORECASE)
   236→   236→   236→                    if m_pj:
   237→   237→   237→                        self._flush_penjelasan()
   238→   238→   238→                        self._penjelasan_pasal = m_pj.group(1)
   239→   239→   239→                    else:
   240→   240→   240→                        self._penjelasan_lines.append(text)
   241→   241→   241→                continue
   242→   242→   242→
   243→   243→   243→            # Check for quoted amendment TRIGGER ("Pasal X diubah sehingga berbunyi:")
   244→   244→   244→            if not self.in_quoted_amendment:
   245→   245→   245→                if self.RE_AMENDMENT_TRIGGER.search(text):
   246→   246→   246→                    # Extract which pasal is being amended
   247→   247→   247→                    m_pasal = re.search(r'Pasal\s+(\d+|\w+)\s+(?:ayat\s*\(\d+\)\s+)?(?:diubah|diganti|dihapus)',
   248→   248→   248→                                         text, re.IGNORECASE)
   249→   249→   249→                    source_pasal = m_pasal.group(1) if m_pasal else "?"
   250→   250→   250→                    self.in_quoted_amendment = True
   251→   251→   251→                    self.quoted_amendment_source = source_pasal
   252→   252→   252→                    self._flush_ayat()
   253→   253→   253→                    # Store the trigger sentence as context for the quoted chunk
   254→   254→   254→                    self.quoted_amendment_lines = [text]
   255→   255→   255→                    continue
   256→   256→   256→
   257→   257→   257→            # Inside quoted amendment: accumulate text
   258→   258→   258→            if self.in_quoted_amendment:
   259→   259→   259→                self.quoted_amendment_lines.append(text)
   260→   260→   260→                if self.RE_QUOTE_END.search(text):
   261→   261→   261→                    self._flush_quoted_amendment()
   262→   262→   262→                # Also check for end of quoted section without closing quote:
   263→   263→   263→                # if we hit a new structural marker, the quote ended implicitly
   264→   264→   264→                if (self.RE_BAB.match(text) or self.RE_PASAL.match(text) or
   265→   265→   265→                    self.RE_PENUTUP.match(text)):
   266→   266→   266→                    self._flush_quoted_amendment()
   267→   267→   267→                    # Don't skip the structural marker - let it be processed below
   268→   268→   268→                    self.in_quoted_amendment = False
   269→   269→   269→                elif self.RE_QUOTE_END.search(text):
   270→   270→   270→                    pass  # already flushed above
   271→   271→   271→                else:
   272→   272→   272→                    continue
   273→   273→   273→
   274→   274→   274→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   275→   275→   275→            m = self.RE_BAB.match(text)
   276→   276→   276→            if m:
   277→   277→   277→                self._flush_ayat()
   278→   278→   278→                self._flush_pasal()
   279→   279→   279→                self.current_bab = m.group(1)
   280→   280→   280→                label_match = re.match(r'^\*?\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   281→   281→   281→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   282→   282→   282→                # Fix #2: if label is empty or just repeats "BAB X", set to None
   283→   283→   283→                if raw_label and raw_label.upper() != f"BAB {m.group(1).upper()}":
   284→   284→   284→                    self.current_bab_label = raw_label
   285→   285→   285→                else:
   286→   286→   286→                    self.current_bab_label = None
   287→   287→   287→                self.current_bagian = None
   288→   288→   288→                self.current_bagian_label = None
   289→   289→   289→                self.current_paragraf = None
   290→   290→   290→                self.current_paragraf_label = None
   291→   291→   291→                # v4: reset pasal when BAB changes to prevent heading-as-chunk
   292→   292→   292→                self.current_pasal = None
   293→   293→   293→                # v6: Check if a Pasal reference is glued to the BAB line
   294→   294→   294→                # (e.g., 'BAB VIIA DEWAN PERWAKILAN DAERAH * Pasal 22C')
   295→   295→   295→                m_glued = self.RE_PASAL_EMBEDDED.search(text)
   296→   296→   296→                if m_glued:
   297→   297→   297→                    self.current_pasal = m_glued.group(1)
   298→   298→   298→                    self.current_ayat = None
   299→   299→   299→                continue
   300→   300→   300→
   301→   301→   301→            m = self.RE_BAGIAN.match(text)
   302→   302→   302→            if m:
   303→   303→   303→                self._flush_ayat()
   304→   304→   304→                self._flush_pasal()
   305→   305→   305→                self.current_bagian = m.group(1)
   306→   306→   306→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   307→   307→   307→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   308→   308→   308→                if raw_label:
   309→   309→   309→                    self.current_bagian_label = raw_label
   310→   310→   310→                else:
   311→   311→   311→                    self.current_bagian_label = None
   312→   312→   312→                self.current_paragraf = None
   313→   313→   313→                self.current_paragraf_label = None
   314→   314→   314→                continue
   315→   315→   315→
   316→   316→   316→            m = self.RE_PARAGRAF.match(text)
   317→   317→   317→            if m:
   318→   318→   318→                self._flush_ayat()
   319→   319→   319→                self._flush_pasal()
   320→   320→   320→                self.current_paragraf = m.group(1)
   321→   321→   321→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   322→   322→   322→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   323→   323→   323→                if raw_label:
   324→   324→   324→                    self.current_paragraf_label = raw_label
   325→   325→   325→                else:
   326→   326→   326→                    self.current_paragraf_label = None
   327→   327→   327→                continue
   328→   328→   328→
   329→   329→   329→            m = self.RE_PASAL.match(text)
   330→   330→   330→            if m:
   331→   331→   331→                self._flush_ayat()
   332→   332→   332→                self._flush_pasal()
   333→   333→   333→                raw_num = m.group(1)
   334→   334→   334→                # v4: normalize space-separated digits ("1 1" -> "11", "1 10" -> "110")
   335→   335→   335→                self.current_pasal = re.sub(r'\s+', '', raw_num) if re.match(r'^\d', raw_num) else raw_num
   336→   336→   336→                self.current_ayat = None
   337→   337→   337→                # v4: capture any remaining text on the same line as pasal heading
   338→   338→   338→                remaining = text[m.end():].strip().lstrip('.').strip()
   339→   339→   339→                if remaining and len(remaining) > 5:
   340→   340→   340→                    self.current_pasal_lines.append(remaining)
   341→   341→   341→                continue
   342→   342→   342→
   343→   343→   343→            m = self.RE_AYAT.match(text)
   344→   344→   344→            if m and self.current_pasal:
   345→   345→   345→                self._flush_ayat()
   346→   346→   346→                self.current_ayat = m.group(1)
   347→   347→   347→                ayat_text = self.RE_AYAT.sub('', text).strip()
   348→   348→   348→                if ayat_text:
   349→   349→   349→                    self.current_ayat_lines.append(ayat_text)
   350→   350→   350→                continue
   351→   351→   351→
   352→   352→   352→            # v6: Check for embedded pasal in non-structural line
   353→   353→   353→            # (e.g., 'PEMILIHAN UMUM * Pasal 22E' — BAB label with glued pasal)
   354→   354→   354→            m_emb = self.RE_PASAL_EMBEDDED.search(text)
   355→   355→   355→            if m_emb and self.current_pasal:
   356→   356→   356→                self._flush_ayat()
   357→   357→   357→                self._flush_pasal()
   358→   358→   358→                self.current_pasal = m_emb.group(1)
   359→   359→   359→                self.current_ayat = None
   360→   360→   360→                continue
   361→   361→   361→
   362→   362→   362→            # If we're inside a pasal but no ayat detected, accumulate lines
   363→   363→   363→            if self.current_pasal:
   364→   364→   364→                if self.current_ayat:
   365→   365→   365→                    self.current_ayat_lines.append(text)
   366→   366→   366→                else:
   367→   367→   367→                    self.current_pasal_lines.append(text)
   368→   368→   368→
   369→   369→   369→        # Flush remaining
   370→   370→   370→        self._flush_ayat()
   371→   371→   371→        self._flush_pasal()
   372→   372→   372→        self._flush_quoted_amendment()
   373→   373→   373→
   374→   374→   374→        # v6: Deduplicate truncated page-break fragments
   375→   375→   375→        self._dedup_truncated_fragments()
   376→   376→   376→
   377→   377→   377→        return self.chunks
   378→   378→   378→
   379→   379→   379→    def _flush_ayat(self):
   380→   380→   380→        """Flush accumulated ayat lines as a chunk."""
   381→   381→   381→        if not self.current_ayat or not self.current_ayat_lines:
   382→   382→   382→            self.current_ayat_lines = []
   383→   383→   383→            return
   384→   384→   384→
   385→   385→   385→        text = ' '.join(self.current_ayat_lines).strip()
   386→   386→   386→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   387→   387→   387→            self.current_ayat_lines = []
   388→   388→   388→            return
   389→   389→   389→
   390→   390→   390→        headings = _make_bab_headings(
   391→   391→   391→            self.current_bab, self.current_bab_label,
   392→   392→   392→            self.current_bagian, self.current_bagian_label,
   393→   393→   393→            self.current_paragraf, self.current_paragraf_label,
   394→   394→   394→            pasal_str=f"Pasal {self.current_pasal}",
   395→   395→   395→            ayat_str=f"Ayat ({self.current_ayat})"
   396→   396→   396→        )
   397→   397→   397→
   398→   398→   398→        path = build_path(headings)
   399→   399→   399→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   400→   400→   400→                            pasal=self.current_pasal, ayat=self.current_ayat)
   401→   401→   401→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   402→   402→   402→
   403→   403→   403→        self.chunks.append({
   404→   404→   404→            "id": chunk_id,
   405→   405→   405→            "text": text,
   406→   406→   406→            "metadata": {
   407→   407→   407→                "doc_type": self.doc_type,
   408→   408→   408→                "issuer": self.issuer,
   409→   409→   409→                "year": self.year,
   410→   410→   410→                "bab": self.current_bab or None,
   411→   411→   411→                "pasal": self.current_pasal,
   412→   412→   412→                "ayat": self.current_ayat,
   413→   413→   413→                "parent": parent_id,
   414→   414→   414→                "path": path,
   415→   415→   415→                "status": "active",
   416→   416→   416→            }
   417→   417→   417→        })
   418→   418→   418→        self.current_ayat_lines = []
   419→   419→   419→
   420→   420→   420→    def _flush_pasal(self):
   421→   421→   421→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   422→   422→   422→        if not self.current_pasal or not self.current_pasal_lines:
   423→   423→   423→            self.current_pasal_lines = []
   424→   424→   424→            return
   425→   425→   425→
   426→   426→   426→        text = ' '.join(self.current_pasal_lines).strip()
   427→   427→   427→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   428→   428→   428→            self.current_pasal_lines = []
   429→   429→   429→            return
   430→   430→   430→
   431→   431→   431→        headings = _make_bab_headings(
   432→   432→   432→            self.current_bab, self.current_bab_label,
   433→   433→   433→            self.current_bagian, self.current_bagian_label,
   434→   434→   434→            self.current_paragraf, self.current_paragraf_label,
   435→   435→   435→            pasal_str=f"Pasal {self.current_pasal}",
   436→   436→   436→        )
   437→   437→   437→        path = build_path(headings)
   438→   438→   438→        chunk_id = build_id(self.doc_type, self.nomor, self.year, pasal=self.current_pasal)
   439→   439→   439→
   440→   440→   440→        self.chunks.append({
   441→   441→   441→            "id": chunk_id,
   442→   442→   442→            "text": text,
   443→   443→   443→            "metadata": {
   444→   444→   444→                "doc_type": self.doc_type,
   445→   445→   445→                "issuer": self.issuer,
   446→   446→   446→                "year": self.year,
   447→   447→   447→                "bab": self.current_bab or None,
   448→   448→   448→                "pasal": self.current_pasal,
   449→   449→   449→                "ayat": None,
   450→   450→   450→                "parent": None,
   451→   451→   451→                "path": path,
   452→   452→   452→                "status": "active",
   453→   453→   453→            }
   454→   454→   454→        })
   455→   455→   455→        self.current_pasal_lines = []

   457→   457→   457→    def _flush_quoted_amendment(self):
   458→   458→   458→        """Flush quoted amendment text as a chunk with status='quoted_amendment'.
   459→   459→   459→        
   460→   460→   460→        Fix for bug #6: instead of silently skipping quoted amendments,
   461→   461→   461→        we now chunk them with a clear status tag so retrieval can
   462→   462→   462→        distinguish live law from historical amendment pointers.
   463→   463→   463→        """
   464→   464→   464→        if not self.quoted_amendment_lines:
   465→   465→   465→            self.in_quoted_amendment = False
   466→   466→   466→            self.quoted_amendment_source = None
   467→   467→   467→            return
   468→   468→   468→
   469→   469→   469→        text = ' '.join(self.quoted_amendment_lines).strip()

   537→   537→   537→        """v6: Remove truncated page-break fragments that duplicate a longer chunk.
   538→   538→   538→        
   539→   539→   539→        When text spans a page break, the bottom-of-page fragment (10-50 chars)
   540→   540→   540→        gets the same pasal:ayat ID as the continuation on the next page.
   541→   541→   541→        Keep the longer chunk, drop the fragment.
   542→   542→   542→        """
   543→   543→   543→        from collections import defaultdict
   544→   544→   544→        id_chunks = defaultdict(list)
   545→   545→   545→        for i, chunk in enumerate(self.chunks):
   546→   546→   546→            id_chunks[chunk["id"]].append(i)
   547→   547→   547→        
   548→   548→   548→        to_remove = set()
   549→   549→   549→        for chunk_id, indices in id_chunks.items():
   550→   550→   550→            if len(indices) < 2:
   551→   551→   551→                continue
   552→   552→   552→            pairs = [(i, len(self.chunks[i]["text"])) for i in indices]
   553→   553→   553→            pairs.sort(key=lambda x: -x[1])
   554→   554→   554→            longest_idx, longest_len = pairs[0]
   555→   555→   555→            
   556→   556→   556→            for idx, text_len in pairs[1:]:
   557→   557→   557→                if text_len < longest_len * 0.5:
   558→   558→   558→                    shorter = self.chunks[idx]["text"].strip()
   559→   559→   559→                    longer = self.chunks[longest_idx]["text"].strip()
   560→   560→   560→                    if longer.endswith(shorter) or shorter in longer:
   561→   561→   561→                        to_remove.add(idx)
   562→   562→   562→        
   563→   563→   563→        if to_remove:
   564→   564→   564→            self.chunks = [c for i, c in enumerate(self.chunks) if i not in to_remove]

   868→   868→   868→def extract_nomor_perkara(lines):
   869→   869→   869→    """Extract case number (e.g. 96/PUU-XVI/2018) from court ruling lines."""

   877→   877→   877→def chunk_pdf(pdf_path: str, directory: str, family_map: Dict) -> List[Dict]:
   878→   878→   878→    """
   879→   879→   879→    Main entry point. Extract + parse a PDF into chunks.
   880→   880→   880→    Returns list of chunk dicts with id, text, metadata.
   881→   881→   881→    """
   882→   882→   882→    dir_config = family_map.get(directory, {})
   883→   883→   883→    family = dir_config.get("family", "A")
   884→   884→   884→    default_doc_type = dir_config.get("doc_type", "Unknown")
   885→   885→   885→    default_issuer = dir_config.get("issuer", "Unknown")
   886→   886→   886→
   887→   887→   887→    try:
   888→   888→   888→        lines, meta = extract_clean(pdf_path)
   889→   889→   889→    except Exception as e:
   890→   890→   890→        return [{"id": f"ERROR:{directory}", "text": str(e), "metadata": {"error": True}}]
   891→   891→   891→
   892→   892→   892→    if not lines:
   893→   893→   893→        return [{"id": f"EMPTY:{directory}", "text": "No extractable text (possibly scanned PDF)", "metadata": {"error": True, "needs_ocr": True}}]
   894→   894→   894→
   895→   895→   895→    # v4 fix: Detect doc_type from title block FIRST, directory as fallback.
   896→   896→   896→    # Use page-1-only RAW lines (pre-boilerplate) since boilerplate stripping
   897→   897→   897→    # removes the actual document title (e.g. "PERATURAN PRESIDEN").
   898→   898→   898→    # Only page 1 to avoid body citations polluting the detection.
   899→   899→   899→    page1_raw = [l["text"].strip() for l in lines if l.get("page") == 1]
   900→   900→   900→    raw_title_block = " ".join(page1_raw[:15]) if page1_raw else ""
   901→   901→   901→    detected_type, detected_issuer = detect_doc_type_from_title(raw_title_block)
   902→   902→   902→    
   903→   903→   903→    # If page 1 was empty (image) or detection failed, try cleaned lines
   904→   904→   904→    if detected_type == "Unknown":
   905→   905→   905→        clean_title = extract_title_block(lines, max_first_n=5)
   906→   906→   906→        detected_type, detected_issuer = detect_doc_type_from_title(clean_title)
   907→   907→   907→    
   908→   908→   908→    # Title block wins when it returns a real type; directory is fallback only
   909→   909→   909→    if detected_type != "Unknown":
   910→   910→   910→        doc_type = detected_type
   911→   911→   911→        issuer = detected_issuer
   912→   912→   912→        if default_doc_type != "Unknown" and detected_type != default_doc_type:
   913→   913→   913→            print(f"  [WARN] doc_type: title={detected_type} overrides dir={default_doc_type} for {directory}")
   914→   914→   914→    elif default_doc_type != "Unknown":
   915→   915→   915→        doc_type = default_doc_type
   916→   916→   916→        issuer = default_issuer
   917→   917→   917→    else:
   918→   918→   918→        doc_type = "Unknown"
   919→   919→   919→        issuer = "Unknown"
   920→   920→   920→    nomor = meta.get("nomor")
   921→   921→   921→    year = meta.get("year")
   922→   922→   922→
   923→   923→   923→    # v4: If page 1 was empty (image), always re-extract nomor/year from post-decision text
   924→   924→   924→    # since the pre-decision text contains citations, not the document's own number.
   925→   925→   925→    page1_has_text = any(l.get("page") == 1 for l in lines)
   926→   926→   926→    if not page1_has_text and meta.get("total_pages", 0) > 1:
   927→   927→   927→        post_nomor, post_year = extract_nomor_year_post_decision(lines)
   928→   928→   928→        if post_nomor and post_nomor != nomor:
   929→   929→   929→            nomor = post_nomor
   930→   930→   930→            if post_year:
   931→   931→   931→                year = post_year
   932→   932→   932→            print(f"  [INFO] Post-decision: nomor={nomor}, year={year}")
   933→   933→   933→        # v4: Filename fallback when page 1 is image and text extraction found wrong number
   934→   934→   934→        if (not post_nomor) or (nomor == meta.get('nomor') and re.search(r'(\d+)[_.](\d{4})', os.path.basename(pdf_path))):
   935→   935→   935→            fn = os.path.basename(pdf_path)
   936→   936→   936→            fn_match = re.search(r'(\d+)[_.](\d{4})', fn)
   937→   937→   937→            if fn_match:
   938→   938→   938→                nomor = fn_match.group(1)
   939→   939→   939→                year = fn_match.group(2)
   940→   940→   940→                print(f"  [INFO] Filename fallback: nomor={nomor}, year={year}")
   941→   941→   941→    
   942→   942→   942→    # Directory-specific overrides for image-cover documents where title can't be read
   943→   943→   943→    if directory == "JDIH_KPU" and doc_type in ("Peraturan", "Unknown"):
   944→   944→   944→        doc_type = "PKPU"
   945→   945→   945→        issuer = "Komisi Pemilihan Umum"
   946→   946→   946→
   947→   947→   947→    # Parse based on family
   948→   948→   948→    if family == "A":
   949→   949→   949→        parser = FamilyAParser(lines, doc_type, nomor, year, issuer)
   950→   950→   950→        chunks = parser.parse()
   951→   951→   951→    elif family == "B":
   952→   952→   952→        parser = FamilyBParser(lines, doc_type, nomor, year, issuer)
   953→   953→   953→        chunks = parser.parse()
   954→   954→   954→    elif family == "C":
   955→   955→   955→        nomor_perkara = extract_nomor_perkara(lines)
   956→   956→   956→        parser = FamilyCParser(lines, doc_type, nomor, year, issuer, nomor_perkara)
   957→   957→   957→        chunks = parser.parse()
   958→   958→   958→    else:
   959→   959→   959→        chunks = [{"id": f"UNKNOWN_FAMILY:{directory}", "text": "", "metadata": {"error": True}}]
   960→   960→   960→
   961→   961→   961→    # Add extraction metadata to all chunks
   962→   962→   962→    for chunk in chunks:
   963→   963→   963→        chunk["metadata"]["source_file"] = meta["file"]
   964→   964→   964→        chunk["metadata"]["directory"] = directory
   965→   965→   965→        chunk["metadata"]["family"] = family
   966→   966→   966→        if nomor:
   967→   967→   967→            chunk["metadata"]["nomor"] = nomor
   968→   968→   968→        if year:
   969→   969→   969→            chunk["metadata"]["year"] = year
   970→   970→   970→
   971→   971→   971→    return chunks
