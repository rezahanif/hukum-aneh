    33→    33→def build_id(doc_type, nomor, year, **parts):
    34→    34→    """Build chunk ID. Handles both hierarchical and flat structures."""
    35→    35→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    36→    36→        # Court rulings use nomor_perkara
    37→    37→        nomor_perkara = parts.get("nomor_perkara", nomor or "?")
    38→    38→        chunk_type = parts.get("chunk_type", "amar")
    39→    39→        chunk_num = parts.get("chunk_num", "?")
    40→    40→        return f"{doc_type}:{nomor_perkara}:{chunk_type}:{chunk_num}"
    41→    41→    elif parts.get("chunk_type") in ("diktum", "pertimbangan"):
    42→    42→        # Family B: flat numbered
    43→    43→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{parts['chunk_type']}:{parts['chunk_num']}"
    44→    44→    else:
    45→    45→        # Family A: hierarchical
    46→    46→        pasal = parts.get("pasal", "?")
    47→    47→        ayat = parts.get("ayat", "?")
    48→    48→        if ayat and ayat != "?":
    49→    49→            return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}:{ayat}"

    70→    70→class FamilyAParser:
    71→    71→    """
    72→    72→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
    73→    73→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
    74→    74→    """
    75→    75→
    76→    76→    # Heading patterns (order matters: higher levels first)
    77→    77→    RE_BAB = re.compile(r'^\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
    78→    78→    RE_BAGIAN = re.compile(r'^\s*BAGIAN\s+(\w+)', re.IGNORECASE)
    79→    79→    RE_PARAGRAF = re.compile(r'^\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
    80→    80→    RE_PASAL = re.compile(r'^\s*pasal\s+([IVXLCDM]+|\d+[a-zA-Z]?)\s*\.?\s*.{0,5}$', re.IGNORECASE)
    81→    81→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
    82→    82→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
    83→    83→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
    84→    84→
    85→    85→    # Quoted amendment detection
    86→    86→    RE_AMENDMENT_TRIGGER = re.compile(
    87→    87→        r'(?:diubah|diganti|dihapus)\s+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
    88→    88→        re.IGNORECASE
    89→    89→    )
    90→    90→    RE_QUOTE_START = re.compile(r'^\s*["“]')
    91→    91→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
    92→    92→
    93→    93→    # Penutup (closing section) marker
    94→    94→    RE_PENUTUP = re.compile(r'^(?:PENUTUP|penutup)\b')

    96→    96→    def __init__(self, lines, doc_type, nomor, year, issuer):
    97→    97→        self.lines = lines
    98→    98→        self.doc_type = doc_type
    99→    99→        self.nomor = nomor
   100→   100→        self.year = year
   101→   101→        self.issuer = issuer
   102→   102→        self.chunks = []
   103→   103→
   104→   104→        # Heading stack
   105→   105→        self.current_bab = None
   106→   106→        self.current_bab_label = None
   107→   107→        self.current_bagian = None
   108→   108→        self.current_bagian_label = None
   109→   109→        self.current_paragraf = None
   110→   110→        self.current_paragraf_label = None
   111→   111→        self.current_pasal = None
   112→   112→        self.current_ayat = None
   113→   113→
   114→   114→        # State
   115→   115→        self.in_preamble = True
   116→   116→        self.in_penutup = False
   117→   117→        self.in_quoted_amendment = False
   118→   118→        self.quoted_amendment_note = None
   119→   119→        self.current_ayat_lines = []
   120→   120→        self.current_pasal_lines = []

   122→   122→    def parse(self) -> List[Dict]:
   123→   123→        """Main parse loop. Returns list of chunks."""
   124→   124→        for i, line in enumerate(self.lines):
   125→   125→            text = line["text"].strip()
   126→   126→            if not text:
   127→   127→                continue
   128→   128→
   129→   129→            text_upper = text.upper()
   130→   130→
   131→   131→            # Detect preamble end / body start
   132→   132→            if self.in_preamble:
   133→   133→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   134→   134→                    self.in_preamble = False
   135→   135→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
   136→   136→                    self.in_preamble = False
   137→   137→                    continue  # skip the decision keyword line itself
   138→   138→                else:
   139→   139→                    continue  # still in preamble
   140→   140→
   141→   141→            # Detect penutup
   142→   142→            if self.RE_PENUTUP.match(text):
   143→   143→                self.in_penutup = True
   144→   144→                # Flush any pending ayat
   145→   145→                self._flush_ayat()
   146→   146→                self._flush_pasal()
   147→   147→                continue
   148→   148→
   149→   149→            if self.in_penutup:
   150→   150→                continue  # skip penutup
   151→   151→
   152→   152→            # Check for quoted amendment
   153→   153→            if not self.in_quoted_amendment:
   154→   154→                if self.RE_AMENDMENT_TRIGGER.search(text):
   155→   155→                    self.in_quoted_amendment = True
   156→   156→                    self.quoted_amendment_note = text[:100]
   157→   157→                    self._flush_ayat()
   158→   158→                    continue
   159→   159→                if self.RE_QUOTE_START.match(text):
   160→   160→                    # Could be inline quote
   161→   161→                    self.in_quoted_amendment = True
   162→   162→                    self.quoted_amendment_note = "Quoted passage (inline)"
   163→   163→
   164→   164→            if self.in_quoted_amendment:
   165→   165→                if self.RE_QUOTE_END.search(text):
   166→   166→                    self.in_quoted_amendment = False
   167→   167→                    self.quoted_amendment_note = None
   168→   168→                continue  # skip all quoted text
   169→   169→
   170→   170→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   171→   171→            m = self.RE_BAB.match(text)
   172→   172→            if m:
   173→   173→                self._flush_ayat()
   174→   174→                self._flush_pasal()
   175→   175→                self.current_bab = m.group(1)
   176→   176→                # Get label text after BAB X
   177→   177→                label_match = re.match(r'^\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   178→   178→                self.current_bab_label = label_match.group(1).strip().rstrip('.') if label_match else f"BAB {m.group(1)}"
   179→   179→                self.current_bagian = None
   180→   180→                self.current_bagian_label = None
   181→   181→                self.current_paragraf = None
   182→   182→                self.current_paragraf_label = None
   183→   183→                continue
   184→   184→
   185→   185→            m = self.RE_BAGIAN.match(text)
   186→   186→            if m:
   187→   187→                self._flush_ayat()
   188→   188→                self._flush_pasal()
   189→   189→                self.current_bagian = m.group(1)
   190→   190→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   191→   191→                self.current_bagian_label = label_match.group(1).strip().rstrip('.') if label_match else f"Bagian {m.group(1)}"
   192→   192→                self.current_paragraf = None
   193→   193→                self.current_paragraf_label = None
   194→   194→                continue
   195→   195→
   196→   196→            m = self.RE_PARAGRAF.match(text)
   197→   197→            if m:
   198→   198→                self._flush_ayat()
   199→   199→                self._flush_pasal()
   200→   200→                self.current_paragraf = m.group(1)
   201→   201→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   202→   202→                self.current_paragraf_label = label_match.group(1).strip().rstrip('.') if label_match else f"Paragraf {m.group(1)}"
   203→   203→                continue
   204→   204→
   205→   205→            m = self.RE_PASAL.match(text)
   206→   206→            if m:
   207→   207→                self._flush_ayat()
   208→   208→                self._flush_pasal()
   209→   209→                self.current_pasal = m.group(1)
   210→   210→                self.current_ayat = None
   211→   211→                continue
   212→   212→
   213→   213→            m = self.RE_AYAT.match(text)
   214→   214→            if m and self.current_pasal:
   215→   215→                self._flush_ayat()
   216→   216→                self.current_ayat = m.group(1)
   217→   217→                # The rest of the line after (N) is the start of ayat text
   218→   218→                ayat_text = self.RE_AYAT.sub('', text).strip()
   219→   219→                if ayat_text:
   220→   220→                    self.current_ayat_lines.append(ayat_text)
   221→   221→                continue
   222→   222→
   223→   223→            # If we're inside a pasal but no ayat detected, accumulate lines
   224→   224→            if self.current_pasal:
   225→   225→                if self.current_ayat:
   226→   226→                    self.current_ayat_lines.append(text)
   227→   227→                else:
   228→   228→                    self.current_pasal_lines.append(text)
   229→   229→
   230→   230→        # Flush remaining
   231→   231→        self._flush_ayat()
   232→   232→        self._flush_pasal()
   233→   233→
   234→   234→        return self.chunks
   235→   235→
   236→   236→    def _flush_ayat(self):
   237→   237→        """Flush accumulated ayat lines as a chunk."""
   238→   238→        if not self.current_ayat or not self.current_ayat_lines:
   239→   239→            self.current_ayat_lines = []
   240→   240→            return
   241→   241→
   242→   242→        text = ' '.join(self.current_ayat_lines).strip()
   243→   243→        if not text:
   244→   244→            self.current_ayat_lines = []
   245→   245→            return
   246→   246→
   247→   247→        headings = [
   248→   248→            f"BAB {self.current_bab}" if self.current_bab else None,
   249→   249→            self.current_bab_label,
   250→   250→            f"Bagian {self.current_bagian}" if self.current_bagian else None,
   251→   251→            self.current_bagian_label,
   252→   252→            f"Paragraf {self.current_paragraf}" if self.current_paragraf else None,
   253→   253→            self.current_paragraf_label,
   254→   254→            f"Pasal {self.current_pasal}" if self.current_pasal else None,
   255→   255→            f"Ayat ({self.current_ayat})",
   256→   256→        ]
   257→   257→        clean_headings = [h for h in headings if h]
   258→   258→
   259→   259→        path = build_path(clean_headings)
   260→   260→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   261→   261→                            pasal=self.current_pasal, ayat=self.current_ayat)
   262→   262→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   263→   263→
   264→   264→        self.chunks.append({
   265→   265→            "id": chunk_id,
   266→   266→            "text": text,
   267→   267→            "metadata": {
   268→   268→                "doc_type": self.doc_type,
   269→   269→                "issuer": self.issuer,
   270→   270→                "year": self.year,
   271→   271→                "bab": self.current_bab or None,
   272→   272→                "pasal": self.current_pasal,
   273→   273→                "ayat": self.current_ayat,
   274→   274→                "parent": parent_id,
   275→   275→                "path": path,
   276→   276→                "status": "active",
   277→   277→            }
   278→   278→        })
   279→   279→        self.current_ayat_lines = []
   280→   280→
   281→   281→    def _flush_pasal(self):
   282→   282→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   283→   283→        if not self.current_pasal or not self.current_pasal_lines:
   284→   284→            self.current_pasal_lines = []
   285→   285→            return
   286→   286→
   287→   287→        text = ' '.join(self.current_pasal_lines).strip()
   288→   288→        if not text:
   289→   289→            self.current_pasal_lines = []
   290→   290→            return
   291→   291→
   292→   292→        headings = [
   293→   293→            f"BAB {self.current_bab}" if self.current_bab else None,
   294→   294→            self.current_bab_label,
   295→   295→            f"Bagian {self.current_bagian}" if self.current_bagian else None,
   296→   296→            self.current_bagian_label,
   297→   297→            f"Paragraf {self.current_paragraf}" if self.current_paragraf else None,
   298→   298→            self.current_paragraf_label,
   299→   299→            f"Pasal {self.current_pasal}",
   300→   300→        ]
   301→   301→        clean_headings = [h for h in headings if h]
   302→   302→        path = build_path(clean_headings)
   303→   303→        chunk_id = build_id(self.doc_type, self.nomor, self.year, pasal=self.current_pasal)
   304→   304→
   305→   305→        self.chunks.append({
   306→   306→            "id": chunk_id,
   307→   307→            "text": text,
   308→   308→            "metadata": {
   309→   309→                "doc_type": self.doc_type,
   310→   310→                "issuer": self.issuer,
   311→   311→                "year": self.year,
   312→   312→                "bab": self.current_bab or None,
   313→   313→                "pasal": self.current_pasal,
   314→   314→                "ayat": None,
   315→   315→                "parent": None,
   316→   316→                "path": path,
   317→   317→                "status": "active",
   318→   318→            }
   319→   319→        })
   320→   320→        self.current_pasal_lines = []

   324→   324→# FAMILY B: DECREE/DECISION PARSER

   326→   326→
   327→   327→class FamilyBParser:
   328→   328→    """
   329→   329→    Parses decree/decision documents: Menimbang > Mengingat > MEMUTUSKAN/MENETAPKAN > Diktum.
   330→   330→    Handles: Keppres, Inpres, Tap_MPR, Kepmen.
   331→   331→    """
   332→   332→
   333→   333→    RE_MENIMBANG = re.compile(r'^\s*menimbang\s*:?\s*$', re.IGNORECASE)
   334→   334→    RE_MENGINGAT = re.compile(r'^\s*mengingat\s*:?\s*$', re.IGNORECASE)
   335→   335→    # Decision keywords: match keyword at/near end of line, allowing a short prefix
   336→   336→    # like "dengan ini" before the keyword. This handles both standalone "MEMUTUSKAN:"
   337→   337→    # and inline "dengan ini menginstruksikan:" patterns.
   338→   338→    RE_MEMUTUSKAN = re.compile(r'^(?:.{0,30}\s+)?memutuskan\s*:?\s*$', re.IGNORECASE)
   339→   339→    RE_MENETAPKAN = re.compile(r'^(?:.{0,30}\s+)?menetapkan\s*:?\s*$', re.IGNORECASE)
   340→   340→    RE_MENINSTRUKSIKAN = re.compile(r'^(?:.{0,30}\s+)?menginstruksikan\s*:?\s*$', re.IGNORECASE)
   341→   341→    # Item markers: trailing space is optional (handles bare "1." or "a.")
   342→   342→    RE_ITEM_LETTER = re.compile(r'^\s*([a-z])\s*\.\s*')
   343→   343→    RE_ITEM_NUMBER = re.compile(r'^\s*(\d+)\s*\.\s*')
   344→   344→    # Indonesian ordinal words used as diktum markers (KESATU, KEDUA, KETIGA, ...)
   345→   345→    RE_ORDINAL = re.compile(
   346→   346→        r'^\s*(KE(?:SATU|DUA|TIGA|EMPAT|LIMA|NAM|TUJUH|DELAPAN|SEMBILAN|SEPULUH|'
   347→   347→        r'SEBELAS|DUA BELAS|TIGA BELAS|EMPAT BELAS|LIMA BELAS|ENAM BELAS|'
   348→   348→        r'TUJUH BELAS|DELAPAN BELAS|SEMBILAN BELAS|DUA PULUH))\b',
   349→   349→        re.IGNORECASE
   350→   350→    )
   351→   351→
   352→   352→    def __init__(self, lines, doc_type, nomor, year, issuer):
   353→   353→        self.lines = lines
   354→   354→        self.doc_type = doc_type
   355→   355→        self.nomor = nomor
   356→   356→        self.year = year
   357→   357→        self.issuer = issuer
   358→   358→        self.chunks = []
   359→   359→
   360→   360→        self.section = "pre"  # pre, menimbang, mengingat, keputusan, done
   361→   361→        self.current_item = None
   362→   362→        self.current_item_lines = []
   363→   363→        self.preamble_lines = []
   364→   364→        self.decision_keyword = None  # MEMUTUSKAN, MENETAPKAN, etc.
   365→   365→
   366→   366→    def parse(self) -> List[Dict]:
   367→   367→        for line in self.lines:
   368→   368→            text = line["text"].strip()
   369→   369→            if not text:
   370→   370→                continue
   371→   371→
   372→   372→            text_upper = text.upper()
   373→   373→
   374→   374→            # Section transitions
   375→   375→            if self.RE_MENIMBANG.match(text):
   376→   376→                self._flush_preamble()
   377→   377→                self.section = "menimbang"
   378→   378→                self.current_item = None
   379→   379→                continue
   380→   380→
   381→   381→            if self.RE_MENGINGAT.match(text):
   382→   382→                self._flush_preamble()
   383→   383→                self.section = "mengingat"
   384→   384→                self.current_item = None
   385→   385→                continue
   386→   386→
   387→   387→            # Decision keyword triggers (regex allows short prefix like "dengan ini").
   388→   388→            # Guard: only trigger if not already in keputusan, to avoid
   389→   389→            # re-triggering on "menetapkan" appearing inside diktum item text.
   390→   390→            if self.section not in ("keputusan", "done"):
   391→   391→                if self.RE_MENINSTRUKSIKAN.match(text):
   392→   392→                    self._flush_item()
   393→   393→                    self.section = "keputusan"
   394→   394→                    self.decision_keyword = "MENGINSTRUKSIKAN"
   395→   395→                    self.current_item = None
   396→   396→                    continue
   397→   397→
   398→   398→                if self.RE_MEMUTUSKAN.match(text):
   399→   399→                    self._flush_item()
   400→   400→                    self.section = "keputusan"
   401→   401→                    self.decision_keyword = "MEMUTUSKAN"
   402→   402→                    self.current_item = None
   403→   403→                    continue
   404→   404→
   405→   405→                if self.RE_MENETAPKAN.match(text):
   406→   406→                    self._flush_item()
   407→   407→                    self.section = "keputusan"
   408→   408→                    self.decision_keyword = "MENETAPKAN"
   409→   409→                    self.current_item = None
   410→   410→                    continue
   411→   411→
   412→   412→            if self.section == "pre":
   413→   413→                continue  # skip title block
   414→   414→
   415→   415→            if self.section == "done":
   416→   416→                continue
   417→   417→
   418→   418→            # Check for signature block (end of substantive content)
   419→   419→            # Must come AFTER pre/done guards to avoid false triggers in preamble.
   420→   420→            text_lower = text.lower()
   421→   421→            if "ttd" in text_lower and len(text) < 20:
   422→   422→                self._flush_item()
   423→   423→                self.section = "done"
   424→   424→                continue
   425→   425→
   426→   426→            # Check for new item in keputusan section
   427→   427→            if self.section == "keputusan":
   428→   428→                m_letter = self.RE_ITEM_LETTER.match(text)
   429→   429→                m_number = self.RE_ITEM_NUMBER.match(text)
   430→   430→                m_ordinal = self.RE_ORDINAL.match(text)
   431→   431→
   432→   432→                if m_letter or m_number or m_ordinal:
   433→   433→                    self._flush_item()
   434→   434→                    if m_ordinal:
   435→   435→                        self.current_item = m_ordinal.group(1).upper()
   436→   436→                    elif m_letter:
   437→   437→                        self.current_item = m_letter.group(1)
   438→   438→                    else:
   439→   439→                        self.current_item = m_number.group(1)
   440→   440→                    # Get text after the item marker
   441→   441→                    m = m_letter or m_number or m_ordinal
   442→   442→                    remaining = text[m.end():].strip()
   443→   443→                    if remaining:
   444→   444→                        self.current_item_lines.append(remaining)
   445→   445→                    continue
   446→   446→
   447→   447→                # Sub-items: continue accumulating under current item
   448→   448→                if self.current_item:
   449→   449→                    self.current_item_lines.append(text)
   450→   450→                continue
   451→   451→
   452→   452→            # menimbang/mengingat: accumulate as preamble (skip for chunking or tag separately)
   453→   453→            # Per the plan, optionally chunk these with section tag
   454→   454→
   455→   455→        self._flush_item()
   456→   456→        return self.chunks

   498→   498→class FamilyCParser:
   499→   499→    """
   500→   500→    Parses court rulings: Duduk Perkara > Menimbang > MENGADILI/MENETAPKAN > Amar.
   501→   501→    Handles: Putusan MK, Putusan MA.
   502→   502→    """
   503→   503→
   504→   504→    RE_MENGADILI = re.compile(r'^\s*mengadili\s*:?\s*$', re.IGNORECASE)
   505→   505→    RE_MENETAPKAN = re.compile(r'^\s*menetapkan\s*:?\s*$', re.IGNORECASE)
   506→   506→    RE_AMAR = re.compile(r'^\s*(\d+)\s*\.\s+')
   507→   507→    RE_PUTUSAN = re.compile(r'^\s*putusan\s*:?\s*$', re.IGNORECASE)
   508→   508→
   509→   509→    def __init__(self, lines, doc_type, nomor, year, issuer, nomor_perkara=None):
   510→   510→        self.lines = lines
   511→   511→        self.doc_type = doc_type
   512→   512→        self.nomor = nomor
   513→   513→        self.year = year
   514→   514→        self.issuer = issuer
   515→   515→        self.nomor_perkara = nomor_perkara
   516→   516→        self.chunks = []
   517→   517→
   518→   518→        self.section = "pre"  # pre, menimbang, mengadili, amar, done
   519→   519→        self.current_amar = None
   520→   520→        self.current_amar_lines = []
   521→   521→        self.pertimbangan_lines = []
   522→   522→
   523→   523→    def parse(self) -> List[Dict]:
   524→   524→        for line in self.lines:
   525→   525→            text = line["text"].strip()
   526→   526→            if not text:
   527→   527→                continue
   528→   528→
   529→   529→            text_upper = text.upper()
   530→   530→
   531→   531→            if self.RE_MENGADILI.match(text):
   532→   532→                self._flush_amar()
   533→   533→                self.section = "amar"
   534→   534→                self.decision_keyword = "MENGADILI"
   535→   535→                continue
   536→   536→
   537→   537→            if self.RE_MENETAPKAN.match(text):
   538→   538→                self._flush_amar()
   539→   539→                self.section = "amar"
   540→   540→                self.decision_keyword = "MENETAPKAN"
   541→   541→                continue
   542→   542→
   543→   543→            if self.section == "pre":
   544→   544→                continue
   545→   545→
   546→   546→            if self.section == "done":
   547→   547→                continue
   548→   548→
   549→   549→            if self.section == "amar":
   550→   550→                m = self.RE_AMAR.match(text)
   551→   551→                if m:
   552→   552→                    self._flush_amar()
   553→   553→                    self.current_amar = m.group(1)
   554→   554→                    remaining = text[m.end():].strip()
   555→   555→                    if remaining:
   556→   556→                        self.current_amar_lines.append(remaining)
   557→   557→                    continue
   558→   558→
   559→   559→                if self.current_amar:
   560→   560→                    self.current_amar_lines.append(text)
   561→   561→
   562→   562→                # Detect end: signature block
   563→   563→                if re.search(r'\bttd\b|\bHAKIM\b|\bKETUA\b', text_upper) and len(text) < 30:
   564→   564→                    self._flush_amar()
   565→   565→                    self.section = "done"
   566→   566→
   567→   567→        self._flush_amar()
   568→   568→        return self.chunks

   581→   581→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   582→   582→                            nomor_perkara=self.nomor_perkara,
   583→   583→                            chunk_type="amar", chunk_num=self.current_amar)
   584→   584→
   585→   585→        self.chunks.append({
   586→   586→            "id": chunk_id,
   587→   587→            "text": text,

   589→   589→                "doc_type": self.doc_type,
   590→   590→                "issuer": self.issuer,
   591→   591→                "year": self.year,
   592→   592→                "nomor_perkara": self.nomor_perkara,

   594→   594→                "decision_keyword": kw,
   595→   595→                "amar": self.current_amar,

   597→   597→                "path": f"{kw} > Amar {self.current_amar}",
   598→   598→                "status": "active",
   599→   599→            }
   600→   600→        })
   601→   601→        self.current_amar_lines = []

   608→   608→def extract_nomor_perkara(lines):
   609→   609→    """Extract case number (e.g. 96/PUU-XVI/2018) from court ruling lines."""
   610→   610→    for line in lines[:30]:
   611→   611→        m = re.search(r'(\d+/[A-Z]+-[A-Z]+/\d{4})', line["text"])
   612→   612→        if m:
   613→   613→            return m.group(1)
   614→   614→    return None

   617→   617→def chunk_pdf(pdf_path: str, directory: str, family_map: Dict) -> List[Dict]:
   618→   618→    """
   619→   619→    Main entry point. Extract + parse a PDF into chunks.
   620→   620→    Returns list of chunk dicts with id, text, metadata.
   621→   621→    """
   622→   622→    # Get family config
   623→   623→    dir_config = family_map.get(directory, {})
   624→   624→    family = dir_config.get("family", "A")
   625→   625→    default_doc_type = dir_config.get("doc_type", "Unknown")
   626→   626→    default_issuer = dir_config.get("issuer", "Unknown")
   627→   627→
   628→   628→    # Extract clean text
   629→   629→    try:
   630→   630→        lines, meta = extract_clean(pdf_path)
   631→   631→    except Exception as e:
   632→   632→        return [{"id": f"ERROR:{directory}", "text": str(e), "metadata": {"error": True}}]
   633→   633→
   634→   634→    if not lines:
   635→   635→        return [{"id": f"EMPTY:{directory}", "text": "No extractable text (possibly scanned PDF)", "metadata": {"error": True, "needs_ocr": True}}]
   636→   636→
   637→   637→    # Detect doc type from title block (override directory default)
   638→   638→    title_block = extract_title_block(lines)
   639→   639→    detected_type, detected_issuer = detect_doc_type_from_title(title_block)
   640→   640→    doc_type = detected_type if detected_type != "Unknown" else default_doc_type
   641→   641→    issuer = detected_issuer if detected_issuer != "Unknown" else default_issuer
   642→   642→    nomor = meta.get("nomor")
   643→   643→    year = meta.get("year")
   644→   644→
   645→   645→    # Parse based on family
   646→   646→    if family == "A":
   647→   647→        parser = FamilyAParser(lines, doc_type, nomor, year, issuer)
   648→   648→        chunks = parser.parse()
   649→   649→    elif family == "B":
   650→   650→        parser = FamilyBParser(lines, doc_type, nomor, year, issuer)
   651→   651→        chunks = parser.parse()
   652→   652→    elif family == "C":
   653→   653→        nomor_perkara = extract_nomor_perkara(lines)
   654→   654→        parser = FamilyCParser(lines, doc_type, nomor, year, issuer, nomor_perkara)
   655→   655→        chunks = parser.parse()
   656→   656→    else:
   657→   657→        chunks = [{"id": f"UNKNOWN_FAMILY:{directory}", "text": "", "metadata": {"error": True}}]
   658→   658→
   659→   659→    # Add extraction metadata to all chunks
   660→   660→    for chunk in chunks:
   661→   661→        chunk["metadata"]["source_file"] = meta["file"]
   662→   662→        chunk["metadata"]["directory"] = directory
   663→   663→        chunk["metadata"]["family"] = family
   664→   664→        if nomor:
   665→   665→            chunk["metadata"]["nomor"] = nomor
   666→   666→        if year:
   667→   667→            chunk["metadata"]["year"] = year
   668→   668→
   669→   669→    return chunks
