    42→def build_id(doc_type, nomor, year, **parts):
    43→    """Build chunk ID. Handles both hierarchical and flat structures."""
    44→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    45→        nomor_perkara = parts.get("nomor_perkara", nomor or "?")
    46→        chunk_type = parts.get("chunk_type", "amar")
    47→        chunk_num = parts.get("chunk_num", "?")
    48→        return f"{doc_type}:{nomor_perkara}:{chunk_type}:{chunk_num}"
    49→    elif parts.get("chunk_type") in ("diktum", "pertimbangan"):
    50→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{parts['chunk_type']}:{parts['chunk_num']}"
    51→    else:
    52→        pasal = parts.get("pasal", "?")
    53→        ayat = parts.get("ayat", "?")
    54→        if ayat and ayat != "?":
    55→            return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}:{ayat}"
    56→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"

    59→def build_parent_id(doc_type, nomor, year, pasal):
    60→    """Build parent ID for ayat chunks."""
    61→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    62→        return None
    63→    return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"

    66→def build_path(headings):
    67→    """Build path string from heading stack. Filters out None and duplicates."""
    68→    seen = set()
    69→    parts = []
    70→    for h in headings:
    71→        if h and h not in seen:
    72→            seen.add(h)
    73→            parts.append(h)
    74→    return " > ".join(parts) if parts else ""

    77→def _make_bab_headings(bab_num, bab_label, bagian_num, bagian_label,
    78→                       paragraf_num, paragraf_label, pasal_str=None, ayat_str=None):
    79→    """Build heading list for path, avoiding duplicate BAB labels.
    80→    
    81→    Fix for bug #2: when BAB has no descriptive label (e.g., just "BAB I"),
    82→    the label defaults to "BAB I" which duplicates the number element.
    83→    Now: if label equals the number element, omit it.
    84→    """
    85→    headings = []
    86→    
    87→    bab_num_str = f"BAB {bab_num}" if bab_num else None
    88→    if bab_num_str:
    89→        headings.append(bab_num_str)
    90→    # Only add label if it's different from the number element and non-empty
    91→    if bab_label and bab_label != bab_num_str:
    92→        headings.append(bab_label)
    93→    
    94→    bagian_num_str = f"Bagian {bagian_num}" if bagian_num else None
    95→    if bagian_num_str:
    96→        headings.append(bagian_num_str)
    97→    if bagian_label and bagian_label != bagian_num_str:
    98→        headings.append(bagian_label)
    99→    
   100→    paragraf_num_str = f"Paragraf {paragraf_num}" if paragraf_num else None
   101→    if paragraf_num_str:
   102→        headings.append(paragraf_num_str)
   103→    if paragraf_label and paragraf_label != paragraf_num_str:
   104→        headings.append(paragraf_label)
   105→    
   106→    if pasal_str:
   107→        headings.append(pasal_str)
   108→    if ayat_str:
   109→        headings.append(ayat_str)
   110→    
   111→    return headings

   114→# ==========================================================================
   115→# FAMILY A: HIERARCHICAL STATUTE PARSER
   116→# ==========================================================================
   117→
   118→class FamilyAParser:
   119→    """
   120→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
   121→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
   122→    """
   123→
   124→    # Heading patterns (order matters: higher levels first)
   125→    # v6: allow '* ' prefix (UUD 1945 amendment markers)
   126→    RE_BAB = re.compile(r'^\*?\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
   127→    RE_BAGIAN = re.compile(r'^\*?\s*BAGIAN\s+(\w+)', re.IGNORECASE)
   128→    RE_PARAGRAF = re.compile(r'^\*?\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
   129→    # v6: capture alpha suffixes AND allow '* ' prefix (UUD 1945 amendment markers)
   130→    RE_PASAL = re.compile(r'^\*?\s*pasal\s+([IVXLCDM]+|\d+(?:\s+\d+)*[a-zA-Z]?)\s*[.}\s]*$', re.IGNORECASE)
   131→    # v6: Match pasal embedded mid-line (e.g., 'BAB VIIA ... * Pasal 22C')
   132→    RE_PASAL_EMBEDDED = re.compile(r'\*\s*Pasal\s+(\d+[a-zA-Z]*)', re.IGNORECASE)
   133→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
   134→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
   135→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
   136→
   137→    # Quoted amendment detection
   138→    # v7: allow optional comma between verb and "sehingga" (PKPU uses "diubah, sehingga")
   139→    RE_AMENDMENT_TRIGGER = re.compile(
   140→        r'(?:diubah|diganti|dihapus)[,\s]+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
   141→        re.IGNORECASE
   142→    )
   143→    RE_QUOTE_START = re.compile(r'^\s*["“]')
   144→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
   145→
   146→    # Penutup (closing section) marker
   147→    RE_PENUTUP = re.compile(r'^(?:KETENTUAN\s+PENUTUP|PENUTUP|penutup)\b')
   148→    # v3: Penjelasan (elucidation) marker — can appear after penutup or directly after last pasal
   149→    RE_PENJELASAN = re.compile(r'^(?:PENJELASAN|penjelasan)\b')
   150→
   151→    def __init__(self, lines, doc_type, nomor, year, issuer):
   152→        self.lines = lines
   153→        self.doc_type = doc_type
   154→        self.nomor = nomor
   155→        self.year = year
   156→        self.issuer = issuer
   157→        self.chunks = []
   158→
   159→        # Heading stack
   160→        self.current_bab = None
   161→        self.current_bab_label = None
   162→        self.current_bagian = None
   163→        self.current_bagian_label = None
   164→        self.current_paragraf = None
   165→        self.current_paragraf_label = None
   166→        self.current_pasal = None
   167→        self.current_ayat = None
   168→
   169→        # State
   170→        self.in_preamble = True
   171→        self.in_penutup = False
   172→        self.in_penjelasan = False  # v3: track elucidation section
   173→        self.in_quoted_amendment = False
   174→        self.quoted_amendment_source = None  # tracks which pasal is being amended
   175→        self.current_ayat_lines = []
   176→        self.current_pasal_lines = []
   177→        self.quoted_amendment_lines = []  # accumulate quoted text for chunking
   178→        self._penjelasan_lines = []  # v3: accumulate Penjelasan text
   179→        self._penjelasan_pasal = None
   180→
   181→    def parse(self) -> List[Dict]:
   182→        """Main parse loop. Returns list of chunks."""
   183→        for i, line in enumerate(self.lines):
   184→            text = line["text"].strip()
   185→            if not text:
   186→                continue
   187→
   188→            text_upper = text.upper()
   189→
   190→            # Detect preamble end / body start
   191→            if self.in_preamble:
   192→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   193→                    self.in_preamble = False
   194→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
   195→                    self.in_preamble = False
   196→                    continue
   197→                else:
   198→                    continue
   199→
   200→            # Detect penutup
   201→            if self.RE_PENUTUP.match(text):
   202→                self._flush_ayat()
   203→                self._flush_pasal()
   204→                self._flush_quoted_amendment()
   205→                self.in_penutup = True
   206→                continue
   207→
   208→            # v3: Detect PENJELASAN even without penutup (many docs go straight to penjelasan)
   209→            if self.RE_PENJELASAN.match(text) and not self.in_penjelasan:
   210→                self._flush_ayat()
   211→                self._flush_pasal()
   212→                self._flush_quoted_amendment()
   213→                self.in_penutup = True  # treat as post-operative
   214→                self.in_penjelasan = True
   215→                self._penjelasan_pasal = None
   216→                continue
   217→
   218→            if self.in_penutup:
   219→                # v3: Detect signature block and Penjelasan as hard boundaries
   220→                text_lower = text.lower()
   221→                # Signature block: lines like "Ditetapkan di Jakarta", "pada tanggal", or ttd/
   222→                # Also detect by very short lines after penutup start
   223→                if (re.search(r'ttd|disahkan|ditetapkan', text_lower) or
   224→                    (len(text) < 15 and text_upper == text and len(text) > 2)):
   225→                    self._flush_penjelasan()  # flush any accumulated penjelasan
   226→                    continue
   227→                # PENJELASAN section header
   228→                if text_upper == "PENJELASAN" or text_upper.startswith("PENJELASAN "):
   229→                    self._flush_penjelasan()
   230→                    self.in_penjelasan = True
   231→                    self._penjelasan_pasal = None
   232→                    continue
   233→                # Inside Penjelasan: track pasal references for IDs
   234→                if self.in_penjelasan:
   235→                    # v5: also capture alpha suffixes in Penjelasan pasal references
   236→                    m_pj = re.match(r'^\s*Pasal\s+(\d+[a-zA-Z]*)', text, re.IGNORECASE)
   237→                    if m_pj:
   238→                        self._flush_penjelasan()
   239→                        self._penjelasan_pasal = m_pj.group(1)
   240→                    else:
   241→                        self._penjelasan_lines.append(text)
   242→                continue
   243→
   244→            # Check for quoted amendment TRIGGER ("Pasal X diubah sehingga berbunyi:")
   245→            if not self.in_quoted_amendment:
   246→                if self.RE_AMENDMENT_TRIGGER.search(text):
   247→                    # Extract which pasal is being amended
   248→                    m_pasal = re.search(r'Pasal\s+(\d+|\w+)\s+(?:ayat\s*\(\d+\)\s+)?(?:diubah|diganti|dihapus)',
   249→                                         text, re.IGNORECASE)
   250→                    source_pasal = m_pasal.group(1) if m_pasal else "?"
   251→                    self.in_quoted_amendment = True
   252→                    self.quoted_amendment_source = source_pasal
   253→                    self._flush_ayat()
   254→                    # Store the trigger sentence as context for the quoted chunk
   255→                    self.quoted_amendment_lines = [text]
   256→                    continue
   257→
   258→            # Inside quoted amendment: accumulate text
   259→            if self.in_quoted_amendment:
   260→                self.quoted_amendment_lines.append(text)
   261→                if self.RE_QUOTE_END.search(text):
   262→                    self._flush_quoted_amendment()
   263→                # Also check for end of quoted section without closing quote:
   264→                # if we hit a new structural marker, the quote ended implicitly
   265→                if (self.RE_BAB.match(text) or self.RE_PASAL.match(text) or
   266→                    self.RE_PENUTUP.match(text)):
   267→                    self._flush_quoted_amendment()
   268→                    # Don't skip the structural marker - let it be processed below
   269→                    self.in_quoted_amendment = False
   270→                elif self.RE_QUOTE_END.search(text):
   271→                    pass  # already flushed above
   272→                else:
   273→                    continue
   274→
   275→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   276→            m = self.RE_BAB.match(text)
   277→            if m:
   278→                self._flush_ayat()
   279→                self._flush_pasal()
   280→                self.current_bab = m.group(1)
   281→                label_match = re.match(r'^\*?\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   282→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   283→                # Fix #2: if label is empty or just repeats "BAB X", set to None
   284→                if raw_label and raw_label.upper() != f"BAB {m.group(1).upper()}":
   285→                    self.current_bab_label = raw_label
   286→                else:
   287→                    self.current_bab_label = None
   288→                self.current_bagian = None
   289→                self.current_bagian_label = None
   290→                self.current_paragraf = None
   291→                self.current_paragraf_label = None
   292→                # v4: reset pasal when BAB changes to prevent heading-as-chunk
   293→                self.current_pasal = None
   294→                # v6: Check if a Pasal reference is glued to the BAB line
   295→                # (e.g., 'BAB VIIA DEWAN PERWAKILAN DAERAH * Pasal 22C')
   296→                m_glued = self.RE_PASAL_EMBEDDED.search(text)
   297→                if m_glued:
   298→                    self.current_pasal = m_glued.group(1)
   299→                    self.current_ayat = None
   300→                continue
   301→
   302→            m = self.RE_BAGIAN.match(text)
   303→            if m:
   304→                self._flush_ayat()
   305→                self._flush_pasal()
   306→                self.current_bagian = m.group(1)
   307→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   308→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   309→                if raw_label:
   310→                    self.current_bagian_label = raw_label
   311→                else:
   312→                    self.current_bagian_label = None
   313→                self.current_paragraf = None
   314→                self.current_paragraf_label = None
   315→                continue
   316→
   317→            m = self.RE_PARAGRAF.match(text)
   318→            if m:
   319→                self._flush_ayat()
   320→                self._flush_pasal()
   321→                self.current_paragraf = m.group(1)
   322→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   323→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   324→                if raw_label:
   325→                    self.current_paragraf_label = raw_label
   326→                else:
   327→                    self.current_paragraf_label = None
   328→                continue
   329→
   330→            m = self.RE_PASAL.match(text)
   331→            if m:
   332→                self._flush_ayat()
   333→                self._flush_pasal()
   334→                raw_num = m.group(1)
   335→                # v4: normalize space-separated digits ("1 1" -> "11", "1 10" -> "110")
   336→                self.current_pasal = re.sub(r'\s+', '', raw_num) if re.match(r'^\d', raw_num) else raw_num
   337→                self.current_ayat = None
   338→                # v4: capture any remaining text on the same line as pasal heading
   339→                remaining = text[m.end():].strip().lstrip('.').strip()
   340→                if remaining and len(remaining) > 5:
   341→                    self.current_pasal_lines.append(remaining)
   342→                continue
   343→
   344→            m = self.RE_AYAT.match(text)
   345→            if m and self.current_pasal:
   346→                self._flush_ayat()
   347→                self.current_ayat = m.group(1)
   348→                ayat_text = self.RE_AYAT.sub('', text).strip()
   349→                if ayat_text:
   350→                    self.current_ayat_lines.append(ayat_text)
   351→                continue
   352→
   353→            # v6: Check for embedded pasal in non-structural line
   354→            # (e.g., 'PEMILIHAN UMUM * Pasal 22E' — BAB label with glued pasal)
   355→            m_emb = self.RE_PASAL_EMBEDDED.search(text)
   356→            if m_emb and self.current_pasal:
   357→                self._flush_ayat()
   358→                self._flush_pasal()
   359→                self.current_pasal = m_emb.group(1)
   360→                self.current_ayat = None
   361→                continue
   362→
   363→            # If we're inside a pasal but no ayat detected, accumulate lines
   364→            if self.current_pasal:
   365→                if self.current_ayat:
   366→                    self.current_ayat_lines.append(text)
   367→                else:
   368→                    self.current_pasal_lines.append(text)
   369→
   370→        # Flush remaining
   371→        self._flush_ayat()
   372→        self._flush_pasal()
   373→        self._flush_quoted_amendment()
   374→
   375→        # v6: Deduplicate truncated page-break fragments
   376→        self._dedup_truncated_fragments()
   377→
   378→        # v7: Flag B↔8 OCR confusion in pasal numbers
   379→        self._flag_ocr_pasal_numbers()
   380→
   381→        return self.chunks
   382→
   383→    def _flush_ayat(self):
   384→        """Flush accumulated ayat lines as a chunk."""
   385→        if not self.current_ayat or not self.current_ayat_lines:
   386→            self.current_ayat_lines = []
   387→            return
   388→
   389→        text = ' '.join(self.current_ayat_lines).strip()
   390→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   391→            self.current_ayat_lines = []
   392→            return
   393→
   394→        headings = _make_bab_headings(
   395→            self.current_bab, self.current_bab_label,
   396→            self.current_bagian, self.current_bagian_label,
   397→            self.current_paragraf, self.current_paragraf_label,
   398→            pasal_str=f"Pasal {self.current_pasal}",
   399→            ayat_str=f"Ayat ({self.current_ayat})"
   400→        )
   401→
   402→        path = build_path(headings)
   403→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   404→                            pasal=self.current_pasal, ayat=self.current_ayat)
   405→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   406→
   407→        self.chunks.append({
   408→            "id": chunk_id,
   409→            "text": text,
   410→            "metadata": {
   411→                "doc_type": self.doc_type,
   412→                "issuer": self.issuer,
   413→                "year": self.year,
   414→                "bab": self.current_bab or None,
   415→                "pasal": self.current_pasal,
   416→                "ayat": self.current_ayat,
   417→                "parent": parent_id,
   418→                "path": path,
   419→                "status": "active",
   420→            }
   421→        })
   422→        self.current_ayat_lines = []
   423→
   424→    def _flush_pasal(self):
   425→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   426→        if not self.current_pasal or not self.current_pasal_lines:
   427→            self.current_pasal_lines = []
   428→            return
   429→
   430→        text = ' '.join(self.current_pasal_lines).strip()
   431→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   432→            self.current_pasal_lines = []
   433→            return
   434→
   435→        headings = _make_bab_headings(
   436→            self.current_bab, self.current_bab_label,
   437→            self.current_bagian, self.current_bagian_label,
   438→            self.current_paragraf, self.current_paragraf_label,
   439→            pasal_str=f"Pasal {self.current_pasal}",
   440→        )
   441→        path = build_path(headings)
   442→        chunk_id = build_id(self.doc_type, self.nomor, self.year, pasal=self.current_pasal)
   443→
   444→        self.chunks.append({
   445→            "id": chunk_id,
   446→            "text": text,
   447→            "metadata": {
   448→                "doc_type": self.doc_type,
   449→                "issuer": self.issuer,
   450→                "year": self.year,
   451→                "bab": self.current_bab or None,
   452→                "pasal": self.current_pasal,
   453→                "ayat": None,
   454→                "parent": None,
   455→                "path": path,
   456→                "status": "active",
   457→            }
   458→        })
   459→        self.current_pasal_lines = []
   460→
   461→    def _flush_quoted_amendment(self):
   462→        """Flush quoted amendment text as a chunk with status='quoted_amendment'.
   463→        
   464→        Fix for bug #6: instead of silently skipping quoted amendments,
   465→        we now chunk them with a clear status tag so retrieval can
   466→        distinguish live law from historical amendment pointers.
   467→        """
   468→        if not self.quoted_amendment_lines:
   469→            self.in_quoted_amendment = False
   470→            self.quoted_amendment_source = None
   471→            return
   472→
   473→        text = ' '.join(self.quoted_amendment_lines).strip()
   474→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   475→            self.quoted_amendment_lines = []
   476→            self.in_quoted_amendment = False
   477→            self.quoted_amendment_source = None
   478→            return
   479→
   480→        source = self.quoted_amendment_source or "?"
   481→        chunk_id = build_id(self.doc_type, self.nomor, self.year,

   540→    def _dedup_truncated_fragments(self):
   541→        """v6+v7: Remove page-break artifacts that share an ID with a real chunk.
   542→        
   543→        v6: When text spans a page break, the bottom-of-page fragment (10-50 chars)
   544→        gets the same pasal:ayat ID as the continuation on the next page.
   545→        Keep the longer chunk, drop the fragment (if it's a substring).
   546→        
   547→        v7: Also catch page-break label chunks (organizational unit names like
   548→        "Sekretariat Presiden", "Deputi Bidang ..." that appear at the top of
   549→        a new page and inherit the current pasal:ayat context). These are NOT
   550→        substrings of the real content, but they are very short (< 80 chars)
   551→        and collide with the real chunk's ID.
   552→        """
   553→        from collections import defaultdict
   554→        id_chunks = defaultdict(list)
   555→        for i, chunk in enumerate(self.chunks):
   556→            id_chunks[chunk["id"]].append(i)
   557→        
   558→        to_remove = set()
   559→        for chunk_id, indices in id_chunks.items():
   560→            if len(indices) < 2:
   561→                continue
   562→            pairs = [(i, len(self.chunks[i]["text"])) for i in indices]
   563→            pairs.sort(key=lambda x: -x[1])
   564→            longest_idx, longest_len = pairs[0]
   565→            
   566→            for idx, text_len in pairs[1:]:
   567→                shorter = self.chunks[idx]["text"].strip()
   568→                longer = self.chunks[longest_idx]["text"].strip()
   569→                # v6: substring-based dedup (truncated page-break tail)
   570→                if text_len < longest_len * 0.5:
   571→                    if longer.endswith(shorter) or shorter in longer:
   572→                        to_remove.add(idx)
   573→                # v7: short page-break label/artifact (< 80 chars)
   574→                # These are org unit names or OCR fragments at top of new page
   575→                elif text_len < 80 and text_len < longest_len * 0.4:
   576→                    to_remove.add(idx)
   577→        
   578→        if to_remove:
   579→            self.chunks = [c for i, c in enumerate(self.chunks) if i not in to_remove]
   580→
   581→    def _flag_ocr_pasal_numbers(self):
   582→        """v7: Detect and flag pasal numbers likely corrupted by B↔8 OCR confusion.
   583→        
   584→        Pattern: In amendment-heavy documents (like UU 41/2014), pasal numbers
   585→        with letter suffixes (36A, 36B, 36C...) are common. If OCR reads 'B' as '8',
   586→        '36B' becomes '368' — a plausible-looking but wrong pasal number.
   587→        
   588→        Detection heuristic: If a pasal number contains a trailing '8' where
   589→        neighbors have letter-suffixed variants (A, C, D...), flag it.
   590→        Also flag any pasal number with '8' replacing a letter position
   591→        (e.g., '918' where '91B' or '91C' would be expected).
   592→        
   593→        This adds metadata._ocr_flag to affected chunks so downstream QA can
   594→        identify and manually correct them.
   595→        """
   596→        # Collect all pasal numbers from chunks
   597→        pasal_order = []  # [(pasal_str, chunk_indices), ...]
   598→        pasal_map = {}  # pasal_str -> [chunk_indices]
   599→        for i, chunk in enumerate(self.chunks):
   600→            p = chunk.get('metadata', {}).get('pasal')
   601→            if p and p != '?' and p not in pasal_map:
   602→                pasal_order.append(p)
   603→            if p and p != '?':
   604→                pasal_map.setdefault(p, []).append(i)
   605→
   606→        # For each pasal number ending in a digit that could be OCR-confused letter:
   607→        # Check if removing the last digit and the same prefix has alpha variants
   608→        OCR_DIGIT_TO_LETTER = {'8': 'B', '0': 'O', '1': 'I', '5': 'S', '6': 'G'}
   609→        flagged = set()
   610→        for p in pasal_order:
   611→            # Only check numeric pasal numbers
   612→            if not re.match(r'^\d+$', str(p)):
   613→                continue
   614→            p_str = str(p)
   615→            if len(p_str) < 2:
   616→                continue
   617→            last_char = p_str[-1]
   618→            if last_char not in OCR_DIGIT_TO_LETTER:
   619→                continue
   620→            prefix = p_str[:-1]
   621→            letter = OCR_DIGIT_TO_LETTER[last_char]
   622→            candidate = prefix + letter
   623→            
   624→            # Check if the alpha variant exists among known pasals
   625→            if candidate in pasal_map:
   626→                flagged.add(p)
   627→                # Also flag all chunks with this pasal number
   628→                for idx in pasal_map.get(p, []):
   629→                    self.chunks[idx]['metadata']['_ocr_flag'] = (
   630→                        f'Possible OCR: pasal "{p}" may be "{candidate}" '
   631→                        f'(last digit {last_char} could be letter {letter})'
   632→                    )
   633→
   634→        if flagged:
   635→            print(f"  [WARN] OCR pasal flag: {flagged} (trailing digit may be letter)")
   636→

   642→class FamilyBParser:

   644→    Parses decree/decision documents: Menimbang > Mengingat > MEMUTUSKAN/MENETAPKAN > Diktum.
   645→    Handles: Keppres, Inpres, Tap_MPR, Kepmen.

   651→    RE_MENIMBANG = re.compile(r'^\s*menimbang\s*:?\s*$', re.IGNORECASE)

   661→        r'SEBELAS|DUA BELAS|TIGA BELAS|EMPAT BELAS|LIMA BELAS|ENAM BELAS|'
   662→        r'TUJUH BELAS|DELAPAN BELAS|SEMBILAN BELAS|DUA PULUH))\b',
   663→        re.IGNORECASE
   664→    )

   666→    # v5: also handle alpha suffixed pasal numbers (e.g. Pasal 68A)
   667→    RE_PASAL_DIKTUM = re.compile(r'^\s*Pasal\s+(\d+[a-zA-Z]?)\s*\.?\s*$', re.IGNORECASE)

   669→    def __init__(self, lines, doc_type, nomor, year, issuer):
   670→        self.lines = lines
   671→        self.doc_type = doc_type
   672→        self.nomor = nomor
   673→        self.year = year
   674→        self.issuer = issuer
   675→        self.chunks = []
   676→
   677→        self.section = "pre"  # pre, menimbang, mengingat, keputusan, done
   678→        self.current_item = None
   679→        self.current_item_lines = []
   680→        self.preamble_lines = []
   681→        self.decision_keyword = None
   682→        self.parent_section = None  # v3: track parent lettered section for inpres diktum IDs

   684→    def parse(self) -> List[Dict]:
   685→        for line in self.lines:
   686→            text = line["text"].strip()
   687→            if not text:
   688→                continue
   689→
   690→            text_upper = text.upper()
   691→
   692→            # Section transitions
   693→            if self.RE_MENIMBANG.match(text):
   694→                self._flush_preamble()
   695→                self.section = "menimbang"
   696→                self.current_item = None
   697→                continue
   698→
   699→            if self.RE_MENGINGAT.match(text):
   700→                self._flush_preamble()
   701→                self.section = "mengingat"
   702→                self.current_item = None
   703→                continue
   704→
   705→            # Decision keyword triggers (use search() for end-anchored patterns)
   706→            if self.section not in ("keputusan", "done"):
   707→                if self.RE_MENINSTRUKSIKAN.search(text):
   708→                    self._flush_item()
   709→                    self.section = "keputusan"
   710→                    self.decision_keyword = "MENGINSTRUKSIKAN"
   711→                    self.current_item = None
   712→                    continue
   713→
   714→                if self.RE_MEMUTUSKAN.search(text):
   715→                    self._flush_item()
   716→                    self.section = "keputusan"
   717→                    self.decision_keyword = "MEMUTUSKAN"
   718→                    self.current_item = None
   719→                    continue
   720→
   721→                if self.RE_MENETAPKAN.search(text):
   722→                    self._flush_item()
   723→                    self.section = "keputusan"
   724→                    self.decision_keyword = "MENETAPKAN"
   725→                    self.current_item = None
   726→                    continue
   727→
   728→            if self.section == "pre":
   729→                continue
   730→
   731→            if self.section == "done":
   732→                continue
   733→
   734→            # v3: Signature block detection (broader)
   735→            text_lower = text.lower()
   736→            if ("ttd" in text_lower and len(text) < 30):
   737→                self._flush_item()
   738→                self.section = "done"
   739→                continue
   740→            # Also stop at formal signing language
   741→            if re.search(r'ditetapkan\s+di|dilantik\s+di', text_lower):
   742→                self._flush_item()
   743→                self.section = "done"
   744→                continue
   745→            # Stop at Penjelasan in Family B documents
   746→            if text_upper == "PENJELASAN":
   747→                self._flush_item()
   748→                self.section = "done"
   749→                continue
   750→
   751→            # In keputusan section: check for item markers
   752→            if self.section == "keputusan":
   753→                m_letter = self.RE_ITEM_LETTER.match(text)
   754→                m_number = self.RE_ITEM_NUMBER.match(text)
   755→                m_ordinal = self.RE_ORDINAL.match(text)
   756→                m_pasal = self.RE_PASAL_DIKTUM.match(text)
   757→
   758→                if m_pasal:
   759→                    # Fix #5: "Pasal X" acts as a diktum divider in some keppres
   760→                    self._flush_item()
   761→                    self.current_item = f"Pasal {m_pasal.group(1)}"
   762→                    continue
   763→
   764→                if m_letter or m_number or m_ordinal:
   765→                    self._flush_item()
   766→                    if m_ordinal:
   767→                        self.current_item = m_ordinal.group(1).upper()
   768→                    elif m_letter:
   769→                        self.current_item = m_letter.group(1)
   770→                    else:
   771→                        self.current_item = m_number.group(1)
   772→                    m = m_letter or m_number or m_ordinal
   773→                    remaining = text[m.end():].strip()
   774→                    if remaining:
   775→                        self.current_item_lines.append(remaining)
   776→                    continue
   777→
   778→                # Sub-items: accumulate under current item
   779→                if self.current_item:
   780→                    self.current_item_lines.append(text)
   781→                continue
   782→
   783→        self._flush_item()
   784→        return self.chunks

   786→    def _flush_item(self):
   787→        if not self.current_item or not self.current_item_lines:
   788→            self.current_item_lines = []
   789→            return
   790→
   791→        text = ' '.join(self.current_item_lines).strip()
   792→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   793→            self.current_item_lines = []
   794→            return
   795→
   796→        kw = self.decision_keyword or "MEMUTUSKAN"
   797→        # v3: Include parent section in diktum ID for inpres (prevents numbering collision)
   798→        diktum_num = self.current_item
   799→        if self.parent_section and diktum != self.parent_section:
   800→            diktum_num = f"{self.parent_section}.{diktum_num}"
   801→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   802→                            chunk_type="diktum", chunk_num=diktum_num)
   803→
   804→        self.chunks.append({
   805→            "id": chunk_id,
   806→            "text": text,
   807→            "metadata": {
   808→                "doc_type": self.doc_type,
   809→                "issuer": self.issuer,
   810→                "year": self.year,
   811→                "nomor": self.nomor,
   812→                "section": "keputusan",
   813→                "decision_keyword": kw,
   814→                "diktum": self.current_item,
   815→                "parent": None,
   816→                "path": f"{kw} > Diktum {self.current_item}",
   817→                "status": "active",
   818→            }
   819→        })
   820→        self.current_item_lines = []
   821→
   822→    def _flush_preamble(self):
   823→        self.preamble_lines = []

   948→def chunk_pdf(pdf_path: str, directory: str, family_map: Dict) -> List[Dict]:
   949→    """
   950→    Main entry point. Extract + parse a PDF into chunks.
   951→    Returns list of chunk dicts with id, text, metadata.
   952→    """
   953→    dir_config = family_map.get(directory, {})
   954→    family = dir_config.get("family", "A")
   955→    default_doc_type = dir_config.get("doc_type", "Unknown")
   956→    default_issuer = dir_config.get("issuer", "Unknown")
   957→
   958→    try:
   959→        lines, meta = extract_clean(pdf_path)
   960→    except Exception as e:
   961→        return [{"id": f"ERROR:{directory}", "text": str(e), "metadata": {"error": True}}]
   962→
   963→    if not lines:
   964→        return [{"id": f"EMPTY:{directory}", "text": "No extractable text (possibly scanned PDF)", "metadata": {"error": True, "needs_ocr": True}}]
   965→
   966→    # v4 fix: Detect doc_type from title block FIRST, directory as fallback.
   967→    # Use page-1-only RAW lines (pre-boilerplate) since boilerplate stripping
   968→    # removes the actual document title (e.g. "PERATURAN PRESIDEN").
   969→    # Only page 1 to avoid body citations polluting the detection.
   970→    page1_raw = [l["text"].strip() for l in lines if l.get("page") == 1]
   971→    raw_title_block = " ".join(page1_raw[:15]) if page1_raw else ""
   972→    detected_type, detected_issuer = detect_doc_type_from_title(raw_title_block)
   973→    
   974→    # If page 1 was empty (image) or detection failed, try cleaned lines
   975→    if detected_type == "Unknown":
   976→        clean_title = extract_title_block(lines, max_first_n=5)
   977→        detected_type, detected_issuer = detect_doc_type_from_title(clean_title)
   978→    
   979→    # Title block wins when it returns a real type; directory is fallback only
   980→    if detected_type != "Unknown":
   981→        doc_type = detected_type
   982→        issuer = detected_issuer
   983→        if default_doc_type != "Unknown" and detected_type != default_doc_type:
   984→            print(f"  [WARN] doc_type: title={detected_type} overrides dir={default_doc_type} for {directory}")
   985→    elif default_doc_type != "Unknown":
   986→        doc_type = default_doc_type
   987→        issuer = default_issuer
   988→    else:
   989→        doc_type = "Unknown"
   990→        issuer = "Unknown"
   991→    nomor = meta.get("nomor")
