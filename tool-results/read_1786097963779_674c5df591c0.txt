    42→def build_id(doc_type, nomor, year, **parts):
    43→    """Build chunk ID. Handles both hierarchical and flat structures."""
    44→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    45→        nomor_perkara = parts.get("nomor_perkara", nomor or "?")
    46→        chunk_type = parts.get("chunk_type", "amar")
    47→        chunk_num = parts.get("chunk_num", "?")
    48→        return f"{doc_type}:{nomor_perkara}:{chunk_type}:{chunk_num}"
    49→    elif parts.get("chunk_type") in ("diktum", "pertimbangan"):
    50→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{parts['chunk_type']}:{parts['chunk_num']}"
    51→    else:
    52→        pasal = parts.get("pasal", "?")
    53→        ayat = parts.get("ayat", "?")
    54→        if ayat and ayat != "?":
    55→            return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}:{ayat}"
    56→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"

    59→def build_parent_id(doc_type, nomor, year, pasal):
    60→    """Build parent ID for ayat chunks."""
    61→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    62→        return None
    63→    return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"

    66→def build_path(headings):
    67→    """Build path string from heading stack. Filters out None and duplicates."""
    68→    seen = set()
    69→    parts = []
    70→    for h in headings:
    71→        if h and h not in seen:
    72→            seen.add(h)
    73→            parts.append(h)
    74→    return " > ".join(parts) if parts else ""

    77→def _make_bab_headings(bab_num, bab_label, bagian_num, bagian_label,
    78→                       paragraf_num, paragraf_label, pasal_str=None, ayat_str=None):
    79→    """Build heading list for path, avoiding duplicate BAB labels.
    80→    
    81→    Fix for bug #2: when BAB has no descriptive label (e.g., just "BAB I"),
    82→    the label defaults to "BAB I" which duplicates the number element.
    83→    Now: if label equals the number element, omit it.
    84→    """
    85→    headings = []
    86→    
    87→    bab_num_str = f"BAB {bab_num}" if bab_num else None
    88→    if bab_num_str:
    89→        headings.append(bab_num_str)
    90→    # Only add label if it's different from the number element and non-empty
    91→    if bab_label and bab_label != bab_num_str:
    92→        headings.append(bab_label)
    93→    
    94→    bagian_num_str = f"Bagian {bagian_num}" if bagian_num else None
    95→    if bagian_num_str:
    96→        headings.append(bagian_num_str)
    97→    if bagian_label and bagian_label != bagian_num_str:
    98→        headings.append(bagian_label)
    99→    
   100→    paragraf_num_str = f"Paragraf {paragraf_num}" if paragraf_num else None
   101→    if paragraf_num_str:
   102→        headings.append(paragraf_num_str)
   103→    if paragraf_label and paragraf_label != paragraf_num_str:
   104→        headings.append(paragraf_label)
   105→    
   106→    if pasal_str:
   107→        headings.append(pasal_str)
   108→    if ayat_str:
   109→        headings.append(ayat_str)
   110→    
   111→    return headings

   114→# ==========================================================================
   115→# FAMILY A: HIERARCHICAL STATUTE PARSER
   116→# ==========================================================================
   117→
   118→class FamilyAParser:
   119→    """
   120→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
   121→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
   122→    """
   123→
   124→    # Heading patterns (order matters: higher levels first)
   125→    RE_BAB = re.compile(r'^\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
   126→    RE_BAGIAN = re.compile(r'^\s*BAGIAN\s+(\w+)', re.IGNORECASE)
   127→    RE_PARAGRAF = re.compile(r'^\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
   128→    # v5: capture alpha suffixes (68A, 68B, ...) AND require heading-like ending
   129→    # to prevent body-text references like "Pasal 2 diangkat" from matching.
   130→    RE_PASAL = re.compile(r'^\s*pasal\s+([IVXLCDM]+|\d+(?:\s+\d+)*[a-zA-Z]?)\s*[.}\s]*$', re.IGNORECASE)
   131→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
   132→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
   133→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
   134→
   135→    # Quoted amendment detection
   136→    RE_AMENDMENT_TRIGGER = re.compile(
   137→        r'(?:diubah|diganti|dihapus)\s+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
   138→        re.IGNORECASE
   139→    )
   140→    RE_QUOTE_START = re.compile(r'^\s*["“]')
   141→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
   142→
   143→    # Penutup (closing section) marker
   144→    RE_PENUTUP = re.compile(r'^(?:KETENTUAN\s+PENUTUP|PENUTUP|penutup)\b')
   145→    # v3: Penjelasan (elucidation) marker — can appear after penutup or directly after last pasal
   146→    RE_PENJELASAN = re.compile(r'^(?:PENJELASAN|penjelasan)\b')
   147→
   148→    def __init__(self, lines, doc_type, nomor, year, issuer):
   149→        self.lines = lines
   150→        self.doc_type = doc_type
   151→        self.nomor = nomor
   152→        self.year = year
   153→        self.issuer = issuer
   154→        self.chunks = []
   155→
   156→        # Heading stack
   157→        self.current_bab = None
   158→        self.current_bab_label = None
   159→        self.current_bagian = None
   160→        self.current_bagian_label = None
   161→        self.current_paragraf = None
   162→        self.current_paragraf_label = None
   163→        self.current_pasal = None
   164→        self.current_ayat = None
   165→
   166→        # State
   167→        self.in_preamble = True
   168→        self.in_penutup = False
   169→        self.in_penjelasan = False  # v3: track elucidation section
   170→        self.in_quoted_amendment = False
   171→        self.quoted_amendment_source = None  # tracks which pasal is being amended
   172→        self.current_ayat_lines = []
   173→        self.current_pasal_lines = []
   174→        self.quoted_amendment_lines = []  # accumulate quoted text for chunking
   175→        self._penjelasan_lines = []  # v3: accumulate Penjelasan text
   176→        self._penjelasan_pasal = None
   177→
   178→    def parse(self) -> List[Dict]:
   179→        """Main parse loop. Returns list of chunks."""
   180→        for i, line in enumerate(self.lines):
   181→            text = line["text"].strip()
   182→            if not text:
   183→                continue
   184→
   185→            text_upper = text.upper()
   186→
   187→            # Detect preamble end / body start
   188→            if self.in_preamble:
   189→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   190→                    self.in_preamble = False
   191→                elif text_upper.startswith("MENETAPKAN") or text_upper.startswith("MEMUTUSKAN"):
   192→                    self.in_preamble = False
   193→                    continue
   194→                else:
   195→                    continue
   196→
   197→            # Detect penutup
   198→            if self.RE_PENUTUP.match(text):
   199→                self._flush_ayat()
   200→                self._flush_pasal()
   201→                self._flush_quoted_amendment()
   202→                self.in_penutup = True
   203→                continue
   204→
   205→            # v3: Detect PENJELASAN even without penutup (many docs go straight to penjelasan)
   206→            if self.RE_PENJELASAN.match(text) and not self.in_penjelasan:
   207→                self._flush_ayat()
   208→                self._flush_pasal()
   209→                self._flush_quoted_amendment()
   210→                self.in_penutup = True  # treat as post-operative
   211→                self.in_penjelasan = True
   212→                self._penjelasan_pasal = None
   213→                continue
   214→
   215→            if self.in_penutup:
   216→                # v3: Detect signature block and Penjelasan as hard boundaries
   217→                text_lower = text.lower()
   218→                # Signature block: lines like "Ditetapkan di Jakarta", "pada tanggal", or ttd/
   219→                # Also detect by very short lines after penutup start
   220→                if (re.search(r'ttd|disahkan|ditetapkan', text_lower) or
   221→                    (len(text) < 15 and text_upper == text and len(text) > 2)):
   222→                    self._flush_penjelasan()  # flush any accumulated penjelasan
   223→                    continue
   224→                # PENJELASAN section header
   225→                if text_upper == "PENJELASAN" or text_upper.startswith("PENJELASAN "):
   226→                    self._flush_penjelasan()
   227→                    self.in_penjelasan = True
   228→                    self._penjelasan_pasal = None
   229→                    continue
   230→                # Inside Penjelasan: track pasal references for IDs
   231→                if self.in_penjelasan:
   232→                    # v5: also capture alpha suffixes in Penjelasan pasal references
   233→                    m_pj = re.match(r'^\s*Pasal\s+(\d+[a-zA-Z]*)', text, re.IGNORECASE)
   234→                    if m_pj:
   235→                        self._flush_penjelasan()
   236→                        self._penjelasan_pasal = m_pj.group(1)
   237→                    else:
   238→                        self._penjelasan_lines.append(text)
   239→                continue
   240→
   241→            # Check for quoted amendment TRIGGER ("Pasal X diubah sehingga berbunyi:")
   242→            if not self.in_quoted_amendment:
   243→                if self.RE_AMENDMENT_TRIGGER.search(text):
   244→                    # Extract which pasal is being amended
   245→                    m_pasal = re.search(r'Pasal\s+(\d+|\w+)\s+(?:ayat\s*\(\d+\)\s+)?(?:diubah|diganti|dihapus)',
   246→                                         text, re.IGNORECASE)
   247→                    source_pasal = m_pasal.group(1) if m_pasal else "?"
   248→                    self.in_quoted_amendment = True
   249→                    self.quoted_amendment_source = source_pasal
   250→                    self._flush_ayat()
   251→                    # Store the trigger sentence as context for the quoted chunk
   252→                    self.quoted_amendment_lines = [text]
   253→                    continue
   254→
   255→            # Inside quoted amendment: accumulate text
   256→            if self.in_quoted_amendment:
   257→                self.quoted_amendment_lines.append(text)
   258→                if self.RE_QUOTE_END.search(text):
   259→                    self._flush_quoted_amendment()
   260→                # Also check for end of quoted section without closing quote:
   261→                # if we hit a new structural marker, the quote ended implicitly
   262→                if (self.RE_BAB.match(text) or self.RE_PASAL.match(text) or
   263→                    self.RE_PENUTUP.match(text)):
   264→                    self._flush_quoted_amendment()
   265→                    # Don't skip the structural marker - let it be processed below
   266→                    self.in_quoted_amendment = False
   267→                elif self.RE_QUOTE_END.search(text):
   268→                    pass  # already flushed above
   269→                else:
   270→                    continue
   271→
   272→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   273→            m = self.RE_BAB.match(text)
   274→            if m:
   275→                self._flush_ayat()
   276→                self._flush_pasal()
   277→                self.current_bab = m.group(1)
   278→                label_match = re.match(r'^\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   279→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   280→                # Fix #2: if label is empty or just repeats "BAB X", set to None
   281→                if raw_label and raw_label.upper() != f"BAB {m.group(1).upper()}":
   282→                    self.current_bab_label = raw_label
   283→                else:
   284→                    self.current_bab_label = None
   285→                self.current_bagian = None
   286→                self.current_bagian_label = None
   287→                self.current_paragraf = None
   288→                self.current_paragraf_label = None
   289→                # v4: reset pasal when BAB changes to prevent heading-as-chunk
   290→                self.current_pasal = None
   291→                continue
   292→
   293→            m = self.RE_BAGIAN.match(text)
   294→            if m:
   295→                self._flush_ayat()
   296→                self._flush_pasal()
   297→                self.current_bagian = m.group(1)
   298→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   299→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   300→                if raw_label:
   301→                    self.current_bagian_label = raw_label
   302→                else:
   303→                    self.current_bagian_label = None
   304→                self.current_paragraf = None
   305→                self.current_paragraf_label = None
   306→                continue
   307→
   308→            m = self.RE_PARAGRAF.match(text)
   309→            if m:
   310→                self._flush_ayat()
   311→                self._flush_pasal()
   312→                self.current_paragraf = m.group(1)
   313→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   314→                raw_label = label_match.group(1).strip().rstrip('.') if label_match else ""
   315→                if raw_label:
   316→                    self.current_paragraf_label = raw_label
   317→                else:
   318→                    self.current_paragraf_label = None
   319→                continue
   320→
   321→            m = self.RE_PASAL.match(text)
   322→            if m:
   323→                self._flush_ayat()
   324→                self._flush_pasal()
   325→                raw_num = m.group(1)
   326→                # v4: normalize space-separated digits ("1 1" -> "11", "1 10" -> "110")
   327→                self.current_pasal = re.sub(r'\s+', '', raw_num) if re.match(r'^\d', raw_num) else raw_num
   328→                self.current_ayat = None
   329→                # v4: capture any remaining text on the same line as pasal heading
   330→                remaining = text[m.end():].strip().lstrip('.').strip()
   331→                if remaining and len(remaining) > 5:
   332→                    self.current_pasal_lines.append(remaining)
   333→                continue
   334→
   335→            m = self.RE_AYAT.match(text)
   336→            if m and self.current_pasal:
   337→                self._flush_ayat()
   338→                self.current_ayat = m.group(1)
   339→                ayat_text = self.RE_AYAT.sub('', text).strip()
   340→                if ayat_text:
   341→                    self.current_ayat_lines.append(ayat_text)
   342→                continue
   343→
   344→            # If we're inside a pasal but no ayat detected, accumulate lines
   345→            if self.current_pasal:
   346→                if self.current_ayat:
   347→                    self.current_ayat_lines.append(text)
   348→                else:
   349→                    self.current_pasal_lines.append(text)
   350→
   351→        # Flush remaining
   352→        self._flush_ayat()
   353→        self._flush_pasal()
   354→        self._flush_quoted_amendment()
   355→
   356→        return self.chunks
   357→
   358→    def _flush_ayat(self):
   359→        """Flush accumulated ayat lines as a chunk."""
   360→        if not self.current_ayat or not self.current_ayat_lines:
   361→            self.current_ayat_lines = []
   362→            return
   363→
   364→        text = ' '.join(self.current_ayat_lines).strip()
   365→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   366→            self.current_ayat_lines = []
   367→            return
   368→
   369→        headings = _make_bab_headings(
   370→            self.current_bab, self.current_bab_label,
   371→            self.current_bagian, self.current_bagian_label,
   372→            self.current_paragraf, self.current_paragraf_label,
   373→            pasal_str=f"Pasal {self.current_pasal}",
   374→            ayat_str=f"Ayat ({self.current_ayat})"
   375→        )
   376→
   377→        path = build_path(headings)
   378→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   379→                            pasal=self.current_pasal, ayat=self.current_ayat)
   380→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   381→
   382→        self.chunks.append({
   383→            "id": chunk_id,
   384→            "text": text,
   385→            "metadata": {
   386→                "doc_type": self.doc_type,
   387→                "issuer": self.issuer,
   388→                "year": self.year,
   389→                "bab": self.current_bab or None,
   390→                "pasal": self.current_pasal,
   391→                "ayat": self.current_ayat,
   392→                "parent": parent_id,
   393→                "path": path,
   394→                "status": "active",
   395→            }
   396→        })
   397→        self.current_ayat_lines = []
   398→
   399→    def _flush_pasal(self):
   400→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   401→        if not self.current_pasal or not self.current_pasal_lines:
   402→            self.current_pasal_lines = []
   403→            return
   404→
   405→        text = ' '.join(self.current_pasal_lines).strip()
   406→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   407→            self.current_pasal_lines = []
   408→            return
   409→
   410→        headings = _make_bab_headings(
   411→            self.current_bab, self.current_bab_label,
   412→            self.current_bagian, self.current_bagian_label,
   413→            self.current_paragraf, self.current_paragraf_label,
   414→            pasal_str=f"Pasal {self.current_pasal}",
   415→        )
   416→        path = build_path(headings)
   417→        chunk_id = build_id(self.doc_type, self.nomor, self.year, pasal=self.current_pasal)
   418→
   419→        self.chunks.append({
   420→            "id": chunk_id,
   421→            "text": text,
   422→            "metadata": {
   423→                "doc_type": self.doc_type,
   424→                "issuer": self.issuer,
   425→                "year": self.year,
   426→                "bab": self.current_bab or None,
   427→                "pasal": self.current_pasal,
   428→                "ayat": None,
   429→                "parent": None,
   430→                "path": path,
   431→                "status": "active",
   432→            }
   433→        })
   434→        self.current_pasal_lines = []
   435→
   436→    def _flush_quoted_amendment(self):
   437→        """Flush quoted amendment text as a chunk with status='quoted_amendment'.
   438→        
   439→        Fix for bug #6: instead of silently skipping quoted amendments,
   440→        we now chunk them with a clear status tag so retrieval can
   441→        distinguish live law from historical amendment pointers.
   442→        """
   443→        if not self.quoted_amendment_lines:
   444→            self.in_quoted_amendment = False
   445→            self.quoted_amendment_source = None
   446→            return
   447→
   448→        text = ' '.join(self.quoted_amendment_lines).strip()
   449→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   450→            self.quoted_amendment_lines = []
   451→            self.in_quoted_amendment = False
   452→            self.quoted_amendment_source = None
   453→            return
   454→
   455→        source = self.quoted_amendment_source or "?"
   456→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   457→                            pasal=source, ayat="quoted")
   458→
   459→        self.chunks.append({
   460→            "id": chunk_id,
   461→            "text": text,
   462→            "metadata": {
   463→                "doc_type": self.doc_type,
   464→                "issuer": self.issuer,
   465→                "year": self.year,
   466→                "bab": self.current_bab or None,
   467→                "pasal": source,
   468→                "ayat": "quoted",
   469→                "parent": None,
   470→                "path": f"[AMANDEMEN Pasal {source}]",
   471→                "status": "quoted_amendment",
   472→                "amends_pasal": source,
   473→            }
   474→        })
   475→        self.quoted_amendment_lines = []
   476→        self.in_quoted_amendment = False
   477→        self.quoted_amendment_source = None
   478→
   479→    def _flush_penjelasan(self):
   480→        """Flush accumulated Penjelasan text as chunk(s) with distinct :penjelasan ID suffix.
   481→        
   482→        v3 fix: Penjelasan (elucidation) gets distinct ID suffix to prevent
   483→        collision with operative article IDs.
   484→        """
   485→        if not self._penjelasan_lines:
   486→            return
   487→
   488→        text = ' '.join(self._penjelasan_lines).strip()
   489→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   490→            self._penjelasan_lines = []
   491→            return
   492→
   493→        pasal_ref = self._penjelasan_pasal or "umum"

   517→# FAMILY B: DECREE/DECISION PARSER

   519→
   520→class FamilyBParser:
   521→    """
   522→    Parses decree/decision documents: Menimbang > Mengingat > MEMUTUSKAN/MENETAPKAN > Diktum.
   523→    Handles: Keppres, Inpres, Tap_MPR, Kepmen.
   524→    
   525→    v2 fix: Recognizes "Pasal X" as diktum divider (keppres pattern where
   526→    diktum items are organized under Pasal headings instead of numbered).
   527→    """
   528→
   529→    RE_MENIMBANG = re.compile(r'^\s*menimbang\s*:?\s*$', re.IGNORECASE)
   530→    RE_MENGINGAT = re.compile(r'^\s*mengingat\s*:?\s*$', re.IGNORECASE)
   531→    # Decision keywords: use $ anchor to match at END of line, allowing any prefix
   532→    RE_MEMUTUSKAN = re.compile(r'memutuskan\s*:?\s*$', re.IGNORECASE)
   533→    RE_MENETAPKAN = re.compile(r'menetapkan\s*:?\s*$', re.IGNORECASE)
   534→    RE_MENINSTRUKSIKAN = re.compile(r'menginstruksikan\s*:?\s*$', re.IGNORECASE)
   535→    RE_ITEM_LETTER = re.compile(r'^\s*([a-z])\s*\.\s*')
   536→    RE_ITEM_NUMBER = re.compile(r'^\s*(\d+)\s*\.\s*')
   537→    RE_ORDINAL = re.compile(
   538→        r'^\s*(KE(?:SATU|DUA|TIGA|EMPAT|LIMA|NAM|TUJUH|DELAPAN|SEMBILAN|SEPULUH|'
   539→        r'SEBELAS|DUA BELAS|TIGA BELAS|EMPAT BELAS|LIMA BELAS|ENAM BELAS|'
   540→        r'TUJUH BELAS|DELAPAN BELAS|SEMBILAN BELAS|DUA PULUH))\b',
   541→        re.IGNORECASE
   542→    )
   543→    # Fix #5: recognize "Pasal X" as diktum divider in Family B
   544→    # v5: also handle alpha suffixed pasal numbers (e.g. Pasal 68A)
   545→    RE_PASAL_DIKTUM = re.compile(r'^\s*Pasal\s+(\d+[a-zA-Z]?)\s*\.?\s*$', re.IGNORECASE)
   546→
   547→    def __init__(self, lines, doc_type, nomor, year, issuer):
   548→        self.lines = lines
   549→        self.doc_type = doc_type
   550→        self.nomor = nomor
   551→        self.year = year
   552→        self.issuer = issuer
   553→        self.chunks = []
   554→
   555→        self.section = "pre"  # pre, menimbang, mengingat, keputusan, done
   556→        self.current_item = None
   557→        self.current_item_lines = []
   558→        self.preamble_lines = []
   559→        self.decision_keyword = None
   560→        self.parent_section = None  # v3: track parent lettered section for inpres diktum IDs
   561→
   562→    def parse(self) -> List[Dict]:
   563→        for line in self.lines:
   564→            text = line["text"].strip()
   565→            if not text:
   566→                continue
   567→
   568→            text_upper = text.upper()
   569→
   570→            # Section transitions
   571→            if self.RE_MENIMBANG.match(text):
   572→                self._flush_preamble()
   573→                self.section = "menimbang"
   574→                self.current_item = None
   575→                continue
   576→
   577→            if self.RE_MENGINGAT.match(text):
   578→                self._flush_preamble()
   579→                self.section = "mengingat"
   580→                self.current_item = None
   581→                continue
   582→
   583→            # Decision keyword triggers (use search() for end-anchored patterns)
   584→            if self.section not in ("keputusan", "done"):
   585→                if self.RE_MENINSTRUKSIKAN.search(text):
   586→                    self._flush_item()
   587→                    self.section = "keputusan"
   588→                    self.decision_keyword = "MENGINSTRUKSIKAN"
   589→                    self.current_item = None
   590→                    continue
   591→
   592→                if self.RE_MEMUTUSKAN.search(text):
   593→                    self._flush_item()
   594→                    self.section = "keputusan"
   595→                    self.decision_keyword = "MEMUTUSKAN"
   596→                    self.current_item = None
   597→                    continue
   598→
   599→                if self.RE_MENETAPKAN.search(text):
   600→                    self._flush_item()
   601→                    self.section = "keputusan"
   602→                    self.decision_keyword = "MENETAPKAN"
   603→                    self.current_item = None
   604→                    continue
   605→
   606→            if self.section == "pre":
   607→                continue
   608→
   609→            if self.section == "done":
   610→                continue
   611→
   612→            # v3: Signature block detection (broader)
   613→            text_lower = text.lower()
   614→            if ("ttd" in text_lower and len(text) < 30):
   615→                self._flush_item()
   616→                self.section = "done"
   617→                continue
   618→            # Also stop at formal signing language
   619→            if re.search(r'ditetapkan\s+di|dilantik\s+di', text_lower):
   620→                self._flush_item()
   621→                self.section = "done"
   622→                continue
   623→            # Stop at Penjelasan in Family B documents
   624→            if text_upper == "PENJELASAN":
   625→                self._flush_item()
   626→                self.section = "done"
   627→                continue
   628→
   629→            # In keputusan section: check for item markers
   630→            if self.section == "keputusan":
   631→                m_letter = self.RE_ITEM_LETTER.match(text)
   632→                m_number = self.RE_ITEM_NUMBER.match(text)
   633→                m_ordinal = self.RE_ORDINAL.match(text)
   634→                m_pasal = self.RE_PASAL_DIKTUM.match(text)
   635→
   636→                if m_pasal:
   637→                    # Fix #5: "Pasal X" acts as a diktum divider in some keppres
   638→                    self._flush_item()
   639→                    self.current_item = f"Pasal {m_pasal.group(1)}"
   640→                    continue
   641→
   642→                if m_letter or m_number or m_ordinal:
   643→                    self._flush_item()
   644→                    if m_ordinal:
   645→                        self.current_item = m_ordinal.group(1).upper()
   646→                    elif m_letter:
   647→                        self.current_item = m_letter.group(1)
   648→                    else:
   649→                        self.current_item = m_number.group(1)
   650→                    m = m_letter or m_number or m_ordinal
   651→                    remaining = text[m.end():].strip()
   652→                    if remaining:
   653→                        self.current_item_lines.append(remaining)
   654→                    continue
   655→
   656→                # Sub-items: accumulate under current item
   657→                if self.current_item:
   658→                    self.current_item_lines.append(text)
   659→                continue
   660→
   661→        self._flush_item()
   662→        return self.chunks
   663→
   664→    def _flush_item(self):
   665→        if not self.current_item or not self.current_item_lines:
   666→            self.current_item_lines = []
   667→            return
   668→
   669→        text = ' '.join(self.current_item_lines).strip()
   670→        if not text or len(text) < MIN_CHUNK_TEXT_LENGTH:
   671→            self.current_item_lines = []
   672→            return
   673→
   674→        kw = self.decision_keyword or "MEMUTUSKAN"
   675→        # v3: Include parent section in diktum ID for inpres (prevents numbering collision)
   676→        diktum_num = self.current_item
   677→        if self.parent_section and diktum != self.parent_section:
   678→            diktum_num = f"{self.parent_section}.{diktum_num}"
   679→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   680→                            chunk_type="diktum", chunk_num=diktum_num)
   681→
   682→        self.chunks.append({
   683→            "id": chunk_id,
   684→            "text": text,
   685→            "metadata": {
   686→                "doc_type": self.doc_type,
   687→                "issuer": self.issuer,
   688→                "year": self.year,
   689→                "nomor": self.nomor,
   690→                "section": "keputusan",
   691→                "decision_keyword": kw,
   692→                "diktum": self.current_item,
   693→                "parent": None,
   694→                "path": f"{kw} > Diktum {self.current_item}",
   695→                "status": "active",
   696→            }
   697→        })
   698→        self.current_item_lines = []
   699→
   700→    def _flush_preamble(self):
   701→        self.preamble_lines = []

   714→    RE_MENGADILI = re.compile(r'^\s*mengadili\s*:?\s*$', re.IGNORECASE)
   715→    RE_MENETAPKAN = re.compile(r'^\s*menetapkan\s*:?\s*$', re.IGNORECASE)
   716→    RE_AMAR = re.compile(r'^\s*(\d+)\s*\.\s+')
   717→    RE_PUTUSAN = re.compile(r'^\s*putusan\s*:?\s*$', re.IGNORECASE)

   719→    def __init__(self, lines, doc_type, nomor, year, issuer, nomor_perkara=None):
   720→        self.lines = lines
   721→        self.doc_type = doc_type
   722→        self.nomor = nomor
   723→        self.year = year
   724→        self.issuer = issuer
   725→        self.nomor_perkara = nomor_perkara
   726→        self.chunks = []

   728→        self.section = "pre"
   729→        self.current_amar = None
   730→        self.current_amar_lines = []
   731→        self.pertimbangan_lines = []

   733→    def parse(self) -> List[Dict]:
   734→        for line in self.lines:
   735→            text = line["text"].strip()
   736→            if not text:
   737→                continue
   738→
   739→            text_upper = text.upper()
   740→
   741→            if self.RE_MENGADILI.match(text):
   742→                self._flush_amar()
   743→                self.section = "amar"
   744→                self.decision_keyword = "MENGADILI"
   745→                continue
   746→
   747→            if self.RE_MENETAPKAN.match(text):
   748→                self._flush_amar()
   749→                self.section = "amar"
   750→                self.decision_keyword = "MENETAPKAN"
   751→                continue
   752→
   753→            if self.section == "pre":
   754→                continue
   755→
   756→            if self.section == "done":
   757→                continue
   758→
   759→            if self.section == "amar":
   760→                m = self.RE_AMAR.match(text)
   761→                if m:
   762→                    self._flush_amar()
   763→                    self.current_amar = m.group(1)
   764→                    remaining = text[m.end():].strip()
   765→                    if remaining:
   766→                        self.current_amar_lines.append(remaining)
   767→                    continue
   768→
   769→                if self.current_amar:
   770→                    self.current_amar_lines.append(text)
   771→
   772→                if re.search(r'\bttd\b|\bHAKIM\b|\bKETUA\b', text_upper) and len(text) < 30:
   773→                    self._flush_amar()
   774→                    self.section = "done"

   776→        self._flush_amar()
   777→        return self.chunks

   789→        kw = getattr(self, 'decision_keyword', 'MENGADILI')
   790→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   791→                            nomor_perkara=self.nomor_perkara,
   792→                            chunk_type="amar", chunk_num=self.current_amar)
   793→
   794→        self.chunks.append({
   795→            "id": chunk_id,
   796→            "text": text,
   797→            "metadata": {
   798→                "doc_type": self.doc_type,
   799→                "issuer": self.issuer,
   800→                "year": self.year,
   801→                "nomor_perkara": self.nomor_perkara,
   802→                "section": "amar",
   803→                "decision_keyword": kw,
   804→                "amar": self.current_amar,
   805→                "parent": None,
   806→                "path": f"{kw} > Amar {self.current_amar}",
   807→                "status": "active",
   808→            }
   809→        })
   810→        self.current_amar_lines = []

   817→def extract_nomor_perkara(lines):
   818→    """Extract case number (e.g. 96/PUU-XVI/2018) from court ruling lines."""
   819→    for line in lines[:30]:
   820→        m = re.search(r'(\d+/[A-Z]+-[A-Z]+/\d{4})', line["text"])
   821→        if m:
   822→            return m.group(1)
   823→    return None

   826→def chunk_pdf(pdf_path: str, directory: str, family_map: Dict) -> List[Dict]:
   827→    """
   828→    Main entry point. Extract + parse a PDF into chunks.
   829→    Returns list of chunk dicts with id, text, metadata.
   830→    """
   831→    dir_config = family_map.get(directory, {})
   832→    family = dir_config.get("family", "A")
   833→    default_doc_type = dir_config.get("doc_type", "Unknown")
   834→    default_issuer = dir_config.get("issuer", "Unknown")
   835→
   836→    try:
   837→        lines, meta = extract_clean(pdf_path)
   838→    except Exception as e:
   839→        return [{"id": f"ERROR:{directory}", "text": str(e), "metadata": {"error": True}}]
   840→
   841→    if not lines:
   842→        return [{"id": f"EMPTY:{directory}", "text": "No extractable text (possibly scanned PDF)", "metadata": {"error": True, "needs_ocr": True}}]
   843→
   844→    # v4 fix: Detect doc_type from title block FIRST, directory as fallback.
   845→    # Use page-1-only RAW lines (pre-boilerplate) since boilerplate stripping
   846→    # removes the actual document title (e.g. "PERATURAN PRESIDEN").
   847→    # Only page 1 to avoid body citations polluting the detection.
   848→    page1_raw = [l["text"].strip() for l in lines if l.get("page") == 1]
   849→    raw_title_block = " ".join(page1_raw[:15]) if page1_raw else ""
   850→    detected_type, detected_issuer = detect_doc_type_from_title(raw_title_block)
   851→    
   852→    # If page 1 was empty (image) or detection failed, try cleaned lines
   853→    if detected_type == "Unknown":
   854→        clean_title = extract_title_block(lines, max_first_n=5)
   855→        detected_type, detected_issuer = detect_doc_type_from_title(clean_title)
   856→    
   857→    # Title block wins when it returns a real type; directory is fallback only
   858→    if detected_type != "Unknown":
   859→        doc_type = detected_type
   860→        issuer = detected_issuer
   861→        if default_doc_type != "Unknown" and detected_type != default_doc_type:
   862→            print(f"  [WARN] doc_type: title={detected_type} overrides dir={default_doc_type} for {directory}")
   863→    elif default_doc_type != "Unknown":
   864→        doc_type = default_doc_type
   865→        issuer = default_issuer
   866→    else:
   867→        doc_type = "Unknown"
   868→        issuer = "Unknown"
   869→    nomor = meta.get("nomor")
   870→    year = meta.get("year")
   871→
   872→    # v4: If page 1 was empty (image), always re-extract nomor/year from post-decision text
   873→    # since the pre-decision text contains citations, not the document's own number.
   874→    page1_has_text = any(l.get("page") == 1 for l in lines)
   875→    if not page1_has_text and meta.get("total_pages", 0) > 1:
   876→        post_nomor, post_year = extract_nomor_year_post_decision(lines)
   877→        if post_nomor and post_nomor != nomor:
   878→            nomor = post_nomor
   879→            if post_year:
   880→                year = post_year
   881→            print(f"  [INFO] Post-decision: nomor={nomor}, year={year}")
   882→        # v4: Filename fallback when page 1 is image and text extraction found wrong number
   883→        if (not post_nomor) or (nomor == meta.get('nomor') and re.search(r'(\d+)[_.](\d{4})', os.path.basename(pdf_path))):
   884→            fn = os.path.basename(pdf_path)
   885→            fn_match = re.search(r'(\d+)[_.](\d{4})', fn)
   886→            if fn_match:
   887→                nomor = fn_match.group(1)
   888→                year = fn_match.group(2)
   889→                print(f"  [INFO] Filename fallback: nomor={nomor}, year={year}")
   890→    
   891→    # Directory-specific overrides for image-cover documents where title can't be read
   892→    if directory == "JDIH_KPU" and doc_type in ("Peraturan", "Unknown"):
   893→        doc_type = "PKPU"
   894→        issuer = "Komisi Pemilihan Umum"
   895→
   896→    # Parse based on family
   897→    if family == "A":
   898→        parser = FamilyAParser(lines, doc_type, nomor, year, issuer)
   899→        chunks = parser.parse()
   900→    elif family == "B":
   901→        parser = FamilyBParser(lines, doc_type, nomor, year, issuer)
   902→        chunks = parser.parse()
   903→    elif family == "C":
   904→        nomor_perkara = extract_nomor_perkara(lines)
   905→        parser = FamilyCParser(lines, doc_type, nomor, year, issuer, nomor_perkara)
   906→        chunks = parser.parse()
   907→    else:
   908→        chunks = [{"id": f"UNKNOWN_FAMILY:{directory}", "text": "", "metadata": {"error": True}}]
   909→
   910→    # Add extraction metadata to all chunks
   911→    for chunk in chunks:
   912→        chunk["metadata"]["source_file"] = meta["file"]
   913→        chunk["metadata"]["directory"] = directory
   914→        chunk["metadata"]["family"] = family
   915→        if nomor:
   916→            chunk["metadata"]["nomor"] = nomor
   917→        if year:
   918→            chunk["metadata"]["year"] = year
   919→
   920→    return chunks
