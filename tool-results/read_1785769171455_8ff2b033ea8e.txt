     1→"""
     2→Family A/B/C parsers for Indonesian legal documents.
     3→
     4→Family A (Hierarchical Statute): BAB > Bagian > Paragraf > Pasal > Ayat > huruf/angka
     5→Family B (Decree/Decision): Menimbang > Mengingat > MEMUTUSKAN/MENETAPKAN > Diktum
     6→Family C (Court Ruling): Duduk Perkara > Menimbang > MENGADILI/MENETAPKAN > Amar
     7→
     8→All parsers return a list of chunks in the target JSON format.
     9→"""
    10→import re
    11→import json
    12→import sys
    13→import os
    14→from typing import List, Dict, Optional, Tuple
    15→
    16→# Add parent dir for imports
    17→sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
    18→from scripts.clean_extractor import extract_clean, extract_title_block, detect_doc_type_from_title, fix_glyph_corruption
    19→

    22→# SHARED UTILITIES

    24→
    25→ROMAN_MAP = {'I':1,'II':2,'III':3,'IV':4,'V':5,'VI':6,'VII':7,'VIII':8,'IX':9,'X':10,
    26→             'XI':11,'XII':12,'XIII':13,'XIV':14,'XV':15,'XVI':16,'XVII':17,'XVIII':18,
    27→             'XIX':19,'XX':20}
    28→
    29→def roman_to_int(s):
    30→    return ROMAN_MAP.get(s.upper(), None)
    31→
    32→
    33→def build_id(doc_type, nomor, year, **parts):
    34→    """Build chunk ID. Handles both hierarchical and flat structures."""
    35→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    36→        # Court rulings use nomor_perkara
    37→        nomor_perkara = parts.get("nomor_perkara", nomor or "?")
    38→        chunk_type = parts.get("chunk_type", "amar")
    39→        chunk_num = parts.get("chunk_num", "?")
    40→        return f"{doc_type}:{nomor_perkara}:{chunk_type}:{chunk_num}"
    41→    elif parts.get("chunk_type") in ("diktum", "pertimbangan"):
    42→        # Family B: flat numbered
    43→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{parts['chunk_type']}:{parts['chunk_num']}"
    44→    else:
    45→        # Family A: hierarchical
    46→        pasal = parts.get("pasal", "?")
    47→        ayat = parts.get("ayat", "?")
    48→        if ayat and ayat != "?":
    49→            return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}:{ayat}"
    50→        return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"
    51→
    52→
    53→def build_parent_id(doc_type, nomor, year, pasal):
    54→    """Build parent ID for ayat chunks."""
    55→    if doc_type in ("Putusan_MK", "Putusan_MA"):
    56→        return None
    57→    return f"{doc_type}:{nomor or '?'}:{year or '?'}:{pasal}"
    58→
    59→
    60→def build_path(headings):
    61→    """Build path string from heading stack."""
    62→    parts = [h for h in headings if h]
    63→    return " > ".join(parts) if parts else ""

    66→# ==========================================================================
    67→# FAMILY A: HIERARCHICAL STATUTE PARSER
    68→# ==========================================================================
    69→
    70→class FamilyAParser:
    71→    """
    72→    Parses hierarchical legal documents: BAB > Bagian > Paragraf > Pasal > Ayat.
    73→    Handles: UU, PP, PerPPU, Perpres, Perda, UUD1945, Permen, PKPU.
    74→    """
    75→
    76→    # Heading patterns (order matters: higher levels first)
    77→    RE_BAB = re.compile(r'^\s*BAB\s+([IVXLCDM]+)\b', re.IGNORECASE)
    78→    RE_BAGIAN = re.compile(r'^\s*BAGIAN\s+(\w+)', re.IGNORECASE)
    79→    RE_PARAGRAF = re.compile(r'^\s*PARAGRAF\s+(\w+)', re.IGNORECASE)
    80→    RE_PASAL = re.compile(r'^\s*pasal\s+(\d+[a-zA-Z]?)\s*\.?\s*.{0,5}$', re.IGNORECASE)
    81→    RE_AYAT = re.compile(r'^\s*\(\s*(\d+[a-zA-Z]?)\s*\)')
    82→    RE_HURUF = re.compile(r'^\s*([a-z])\s*\.\s')
    83→    RE_ANGKA = re.compile(r'^\s*(\d+)\s*\.\s')
    84→
    85→    # Quoted amendment detection
    86→    RE_AMENDMENT_TRIGGER = re.compile(
    87→        r'(?:diubah|diganti|dihapus)\s+sehingga\s+berbunyi\s+(?:sebagai\s+berikut|:\s*")',
    88→        re.IGNORECASE
    89→    )
    90→    RE_QUOTE_START = re.compile(r'^\s*["“]')
    91→    RE_QUOTE_END = re.compile(r'["”]\s*[.;]?\s*$')
    92→
    93→    # Penutup (closing section) marker
    94→    RE_PENUTUP = re.compile(r'^(?:PENUTUP|penutup)\b')
    95→
    96→    def __init__(self, lines, doc_type, nomor, year, issuer):
    97→        self.lines = lines
    98→        self.doc_type = doc_type
    99→        self.nomor = nomor
   100→        self.year = year
   101→        self.issuer = issuer
   102→        self.chunks = []
   103→
   104→        # Heading stack
   105→        self.current_bab = None
   106→        self.current_bab_label = None
   107→        self.current_bagian = None
   108→        self.current_bagian_label = None
   109→        self.current_paragraf = None
   110→        self.current_paragraf_label = None
   111→        self.current_pasal = None
   112→        self.current_ayat = None
   113→
   114→        # State
   115→        self.in_preamble = True
   116→        self.in_penutup = False
   117→        self.in_quoted_amendment = False
   118→        self.quoted_amendment_note = None
   119→        self.current_ayat_lines = []
   120→        self.current_pasal_lines = []
   121→
   122→    def parse(self) -> List[Dict]:
   123→        """Main parse loop. Returns list of chunks."""
   124→        for i, line in enumerate(self.lines):
   125→            text = line["text"].strip()
   126→            if not text:
   127→                continue
   128→
   129→            text_upper = text.upper()
   130→
   131→            # Detect preamble end / body start
   132→            if self.in_preamble:
   133→                if self.RE_BAB.match(text) or self.RE_PASAL.match(text):
   134→                    self.in_preamble = False
   135→                elif text_upper.startswith("MENETAPKAN"):
   136→                    self.in_preamble = False
   137→                    continue  # skip "Menetapkan" line itself
   138→                else:
   139→                    continue  # still in preamble
   140→
   141→            # Detect penutup
   142→            if self.RE_PENUTUP.match(text):
   143→                self.in_penutup = True
   144→                # Flush any pending ayat
   145→                self._flush_ayat()
   146→                self._flush_pasal()
   147→                continue
   148→
   149→            if self.in_penutup:
   150→                continue  # skip penutup
   151→
   152→            # Check for quoted amendment
   153→            if not self.in_quoted_amendment:
   154→                if self.RE_AMENDMENT_TRIGGER.search(text):
   155→                    self.in_quoted_amendment = True
   156→                    self.quoted_amendment_note = text[:100]
   157→                    self._flush_ayat()
   158→                    continue
   159→                if self.RE_QUOTE_START.match(text):
   160→                    # Could be inline quote
   161→                    self.in_quoted_amendment = True
   162→                    self.quoted_amendment_note = "Quoted passage (inline)"
   163→
   164→            if self.in_quoted_amendment:
   165→                if self.RE_QUOTE_END.search(text):
   166→                    self.in_quoted_amendment = False
   167→                    self.quoted_amendment_note = None
   168→                continue  # skip all quoted text
   169→
   170→            # Match structural elements (order: BAB > Bagian > Paragraf > Pasal > Ayat > huruf > angka)
   171→            m = self.RE_BAB.match(text)
   172→            if m:
   173→                self._flush_ayat()
   174→                self._flush_pasal()
   175→                self.current_bab = m.group(1)
   176→                # Get label text after BAB X
   177→                label_match = re.match(r'^\s*BAB\s+[IVXLCDM]+\s+(.*)', text, re.IGNORECASE)
   178→                self.current_bab_label = label_match.group(1).strip().rstrip('.') if label_match else f"BAB {m.group(1)}"
   179→                self.current_bagian = None
   180→                self.current_bagian_label = None
   181→                self.current_paragraf = None
   182→                self.current_paragraf_label = None
   183→                continue
   184→
   185→            m = self.RE_BAGIAN.match(text)
   186→            if m:
   187→                self._flush_ayat()
   188→                self._flush_pasal()
   189→                self.current_bagian = m.group(1)
   190→                label_match = re.match(r'^\s*BAGIAN\s+\w+\s+(.*)', text, re.IGNORECASE)
   191→                self.current_bagian_label = label_match.group(1).strip().rstrip('.') if label_match else f"Bagian {m.group(1)}"
   192→                self.current_paragraf = None
   193→                self.current_paragraf_label = None
   194→                continue
   195→
   196→            m = self.RE_PARAGRAF.match(text)
   197→            if m:
   198→                self._flush_ayat()
   199→                self._flush_pasal()
   200→                self.current_paragraf = m.group(1)
   201→                label_match = re.match(r'^\s*PARAGRAF\s+\w+\s+(.*)', text, re.IGNORECASE)
   202→                self.current_paragraf_label = label_match.group(1).strip().rstrip('.') if label_match else f"Paragraf {m.group(1)}"
   203→                continue
   204→
   205→            m = self.RE_PASAL.match(text)
   206→            if m:
   207→                self._flush_ayat()
   208→                self._flush_pasal()
   209→                self.current_pasal = m.group(1)
   210→                self.current_ayat = None
   211→                continue
   212→
   213→            m = self.RE_AYAT.match(text)
   214→            if m and self.current_pasal:
   215→                self._flush_ayat()
   216→                self.current_ayat = m.group(1)
   217→                # The rest of the line after (N) is the start of ayat text
   218→                ayat_text = self.RE_AYAT.sub('', text).strip()
   219→                if ayat_text:
   220→                    self.current_ayat_lines.append(ayat_text)
   221→                continue
   222→
   223→            # If we're inside a pasal but no ayat detected, accumulate lines
   224→            if self.current_pasal:
   225→                if self.current_ayat:
   226→                    self.current_ayat_lines.append(text)
   227→                else:
   228→                    self.current_pasal_lines.append(text)
   229→
   230→        # Flush remaining
   231→        self._flush_ayat()
   232→        self._flush_pasal()
   233→
   234→        return self.chunks
   235→
   236→    def _flush_ayat(self):
   237→        """Flush accumulated ayat lines as a chunk."""
   238→        if not self.current_ayat or not self.current_ayat_lines:
   239→            self.current_ayat_lines = []
   240→            return
   241→
   242→        text = ' '.join(self.current_ayat_lines).strip()
   243→        if not text:
   244→            self.current_ayat_lines = []
   245→            return
   246→
   247→        headings = [
   248→            f"BAB {self.current_bab}" if self.current_bab else None,
   249→            self.current_bab_label,
   250→            f"Bagian {self.current_bagian}" if self.current_bagian else None,
   251→            self.current_bagian_label,
   252→            f"Paragraf {self.current_paragraf}" if self.current_paragraf else None,
   253→            self.current_paragraf_label,
   254→            f"Pasal {self.current_pasal}" if self.current_pasal else None,
   255→            f"Ayat ({self.current_ayat})",
   256→        ]
   257→        clean_headings = [h for h in headings if h]
   258→
   259→        path = build_path(clean_headings)
   260→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   261→                            pasal=self.current_pasal, ayat=self.current_ayat)
   262→        parent_id = build_parent_id(self.doc_type, self.nomor, self.year, self.current_pasal)
   263→
   264→        self.chunks.append({
   265→            "id": chunk_id,
   266→            "text": text,
   267→            "metadata": {
   268→                "doc_type": self.doc_type,
   269→                "issuer": self.issuer,
   270→                "year": self.year,
   271→                "bab": self.current_bab or None,
   272→                "pasal": self.current_pasal,
   273→                "ayat": self.current_ayat,
   274→                "parent": parent_id,
   275→                "path": path,
   276→                "status": "active",
   277→            }
   278→        })
   279→        self.current_ayat_lines = []
   280→
   281→    def _flush_pasal(self):
   282→        """Flush pasal-only content (no ayat detected) as a single chunk."""
   283→        if not self.current_pasal or not self.current_pasal_lines:
   284→            self.current_pasal_lines = []
   285→            return
   286→
   287→        text = ' '.join(self.current_pasal_lines).strip()
   288→        if not text:
   289→            self.current_pasal_lines = []
   290→            return
   291→
   292→        headings = [
   293→            f"BAB {self.current_bab}" if self.current_bab else None,
   294→            self.current_bab_label,
   295→            f"Bagian {self.current_bagian}" if self.current_bagian else None,
   296→            self.current_bagian_label,
   297→            f"Paragraf {self.current_paragraf}" if self.current_paragraf else None,
   298→            self.current_paragraf_label,
   299→            f"Pasal {self.current_pasal}",
   300→        ]
   301→        clean_headings = [h for h in headings if h]
   302→        path = build_path(clean_headings)
   303→        chunk_id = build_id(self.doc_type, self.nomor, self.year, pasal=self.current_pasal)
   304→
   305→        self.chunks.append({
   306→            "id": chunk_id,
   307→            "text": text,
   308→            "metadata": {
   309→                "doc_type": self.doc_type,
   310→                "issuer": self.issuer,
   311→                "year": self.year,
   312→                "bab": self.current_bab or None,
   313→                "pasal": self.current_pasal,
   314→                "ayat": None,
   315→                "parent": None,
   316→                "path": path,
   317→                "status": "active",
   318→            }
   319→        })
   320→        self.current_pasal_lines = []
   321→

   326→
   327→class FamilyBParser:
   328→    """
   329→    Parses decree/decision documents: Menimbang > Mengingat > MEMUTUSKAN/MENETAPKAN > Diktum.
   330→    Handles: Keppres, Inpres, Tap_MPR, Kepmen.
   331→    """
   332→
   333→    RE_MENIMBANG = re.compile(r'^\s*menimbang\s*:?\s*$', re.IGNORECASE)
   334→    RE_MENGINGAT = re.compile(r'^\s*mengingat\s*:?\s*$', re.IGNORECASE)
   335→    RE_MEMUTUSKAN = re.compile(r'^\s*memutuskan\s*:?\s*$', re.IGNORECASE)
   336→    RE_MENETAPKAN = re.compile(r'^\s*menetapkan\s*:?\s*$', re.IGNORECASE)
   337→    RE_MENINSTRUKSIKAN = re.compile(r'^\s*menginstruksikan\s*:?\s*$', re.IGNORECASE)
   338→    RE_ITEM_LETTER = re.compile(r'^\s*([a-z])\s*\.\s+')
   339→    RE_ITEM_NUMBER = re.compile(r'^\s*(\d+)\s*\.\s+')
   340→
   341→    def __init__(self, lines, doc_type, nomor, year, issuer):
   342→        self.lines = lines
   343→        self.doc_type = doc_type
   344→        self.nomor = nomor
   345→        self.year = year
   346→        self.issuer = issuer
   347→        self.chunks = []
   348→
   349→        self.section = "pre"  # pre, menimbang, mengingat, keputusan, done
   350→        self.current_item = None
   351→        self.current_item_lines = []
   352→        self.preamble_lines = []
   353→        self.decision_keyword = None  # MEMUTUSKAN, MENETAPKAN, etc.
   354→
   355→    def parse(self) -> List[Dict]:
   356→        for line in self.lines:
   357→            text = line["text"].strip()
   358→            if not text:
   359→                continue
   360→
   361→            text_upper = text.upper()
   362→
   363→            # Section transitions
   364→            if self.RE_MENIMBANG.match(text):
   365→                self._flush_preamble()
   366→                self.section = "menimbang"
   367→                self.current_item = None
   368→                continue
   369→
   370→            if self.RE_MENGINGAT.match(text):
   371→                self._flush_preamble()
   372→                self.section = "mengingat"
   373→                self.current_item = None
   374→                continue
   375→
   376→            if self.RE_MEMUTUSKAN.match(text):
   377→                self._flush_item()
   378→                self.section = "keputusan"
   379→                self.decision_keyword = "MEMUTUSKAN"
   380→                self.current_item = None
   381→                continue
   382→
   383→            if self.RE_MENETAPKAN.match(text):
   384→                self._flush_item()
   385→                self.section = "keputusan"
   386→                self.decision_keyword = "MENETAPKAN"
   387→                self.current_item = None
   388→                continue
   389→
   390→            if self.RE_MENINSTRUKSIKAN.match(text):
   391→                self._flush_item()
   392→                self.section = "keputusan"
   393→                self.decision_keyword = "MENGINSTRUKSIKAN"
   394→                self.current_item = None
   395→                continue
   396→
   397→            # Check for judge signature block (end of substantive content)
   398→            text_lower = text.lower()
   399→            if "ttd" in text_lower and len(text) < 20:
   400→                self._flush_item()
   401→                self.section = "done"
   402→                continue
   403→
   404→            if self.section == "pre":
   405→                continue  # skip title block
   406→
   407→            if self.section == "done":
   408→                continue
   409→
   410→            # Check for new item in keputusan section
   411→            if self.section == "keputusan":
   412→                m_letter = self.RE_ITEM_LETTER.match(text)
   413→                m_number = self.RE_ITEM_NUMBER.match(text)
   414→
   415→                if m_letter or m_number:
   416→                    self._flush_item()
   417→                    self.current_item = m_letter.group(1) if m_letter else m_number.group(1)
   418→                    # Get text after the item marker
   419→                    m = m_letter or m_number
   420→                    remaining = text[m.end():].strip()
   421→                    if remaining:
   422→                        self.current_item_lines.append(remaining)
   423→                    continue
   424→
   425→                # Sub-items: continue accumulating under current item
   426→                if self.current_item:
   427→                    self.current_item_lines.append(text)
   428→                continue
   429→
   430→            # menimbang/mengingat: accumulate as preamble (skip for chunking or tag separately)
   431→            # Per the plan, optionally chunk these with section tag
   432→
   433→        self._flush_item()
   434→        return self.chunks
   435→
   436→    def _flush_item(self):
   437→        if not self.current_item or not self.current_item_lines:
   438→            self.current_item_lines = []
   439→            return
   440→
   441→        text = ' '.join(self.current_item_lines).strip()
   442→        if not text:
   443→            self.current_item_lines = []
   444→            return
   445→
   446→        kw = self.decision_keyword or "MEMUTUSKAN"
   447→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   448→                            chunk_type="diktum", chunk_num=self.current_item)
   449→
   450→        self.chunks.append({
   451→            "id": chunk_id,
   452→            "text": text,
   453→            "metadata": {
   454→                "doc_type": self.doc_type,
   455→                "issuer": self.issuer,
   456→                "year": self.year,
   457→                "nomor": self.nomor,
   458→                "section": "keputusan",
   459→                "decision_keyword": kw,
   460→                "diktum": self.current_item,
   461→                "parent": None,
   462→                "path": f"{kw} > Diktum {self.current_item}",
   463→                "status": "active",
   464→            }
   465→        })
   466→        self.current_item_lines = []
   467→
   468→    def _flush_preamble(self):
   469→        self.preamble_lines = []  # Could optionally save

   473→# FAMILY C: COURT RULING PARSER
   474→# ==========================================================================
   475→
   476→class FamilyCParser:
   477→    """
   478→    Parses court rulings: Duduk Perkara > Menimbang > MENGADILI/MENETAPKAN > Amar.
   479→    Handles: Putusan MK, Putusan MA.
   480→    """
   481→
   482→    RE_MENGADILI = re.compile(r'^\s*mengadili\s*:?\s*$', re.IGNORECASE)
   483→    RE_MENETAPKAN = re.compile(r'^\s*menetapkan\s*:?\s*$', re.IGNORECASE)
   484→    RE_AMAR = re.compile(r'^\s*(\d+)\s*\.\s+')
   485→    RE_PUTUSAN = re.compile(r'^\s*putusan\s*:?\s*$', re.IGNORECASE)
   486→
   487→    def __init__(self, lines, doc_type, nomor, year, issuer, nomor_perkara=None):
   488→        self.lines = lines
   489→        self.doc_type = doc_type
   490→        self.nomor = nomor
   491→        self.year = year
   492→        self.issuer = issuer
   493→        self.nomor_perkara = nomor_perkara
   494→        self.chunks = []
   495→
   496→        self.section = "pre"  # pre, menimbang, mengadili, amar, done
   497→        self.current_amar = None
   498→        self.current_amar_lines = []
   499→        self.pertimbangan_lines = []
   500→
   501→    def parse(self) -> List[Dict]:
   502→        for line in self.lines:
   503→            text = line["text"].strip()
   504→            if not text:
   505→                continue
   506→
   507→            text_upper = text.upper()
   508→
   509→            if self.RE_MENGADILI.match(text):
   510→                self._flush_amar()
   511→                self.section = "amar"
   512→                self.decision_keyword = "MENGADILI"
   513→                continue
   514→
   515→            if self.RE_MENETAPKAN.match(text):
   516→                self._flush_amar()
   517→                self.section = "amar"
   518→                self.decision_keyword = "MENETAPKAN"
   519→                continue
   520→
   521→            if self.section == "pre":
   522→                continue
   523→
   524→            if self.section == "done":
   525→                continue
   526→
   527→            if self.section == "amar":
   528→                m = self.RE_AMAR.match(text)
   529→                if m:
   530→                    self._flush_amar()
   531→                    self.current_amar = m.group(1)
   532→                    remaining = text[m.end():].strip()
   533→                    if remaining:
   534→                        self.current_amar_lines.append(remaining)
   535→                    continue
   536→
   537→                if self.current_amar:
   538→                    self.current_amar_lines.append(text)
   539→
   540→                # Detect end: signature block
   541→                if re.search(r'\bttd\b|\bHAKIM\b|\bKETUA\b', text_upper) and len(text) < 30:
   542→                    self._flush_amar()
   543→                    self.section = "done"
   544→
   545→        self._flush_amar()
   546→        return self.chunks
   547→
   548→    def _flush_amar(self):
   549→        if not self.current_amar or not self.current_amar_lines:
   550→            self.current_amar_lines = []
   551→            return
   552→
   553→        text = ' '.join(self.current_amar_lines).strip()
   554→        if not text:
   555→            self.current_amar_lines = []
   556→            return
   557→
   558→        kw = getattr(self, 'decision_keyword', 'MENGADILI')
   559→        chunk_id = build_id(self.doc_type, self.nomor, self.year,
   560→                            nomor_perkara=self.nomor_perkara,
   561→                            chunk_type="amar", chunk_num=self.current_amar)
   562→
   563→        self.chunks.append({
   564→            "id": chunk_id,
   565→            "text": text,
   566→            "metadata": {
   567→                "doc_type": self.doc_type,
   568→                "issuer": self.issuer,
   569→                "year": self.year,
   570→                "nomor_perkara": self.nomor_perkara,
   571→                "section": "amar",
   572→                "decision_keyword": kw,
   573→                "amar": self.current_amar,
   574→                "parent": None,
   575→                "path": f"{kw} > Amar {self.current_amar}",
   576→                "status": "active",
   577→            }
   578→        })
   579→        self.current_amar_lines = []

   586→def extract_nomor_perkara(lines):
   587→    """Extract case number (e.g. 96/PUU-XVI/2018) from court ruling lines."""
   588→    for line in lines[:30]:
   589→        m = re.search(r'(\d+/[A-Z]+-[A-Z]+/\d{4})', line["text"])
   590→        if m:
   591→            return m.group(1)
   592→    return None

   595→def chunk_pdf(pdf_path: str, directory: str, family_map: Dict) -> List[Dict]:
   596→    """
   597→    Main entry point. Extract + parse a PDF into chunks.
   598→    Returns list of chunk dicts with id, text, metadata.
   599→    """
   600→    # Get family config
   601→    dir_config = family_map.get(directory, {})
   602→    family = dir_config.get("family", "A")
   603→    default_doc_type = dir_config.get("doc_type", "Unknown")
   604→    default_issuer = dir_config.get("issuer", "Unknown")
   605→
   606→    # Extract clean text
   607→    try:
   608→        lines, meta = extract_clean(pdf_path)
   609→    except Exception as e:
   610→        return [{"id": f"ERROR:{directory}", "text": str(e), "metadata": {"error": True}}]
   611→
   612→    if not lines:
   613→        return [{"id": f"EMPTY:{directory}", "text": "No extractable text (possibly scanned PDF)", "metadata": {"error": True, "needs_ocr": True}}]
   614→
   615→    # Detect doc type from title block (override directory default)
   616→    title_block = extract_title_block(lines)
   617→    detected_type, detected_issuer = detect_doc_type_from_title(title_block)
   618→    doc_type = detected_type if detected_type != "Unknown" else default_doc_type
   619→    issuer = detected_issuer if detected_issuer != "Unknown" else default_issuer
   620→    nomor = meta.get("nomor")
   621→    year = meta.get("year")
   622→
   623→    # Parse based on family
   624→    if family == "A":
   625→        parser = FamilyAParser(lines, doc_type, nomor, year, issuer)
   626→        chunks = parser.parse()
   627→    elif family == "B":
   628→        parser = FamilyBParser(lines, doc_type, nomor, year, issuer)
   629→        chunks = parser.parse()
   630→    elif family == "C":
   631→        nomor_perkara = extract_nomor_perkara(lines)
   632→        parser = FamilyCParser(lines, doc_type, nomor, year, issuer, nomor_perkara)
   633→        chunks = parser.parse()
   634→    else:
   635→        chunks = [{"id": f"UNKNOWN_FAMILY:{directory}", "text": "", "metadata": {"error": True}}]
   636→
   637→    # Add extraction metadata to all chunks
   638→    for chunk in chunks:
   639→        chunk["metadata"]["source_file"] = meta["file"]
   640→        chunk["metadata"]["directory"] = directory
   641→        chunk["metadata"]["family"] = family
   642→        if nomor:
   643→            chunk["metadata"]["nomor"] = nomor
   644→        if year:
   645→            chunk["metadata"]["year"] = year
   646→
   647→    return chunks
